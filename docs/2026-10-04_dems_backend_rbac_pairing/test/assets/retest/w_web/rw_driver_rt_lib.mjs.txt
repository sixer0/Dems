// - OS-assigned DevTools port (--remote-debugging-port=0), read from the profile's DevToolsActivePort; the listening
//   socket is checked to belong to the PID this process spawned (lsof) before any connection.
// - Fresh profile under $TMPDIR (/tmp/claude-501), removed on close; the browser is killed on any exit of this process.
// - Fetch interception fails every request whose origin is not http://127.0.0.1:5178 (data:/blob: allowed, not network).
// - The demo password is read from README (Quick start table) inside this process and only handed to Input.insertText.

function readPassword(username) {
  const re = new RegExp('^\\|\\s*`' + username.replace(/\./g, '\\.') + '`\\s*\\|')
  const row = readFileSync(README, 'utf8').split('\n').find((line) => re.test(line))
  if (!row) throw new Error(`README row for ${username} not found`)
  const value = (row.split('|').map((cell) => cell.trim())[4] ?? '').replace(/`/g, '')
  if (!value) throw new Error('README password cell empty')
  return value
}
const SECRETS = [...new Set(['andi.pratama', 'budi.santoso', 'rina.wijaya'].map(readPassword))]

export function redactString(input) {
  let t = String(input)
  for (const secret of SECRETS) t = t.split(secret).join('***')
  t = t.replace(/(?<![0-9A-Za-z])\d{6}(?![0-9A-Za-z])/g, '######')
  t = t.replace(/(?<![0-9A-Za-z])\d{3}[\s   .\-–]\d{3}(?![0-9A-Za-z])/g, '### ###')
  t = t.replace(/\b(ses|dvc|bio)_[A-Za-z0-9_-]+/g, '$1_***')
  t = t.replace(/token=[^&\s"]+/g, 'token=***')
  return t
}
export function redact(value) {
  if (typeof value === 'string') return redactString(value)
  if (Array.isArray(value)) return value.map(redact)
  if (value && typeof value === 'object') return Object.fromEntries(Object.entries(value).map(([k, v]) => [k, redact(v)]))
  return value
}

/** Run context: log lines and observations go to ASSETS/<run>_log.txt and <run>_observations.json (redacted). */
export function makeRun(run) {
  const lines = []
  const obs = {}
  const log = (line) => {
    const stamped = `${new Date().toISOString()} ${redactString(line)}`
    lines.push(stamped)
    console.log(stamped)
  }
  return {
    run,
    log,
    obs,
    record(key, value) {
      obs[key] = redact(value)
      log(`${key}: ${JSON.stringify(obs[key]).slice(0, 1500)}`)
    },
    flush() {
      writeFileSync(join(ASSETS, `${run}_log.txt`), `${lines.join('\n')}\n`)
      writeFileSync(join(ASSETS, `${run}_observations.json`), `${JSON.stringify(redact(obs), null, 1)}\n`)
    },
  }
}

export async function launch(log) {
  const profile = mkdtempSync(join(PROFILE_BASE, 'rw_profile_'))
  const child = spawn(
    SHELL,
    ['--headless', '--remote-debugging-port=0', `--user-data-dir=${profile}`, '--no-first-run', '--no-default-browser-check',
      '--disable-background-networking', '--disable-component-update', '--disable-sync', '--hide-scrollbars', '--mute-audio',
      '--force-color-profile=srgb', 'about:blank'],
    { stdio: ['ignore', 'ignore', 'ignore'] },
  )
  const kill = () => { try { child.kill('SIGKILL') } catch { /* gone */ } }
  process.on('exit', () => { kill(); try { rmSync(profile, { recursive: true, force: true }) } catch { /* gone */ } })
  for (const sig of ['SIGINT', 'SIGTERM']) process.on(sig, () => process.exit(130))
  const portFile = join(profile, 'DevToolsActivePort')
  for (let i = 0; i < 150 && !existsSync(portFile); i += 1) await sleep(100)
  if (!existsSync(portFile)) throw new Error('DevToolsActivePort not written')
  const [portLine, wsPath] = readFileSync(portFile, 'utf8').trim().split('\n')
  const port = Number(portLine)
  const owner = execFileSync('lsof', ['-nP', '-a', '-p', String(child.pid), `-iTCP:${port}`, '-sTCP:LISTEN'], { encoding: 'utf8' })
  if (!owner.includes(String(child.pid))) throw new Error('CDP port is not owned by the spawned browser')
  const version = await (await fetch(`http://127.0.0.1:${port}/json/version`)).json()
  if (!String(version.webSocketDebuggerUrl).endsWith(wsPath)) throw new Error('DevTools endpoint does not name the spawned browser')
  log(`browser pid ${child.pid}, ${version.Browser}, CDP port ${port} (OS-assigned), owner verified (lsof + /json/version), profile ${profile.split('/').pop()}`)
  const cdp = await Cdp.connect(`ws://127.0.0.1:${port}${wsPath}`)
  return {
    cdp,
    pid: child.pid,
    browser: version.Browser,
    async close() {
      try { await cdp.send('Browser.close') } catch { /* closing */ }
      await sleep(600)
      kill()
      rmSync(profile, { recursive: true, force: true })
      log(`browser closed (pid ${child.pid}), profile removed: ${!existsSync(profile)}`)
    },
  }
}

export class Cdp {
  static connect(url) {
    return new Promise((resolve, reject) => {
      const ws = new WebSocket(url)
      ws.onopen = () => resolve(new Cdp(ws))
      ws.onerror = (event) => reject(new Error(`ws error ${event.message ?? ''}`))
    })
  }
  constructor(ws) {
    this.ws = ws
    this.id = 0
    this.pending = new Map()
    this.listeners = new Set()
    ws.onmessage = (event) => {
      const message = JSON.parse(event.data)
      if (message.id !== undefined) {
        const entry = this.pending.get(message.id)
        if (!entry) return
        this.pending.delete(message.id)
        if (message.error) entry.reject(new Error(`${entry.method}: ${message.error.message}`))
        else entry.resolve(message.result)
      } else for (const listener of this.listeners) listener(message)
    }
  }
  send(method, params = {}, sessionId) {
    const id = ++this.id
    return new Promise((resolve, reject) => {
      this.pending.set(id, { resolve, reject, method })
      this.ws.send(JSON.stringify({ id, method, params, ...(sessionId ? { sessionId } : {}) }))
    })
  }
  on(listener) {
    this.listeners.add(listener)
    return () => this.listeners.delete(listener)
  }
}

const q = (sel) => (/[\[\].#\s>:,=]/.test(sel) ? sel : `[data-testid="${sel}"]`)

export async function openPage(cdp, log) {
  const { targetId } = await cdp.send('Target.createTarget', { url: 'about:blank' })
  const { sessionId } = await cdp.send('Target.attachToTarget', { targetId, flatten: true })
  const s = (method, params) => cdp.send(method, params, sessionId)
  // Per-page-label event record: requests (origin + path, no query values), ws, >=400 responses, console errors and
  // warnings, exceptions, Log entries, blocked other-origin requests. Bodies, cookies and frame payloads are never read.
  const events = []
  const state = { label: 'start' }
  const reqs = new Map()
  cdp.on((message) => {
    if (message.sessionId !== sessionId) return
    const p = message.params
    const push = (kind, data) => events.push({ t: Date.now(), label: state.label, kind, ...redact(data) })
    switch (message.method) {
      case 'Runtime.exceptionThrown':
        push('exception', { text: String(p.exceptionDetails?.exception?.description ?? p.exceptionDetails?.text ?? '').split('\n')[0].slice(0, 240) })
        break
      case 'Runtime.consoleAPICalled':
        if (p.type === 'error' || p.type === 'warning' || p.type === 'assert')
          push('console', { level: p.type, text: (p.args ?? []).map((a) => a.value ?? a.description ?? '').join(' ').slice(0, 300) })
        break
      case 'Log.entryAdded':
        if (p.entry.level === 'error' || p.entry.level === 'warning') push('log', { level: p.entry.level, source: p.entry.source, text: String(p.entry.text).slice(0, 240), url: p.entry.url ? p.entry.url.split('?')[0] : undefined })
        break
      case 'Network.requestWillBeSent': {
        let origin = 'unparsable'
        let path = ''
        try { const u = new URL(p.request.url); origin = u.protocol === 'data:' || u.protocol === 'blob:' ? u.protocol : u.origin; path = u.protocol === 'data:' ? '' : u.pathname } catch { /* keep */ }
        reqs.set(p.requestId, { method: p.request.method, path })
        push('request', { origin, method: p.request.method, path })
        break
      }
      case 'Network.responseReceived':
        if (p.response.status >= 400) {
          const r = reqs.get(p.requestId) ?? {}
          push('response4xx', { status: p.response.status, method: r.method, path: r.path })
        }
        break
      case 'Network.webSocketCreated': {
        let origin = 'unparsable'
        let path = ''
        try { const u = new URL(p.url); origin = u.origin; path = u.pathname } catch { /* keep */ }
        push('ws', { origin, path })
        break
      }
      case 'Fetch.requestPaused': {
        const url = p.request.url
        let allowed = false
        try { const u = new URL(url); allowed = u.origin === ORIGIN || u.protocol === 'data:' || u.protocol === 'blob:' } catch { allowed = false }
        if (allowed) s('Fetch.continueRequest', { requestId: p.requestId }).catch(() => undefined)
        else {
          push('blocked', { url: url.split('?')[0].slice(0, 100) })
          s('Fetch.failRequest', { requestId: p.requestId, errorReason: 'BlockedByClient' }).catch(() => undefined)
        }
        break
      }
      default:
    }
  })
  await s('Page.enable')
  await s('Runtime.enable')
  await s('Network.enable')
  await s('Log.enable')
  await s('Fetch.enable', { patterns: [{ urlPattern: '*' }] })

  const page = {
    events,
    send: s,
    setLabel(label) { state.label = label; log(`== page: ${label}`) },
    get label() { return state.label },
    async viewport(width, height, mobile = false) {
      await s('Emulation.setDeviceMetricsOverride', { width, height, deviceScaleFactor: 1, mobile })
      log(`viewport ${width}x${height}`)
    },
    async goto(path, settle = 1500) {
      await s('Page.navigate', { url: `${ORIGIN}${path}` })
      await sleep(300)
      await page.waitForFn('document.readyState === "complete"', 20000)
      await sleep(settle)
    },
    async eval(expression, label = 'eval') {
      const result = await s('Runtime.evaluate', { expression, awaitPromise: true, returnByValue: true })
      if (result.exceptionDetails) throw new Error(`page script failed: ${label}`)
      return result.result.value
    },
    async waitForFn(expression, timeout = 15000) {
      const until = Date.now() + timeout
      while (Date.now() < until) {
        try { if (await page.eval(expression, 'waitForFn')) return true } catch { /* navigating */ }
        await sleep(150)
      }
      return false
    },
    async exists(sel) {
      return page.eval(`document.querySelector(${JSON.stringify(q(sel))}) !== null`, `exists ${sel}`).catch(() => false)
    },
    async waitFor(sel, present = true, timeout = 15000) {
      const until = Date.now() + timeout
      while (Date.now() < until) {
        if ((await page.exists(sel)) === present) return true
        await sleep(150)
      }
      return false
    },
    async waitText(needle, timeout = 10000, scope = 'body') {
      const until = Date.now() + timeout
      while (Date.now() < until) {
        const found = await page.eval(`(document.querySelector(${JSON.stringify(scope)})?.innerText ?? '').includes(${JSON.stringify(needle)})`, 'waitText').catch(() => false)
        if (found) return true
        await sleep(150)
      }
      return false
    },
    async text(sel, max = 600) {
      const t = await page.eval(`(() => { const el = document.querySelector(${JSON.stringify(q(sel))}); return el ? String(el.innerText ?? el.textContent ?? '').replace(/\\s+/g, ' ').trim() : null })()`, `text ${sel}`)
      return t === null ? null : redactString(t.slice(0, max))
    },
    async attr(sel, name) {
      return page.eval(`document.querySelector(${JSON.stringify(q(sel))})?.getAttribute(${JSON.stringify(name)}) ?? null`, `attr ${sel}`)
    },
    /** Real mouse click at the element's centre once its position is stable; reports whether the hit target is the element. */
    async click(sel, opts = {}) {
      const S = JSON.stringify(q(sel))
      const box = await page.eval(`(() => { const el = document.querySelector(${S}); if (!el) return null; el.scrollIntoView({ block: 'center', inline: 'nearest', behavior: 'instant' }); return { disabled: el.disabled === true || el.getAttribute('aria-disabled') === 'true' || el.hasAttribute('data-disabled') } })()`, `box ${sel}`)
      if (!box) return { found: false }
      const measure = () => page.eval(`(() => { const el = document.querySelector(${S}); if (!el) return null; const r = el.getBoundingClientRect(); return { x: Math.round(r.left + r.width / 2), y: Math.round(r.top + r.height / 2) } })()`, `box2 ${sel}`)
      let prev = await measure()
      for (let i = 0; i < 30; i += 1) {
        await sleep(100)
        const cur = await measure()
        if (cur && prev && cur.x === prev.x && cur.y === prev.y) break
        prev = cur
      }
      if (!prev) return { found: false }
      const { x, y } = prev
      const hit = await page.eval(`(() => { const el = document.querySelector(${S}); const top = document.elementFromPoint(${x}, ${y}); return !!(el && top && (el === top || el.contains(top) || top.contains(el))) })()`, `hit ${sel}`)
      await s('Input.dispatchMouseEvent', { type: 'mouseMoved', x, y })
      await s('Input.dispatchMouseEvent', { type: 'mousePressed', x, y, button: 'left', clickCount: 1 })
      await sleep(opts.hold ?? 60)
      await s('Input.dispatchMouseEvent', { type: 'mouseReleased', x, y, button: 'left', clickCount: 1 })
      await sleep(opts.after ?? 400)
      return { found: true, disabled: box.disabled, hit }
    },
    async clickByText(cssSel, needle, opts = {}) {
      const marked = await page.eval(`(() => { const want = ${JSON.stringify(needle)}; const el = [...document.querySelectorAll(${JSON.stringify(cssSel)})].find((e) => { const t = (e.innerText ?? e.textContent ?? '').replace(/\\s+/g, ' ').trim(); return (${opts.exact ? 't === want' : 't.includes(want)'}) && e.getClientRects().length > 0 }); if (!el) return false; document.querySelectorAll('[data-rw-target]').forEach((e) => e.removeAttribute('data-rw-target')); el.setAttribute('data-rw-target', '1'); return true })()`, `mark ${needle}`)
      if (!marked) return { found: false }
      return page.click('[data-rw-target="1"]', opts)
    },
    async hover(sel) {
      const box = await page.eval(`(() => { const el = document.querySelector(${JSON.stringify(q(sel))}); if (!el) return null; const r = el.getBoundingClientRect(); return { x: Math.round(r.left + r.width / 2), y: Math.round(r.top + r.height / 2) } })()`, `hover ${sel}`)
      if (!box) return false
      await s('Input.dispatchMouseEvent', { type: 'mouseMoved', x: box.x, y: box.y })
      return true
    },
    async key(keyName) {
      const map = { Enter: 13, Escape: 27, ArrowDown: 40, ArrowUp: 38, Tab: 9, Backspace: 8 }
      const vk = map[keyName] ?? 0
      await s('Input.dispatchKeyEvent', { type: 'keyDown', key: keyName, code: keyName, windowsVirtualKeyCode: vk, text: keyName === 'Enter' ? '\r' : undefined })
      await s('Input.dispatchKeyEvent', { type: 'keyUp', key: keyName, code: keyName, windowsVirtualKeyCode: vk })
      await sleep(150)
    },
    async type(sel, value, { clear = true } = {}) {
      const ok = await page.eval(`(() => { let el = document.querySelector(${JSON.stringify(q(sel))}); if (!el) return false; if (!['INPUT','TEXTAREA'].includes(el.tagName)) el = el.querySelector('input, textarea'); if (!el) return false; el.scrollIntoView({ block: 'center' }); el.focus(); if (${clear}) el.select?.(); return document.activeElement === el })()`, `focus ${sel}`)
      if (!ok) throw new Error(`cannot focus ${sel}`)
      if (clear) {
        await s('Input.dispatchKeyEvent', { type: 'keyDown', key: 'Backspace', code: 'Backspace', windowsVirtualKeyCode: 8 })
        await s('Input.dispatchKeyEvent', { type: 'keyUp', key: 'Backspace', code: 'Backspace', windowsVirtualKeyCode: 8 })
      }
      if (value) await s('Input.insertText', { text: value })
      await sleep(200)
    },
    async typePasswordOf(sel, username) {
      return page.type(sel, readPassword(username))
    },
    /** Blurs any pairing code before a capture: the code and QR elements, and text nodes that are only a 6-digit code. */
    async mask() {
      return page.eval(`(() => {
        let style = document.getElementById('rw-mask')
        if (!style) { style = document.createElement('style'); style.id = 'rw-mask'; document.head.appendChild(style) }
        style.textContent = '[data-testid="pd-code"], [data-testid="pd-qr"], [data-rw-mask] { filter: blur(18px) !important; }'
        let n = 0
        const walker = document.createTreeWalker(document.body, NodeFilter.SHOW_TEXT)
        while (walker.nextNode()) {
          const t = (walker.currentNode.nodeValue ?? '').trim()
          if (/^\\d{3}[\\s\\u00a0\\u2009\\u202f.\\-–]?\\d{3}$/.test(t) && walker.currentNode.parentElement) { walker.currentNode.parentElement.setAttribute('data-rw-mask', '1'); n++ }
        }
        const code = document.querySelector('[data-testid="pd-code"]')
        return { masked: n, codeShown: !!code, codeFilter: code ? getComputedStyle(code).filter : null }
      })()`, 'mask')
    },
    async shot(name, clip) {
      const m = await page.mask()
      if (m.codeShown && !/blur/.test(String(m.codeFilter))) throw new Error('pairing code element not blurred: capture refused')
      const params = { format: 'png', captureBeyondViewport: false }
      if (clip) params.clip = { ...clip, scale: 1 }
      const { data } = await s('Page.captureScreenshot', params)
      writeFileSync(join(ASSETS, name), Buffer.from(data, 'base64'))
      log(`screenshot ${name}${m.masked || m.codeShown ? ` (masked ${m.masked}, code element ${m.codeShown})` : ''}`)
      return name
    },
    async fetchJson(path, project = '(b) => b', method = 'GET') {
      const res = await page.eval(`(async () => { const r = await fetch(${JSON.stringify(path)}, { method: ${JSON.stringify(method)}, credentials: 'same-origin', headers: { accept: 'application/json' } }); let b = null; try { b = await r.json() } catch {} let p = null; try { p = (${project})(b) } catch (e) { p = '(projection failed)' } return { status: r.status, body: p } })()`, `fetch ${path}`)
      return redact(res)
    },
    async auditTail(limit = 5, extra = '') {
      return page.fetchJson(`/api/v1/audit?order=DESC&limit=${limit}${extra}`, `(b) => ({ total: b?.total, rows: (b?.rows ?? []).map((r) => ({ seq: r.seq, at: new Date(r.at).toISOString(), type: r.type, severity: r.severity, actorId: r.actorId, sessionId: r.sessionId, evidenceId: r.evidenceId, deviceId: r.deviceId, message: String(r.message).slice(0, 160) })) })`)
    },
    async facts() {
      return redact(await page.eval(`(() => ({
        href: location.href.replace(location.origin, ''),
        title: document.title,
        page: [...document.querySelectorAll('[data-testid^="page-"]')].map((e) => e.getAttribute('data-testid')),
        mode: document.querySelector('[data-testid="shell-mode"]')?.innerText?.replace(/\\s+/g, ' ').trim() ?? null,
        h1: document.querySelector('h1')?.innerText?.trim() ?? null,
        toasts: [...document.querySelectorAll('[data-slot="toast"]')].map((e) => e.innerText.replace(/\\s+/g, ' ').trim().slice(0, 200)),
        dialogs: [...document.querySelectorAll('[role="dialog"], [role="alertdialog"]')].map((e) => (e.getAttribute('data-testid') ?? '') + ': ' + e.innerText.replace(/\\s+/g, ' ').trim().slice(0, 240)),
        alerts: [...document.querySelectorAll('[role="alert"]')].map((e) => e.innerText.replace(/\\s+/g, ' ').trim().slice(0, 200)).filter(Boolean),
        scrollWidth: document.documentElement.scrollWidth, clientWidth: document.documentElement.clientWidth,
      }))()`, 'facts'))
    },
    async menuItems() {
      return redact(await page.eval(`[...document.querySelectorAll('[role="menu"] [role="menuitem"]')].map((e) => ({ text: e.innerText.replace(/\\s+/g, ' ').trim().slice(0, 120), disabled: e.getAttribute('aria-disabled') === 'true' || e.hasAttribute('data-disabled') }))`, 'menu'))
    },
    async elementClip(sel, pad = 10, maxH = 900) {
      return page.eval(`(() => { const el = document.querySelector(${JSON.stringify(q(sel))}); if (!el) return null; const b = el.getBoundingClientRect(); return { x: Math.max(0, b.left - ${pad}), y: Math.max(0, b.top - ${pad}), width: Math.min(document.documentElement.clientWidth, b.width + ${pad * 2}), height: Math.min(${maxH}, b.height + ${pad * 2}) } })()`, 'clip')
    },
  }
  return page
}

/** Signs a demo operator in through the Login form (password never echoed). */
export async function signIn(page, username, log) {
  if (!(await page.exists('page-login'))) await page.goto('/login')
  await page.waitFor('[data-testid="login-username-field"] input', true, 20000)
  await page.waitForFn(`(() => { const b = document.querySelector('[data-testid="login-submit"]'); return !!b && !b.hasAttribute('disabled') })()`, 20000)
  await page.type('login-username-field', username)
  await page.typePasswordOf('login-password-field', username)
  await page.click('login-submit')
  const ok = await page.waitFor('page-dashboard', true, 25000)
  await sleep(2500)
  log(`sign-in ${username}: ${ok ? 'landed on /dashboard' : 'FAILED to reach the dashboard'}`)
  return ok
}

/** Sign out through the user menu; returns the login state. */
export async function signOutMenu(page, log, shotName) {
  await page.click('user-menu')
  await sleep(700)
  const items = await page.menuItems()
  const res = await page.clickByText('[role="menu"] [role="menuitem"]', 'Sign out')
  const login = await page.waitFor('page-login', true, 15000)
  await sleep(1200)
  const out = { menu: items, clicked: res.found, login, state: await page.attr('page-login', 'data-state'), alert: await page.text('login-alert', 200) }
  if (shotName) out.shot = await page.shot(shotName)
  log(`signed out via user menu: ${JSON.stringify(out)}`)
  return out
}

/** Per-label summary of console and network facts. */
export function summarizeEvents(events) {
  const by = {}
  for (const e of events) {
    const b = (by[e.label] ??= { requests: 0, origins: {}, ws: [], errors: [], warnings: [], exceptions: [], log: [], responses4xx: [], blocked: [] })
    if (e.kind === 'request') { b.requests += 1; b.origins[e.origin] = (b.origins[e.origin] ?? 0) + 1 }
    if (e.kind === 'ws') b.ws.push(`${e.origin}${e.path}`)
    if (e.kind === 'console') (e.level === 'error' || e.level === 'assert' ? b.errors : b.warnings).push(e.text)
    if (e.kind === 'exception') b.exceptions.push(e.text)
    if (e.kind === 'log') b.log.push(`${e.level} ${e.source}: ${e.text}${e.url ? ` (${e.url.replace(ORIGIN, '')})` : ''}`)
    if (e.kind === 'response4xx') b.responses4xx.push(`${e.status} ${e.method} ${e.path}`)
    if (e.kind === 'blocked') b.blocked.push(e.url)
  }
  return by
}

export function writeEvents(run, events) {
  writeFileSync(join(ASSETS, `${run}_cdp_events.jsonl`), events.map((e) => JSON.stringify(redact(e))).join('\n') + '\n')
  const summary = summarizeEvents(events)
  writeFileSync(join(ASSETS, `${run}_console_network.json`), `${JSON.stringify(redact(summary), null, 1)}\n`)
  return summary
}
