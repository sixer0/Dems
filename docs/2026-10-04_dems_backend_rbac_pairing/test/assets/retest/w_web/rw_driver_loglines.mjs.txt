// Request lines of the lane-3 Core log in a time window: time (UTC), method, route, status, principal. Nothing else.
// Usage: node loglines.mjs <fromISO> <toISO> [routeRegex]
import { readFileSync } from 'node:fs'
const [from, to, re] = process.argv.slice(2)
const a = Date.parse(from), b = Date.parse(to)
const rx = re ? new RegExp(re) : null
for (const line of readFileSync('/Users/gsp-budi/Documents/DEMS/.cache/manual/lane3-server.log', 'utf8').split('\n')) {
  if (!line.startsWith('{')) continue
  let j; try { j = JSON.parse(line) } catch { continue }
  if (j.msg !== 'request completed' || j.time < a || j.time > b) continue
  const principal = j.principal ? (typeof j.principal === 'string' ? j.principal : `${j.principal.kind ?? ''}:${j.principal.operatorId ?? j.principal.id ?? ''}`) : '-'
  const out = `${new Date(j.time).toISOString()} ${j.method} ${j.route} ${j.status} ${principal}`
  if (!rx || rx.test(out)) console.log(out.replace(/(ses|dvc|bio)_[A-Za-z0-9_-]+/g, '$1_***'))
}
