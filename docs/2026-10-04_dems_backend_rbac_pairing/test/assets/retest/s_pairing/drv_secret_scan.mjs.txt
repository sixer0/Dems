// R-S final scan of every output file (adapted from round 1's drv_10_secret_scan.mjs.txt):
// (1) the demo password (from the seed constant; never printed), counted in every file including images and
//     video (latin1 bytes);
// (2) digit-bounded 6-digit numbers in the text files, each classified with its context so it can be explained;
// (3) six digits separated by single spaces, dots, commas or dashes (a code written "spaced") in the text files.
// The pairing code itself is no longer held by any process (the issue-and-redeem driver counted it in every file
// present at 13:01:36 and then dropped it); this scan looks for anything code-shaped.
//   node --import tsx secret_scan.mjs <file or folder>...
import { readFileSync, readdirSync, statSync } from 'node:fs'
import { join } from 'node:path'
import { DEMO_OPERATOR_PASSWORD } from '/Users/gsp-budi/Documents/DEMS/packages/shared/src/sim/dataset.ts'

const files = []
for (const p of process.argv.slice(2)) {
  if (statSync(p).isDirectory()) for (const n of readdirSync(p).sort()) { const f = join(p, n); if (statSync(f).isFile()) files.push(f) }
  else files.push(p)
}
const binary = /\.(png|mp4)$/i
let passwordTotal = 0
const hits = []
const spacedHits = []
for (const f of files) {
  const t = readFileSync(f).toString('latin1')
  const pw = t.split(DEMO_OPERATOR_PASSWORD).length - 1
  passwordTotal += pw
  if (pw) console.log(`PASSWORD FOUND in ${f}: ${pw}`)
  if (binary.test(f)) continue
  const re = /(?<![0-9])[0-9]{6}(?![0-9])/g
  let m
  while ((m = re.exec(t)) !== null) hits.push({ file: f.split('/').pop(), match: m[0], before: t.slice(Math.max(0, m.index - 24), m.index), after: t.slice(m.index + 6, m.index + 16) })
  const sp = /(?<![0-9])[0-9](?:[ .,\-][0-9]){5}(?![0-9])/g
  while ((m = sp.exec(t)) !== null) spacedHits.push({ file: f.split('/').pop(), context: t.slice(Math.max(0, m.index - 30), m.index + 40).replace(/\s+/g, ' ') })
}
console.log(`files scanned: ${files.length} (${files.filter((f) => binary.test(f)).length} PNG/MP4 counted for the password only); password occurrences: ${passwordTotal}`)
const kindOf = (h) => {
  const b = h.before, a = h.after
  if (/AUD-$/.test(b)) return 'audit row ID (AUD-NNNNNN)'
  if (/PCD-$/.test(b)) return 'pairing-code ID (PCD-NNNNNN; an ID, not the code)'
  if (/"bytes": ?$/.test(b) || /^ bytes/.test(a) || /\($/.test(b) && /^ bytes/.test(a)) return 'byte size of a capture file'
  if (/\[1006\/$/.test(b) || /\[1006\/[0-9]{6}\.$/.test(b)) return 'Chromium log timestamp (HHMMSS / microseconds)'
  if (/[0-9a-f]-?$/i.test(b) && /^-?[0-9a-f]/i.test(a) || /[0-9a-f]{2}-$/i.test(b) || /^-[0-9a-f]{4}/i.test(a) || /^[0-9a-f]+-/i.test(a)) return 'fragment of a request UUID or socket connection ID'
  if (/epoch 1791\d*\.?$/.test(b) || /1791$/.test(b) || /^\d*\.\d{3}/.test(a) && /1791\d*$/.test(b)) return 'part of an epoch timestamp'
  if (/expiresAt"?: ?1791$/.test(b)) return 'part of an epoch timestamp'
  if (/(width|height|size|\d)x$/.test(b) || /^x\d/.test(a)) return 'image size'
  if (/R-S-\d$/.test(b)) return 'defect ID'
  return 'OTHER'
}
const byKind = {}
for (const h of hits) { const k = kindOf(h); byKind[k] = (byKind[k] ?? 0) + 1; if (k === 'OTHER') console.log(`  OTHER ${h.file}: "…${h.before.replace(/\s+/g, ' ')}[6 digits]${h.after.replace(/\s+/g, ' ')}…"`) }
console.log(`digit-bounded 6-digit numbers in text files: ${hits.length}; classified: ${JSON.stringify(byKind)}`)
console.log(`six digits separated by single spaces/dots/commas/dashes in text files: ${spacedHits.length}`)
for (const s of spacedHits) console.log(`  SPACED ${s.file}: "${s.context}"`)
