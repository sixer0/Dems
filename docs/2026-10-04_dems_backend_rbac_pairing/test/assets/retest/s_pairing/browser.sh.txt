#!/bin/sh
# R-S re-test: this row's OWN headless shell (Playwright cache chromium_headless_shell-1243), raw CDP on an
# OS-assigned port (--remote-debugging-port=0, read back from DevToolsActivePort), fresh profile in $TMPDIR.
#   sh browser.sh start | check | stop
set -u
SP=/private/tmp/claude-501/-Users-gsp-budi-Documents-DEMS/0d71fd65-4a51-4019-9c48-5c92d11bbdaf/scratchpad/rs
OUT=/Users/gsp-budi/Documents/DEMS/docs/2026-10-04_dems_backend_rbac_pairing/test/assets/retest/s_pairing
BIN="$HOME/Library/Caches/ms-playwright/chromium_headless_shell-1243/chrome-headless-shell-mac-arm64/chrome-headless-shell"
TMPBASE=${TMPDIR:-/tmp/claude-501}
case "$1" in
start)
  PROFILE="$TMPBASE/rs_pairing_profile_$(date +%H%M%S)"
  mkdir -p "$PROFILE"
  echo "$PROFILE" > "$SP/profile_dir"
  nohup "$BIN" --remote-debugging-port=0 --remote-debugging-address=127.0.0.1 --user-data-dir="$PROFILE" \
    --no-first-run --no-default-browser-check --window-size=1920,1080 about:blank > "$OUT/chrome.log" 2>&1 &
  PID=$!
  echo "$PID" > "$SP/browser_pid"
  i=0; until [ -s "$PROFILE/DevToolsActivePort" ] || [ $i -ge 100 ]; do i=$((i + 1)); sleep 0.2; done
  PORT=$(head -n 1 "$PROFILE/DevToolsActivePort")
  echo "$PORT" > "$SP/cdp_port"
  echo "-- started pid $PID; profile $PROFILE; CDP port $PORT (OS-assigned)"
  ps -p "$PID" -o pid=,command= | cut -c1-260
  ;;
check)
  PID=$(cat "$SP/browser_pid"); PROFILE=$(cat "$SP/profile_dir"); PORT=$(cat "$SP/cdp_port")
  CMD=$(ps -p "$PID" -o command= 2>/dev/null)
  case "$CMD" in *"$PROFILE"*) echo "-- pid $PID is this row's browser (profile $PROFILE), CDP port $PORT";; *) echo "-- pid $PID is NOT this row's browser: '$CMD'"; exit 3;; esac
  lsof -nP -a -p "$PID" -iTCP -sTCP:LISTEN 2>/dev/null | tail -n +2 | awk '{print "   listening:", $9}'
  ;;
stop)
  PID=$(cat "$SP/browser_pid"); PROFILE=$(cat "$SP/profile_dir")
  CMD=$(ps -p "$PID" -o command= 2>/dev/null)
  case "$CMD" in
    *"$PROFILE"*) kill "$PID"; sleep 2; ps -p "$PID" >/dev/null 2>&1 && kill -9 "$PID"; echo "-- stopped pid $PID";;
    "") echo "-- pid $PID already gone";;
    *) echo "-- pid $PID is not this row's browser; nothing killed"; exit 3;;
  esac
  LEFT=$(ps -axo pid=,command= | grep -F "$PROFILE" | grep -v grep | awk '{print $1}')
  [ -n "$LEFT" ] && { echo "-- helper processes of this profile still running: $LEFT; stopping them"; kill $LEFT 2>/dev/null; }
  rm -rf "$PROFILE" && echo "-- profile removed: $PROFILE"
  ;;
esac
