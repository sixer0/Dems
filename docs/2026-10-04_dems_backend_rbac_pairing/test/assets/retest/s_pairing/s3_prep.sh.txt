#!/bin/sh
# R-S step 3 prep: Stack Check, then Pair (round 1's history: Stack Check under Pair), then focus code slot 0
# (bounds from a UI dump of Pair; allowed) and check the keypad. emulator-5558 ONLY. No digits typed here.
set -u
OUT=/Users/gsp-budi/Documents/DEMS/docs/2026-10-04_dems_backend_rbac_pairing/test/assets/retest/s_pairing
S=emulator-5558
APP=com.dems.demo.tablet
a() { adb -s "$S" "$@"; }
now() { python3 -c 'import time; t=time.time(); print(time.strftime("%H:%M:%S", time.localtime(t)) + ".%03d" % int((t % 1) * 1000))'; }
kg() { a shell dumpsys window 2>/dev/null | tr -d '\r' | grep -o 'isKeyguardShowing=[a-z]*' | head -n 1; }
link() { echo "-- host $(now) warm link '$1'"; a shell "am start -W -a android.intent.action.VIEW -d '$1' $APP" 2>&1 | tr -d '\r' | grep -E "Status|Error" | sed 's/^/   /'; }
echo "== R-S step 3 prep — host $(now); $(kg); uptime: $(uptime | sed 's/.*load/load/')"
link 'demsdemo:///stack-check'; sleep 5
a exec-out screencap -p > "$OUT/03_phone_stack_check_visited.png"; echo "   screencap 03_phone_stack_check_visited.png at host $(now)"
link 'demsdemo:///pair'; sleep 5
a shell uiautomator dump /sdcard/rs_ui.xml >/dev/null 2>&1; a shell cat /sdcard/rs_ui.xml > "$OUT/03_ui_pair.xml"; a shell rm -f /sdcard/rs_ui.xml
echo "   UI dump 03_ui_pair.xml ($(wc -c < "$OUT/03_ui_pair.xml" | tr -d ' ') bytes): screen ids [$(grep -o 'resource-id="screen-[a-z-]*"' "$OUT/03_ui_pair.xml" | sed 's/resource-id=//; s/"//g' | sort -u | tr '\n' ' ')]"
B=$(grep -o 'resource-id="pairing-code-slot-0"[^>]*bounds="\[[0-9]*,[0-9]*\]\[[0-9]*,[0-9]*\]"' "$OUT/03_ui_pair.xml" | grep -o 'bounds="[^"]*"' | head -n 1)
[ -z "$B" ] && B=$(grep -o '<node[^>]*resource-id="pairing-code-slot-0"[^>]*>' "$OUT/03_ui_pair.xml" | grep -o 'bounds="[^"]*"' | head -n 1)
echo "   pairing-code-slot-0 $B"
X=$(echo "$B" | sed -E 's/bounds="\[([0-9]+),([0-9]+)\]\[([0-9]+),([0-9]+)\]"/\1 \3/' | awk '{print int(($1+$2)/2)}')
Y=$(echo "$B" | sed -E 's/bounds="\[([0-9]+),([0-9]+)\]\[([0-9]+),([0-9]+)\]"/\2 \4/' | awk '{print int(($1+$2)/2)}')
echo "-- $(kg); tap the centre of pairing-code-slot-0: $X $Y at host $(now)"
a shell input tap "$X" "$Y"; sleep 2
IME=$(a shell dumpsys input_method | tr -d '\r' | grep -o 'mInputShown=[a-z]*' | head -n 1)
if [ "$IME" != "mInputShown=true" ]; then echo "   $IME after the first tap; tap again"; a shell input tap "$X" "$Y"; sleep 2; IME=$(a shell dumpsys input_method | tr -d '\r' | grep -o 'mInputShown=[a-z]*' | head -n 1); fi
a exec-out screencap -p > "$OUT/03_phone_code_field_focused.png"
echo "   screencap 03_phone_code_field_focused.png at host $(now) (empty slots); IME $IME; focus: $(a shell dumpsys window | tr -d '\r' | grep -m 1 mCurrentFocus | sed 's/^ *//')"
