import os, re, json
R='/Users/gsp-budi/Documents/DEMS'
web_in=['A1','A3','A4','A5','A6','A7','A8','A9','A10','A11']+[f'B-{i}' for i in range(1,11)]+['C-7']+[f'D-{i}' for i in range(1,7)]
mob_in=[f'S1-{i}' for i in range(1,9)]+[f'S2-{i}' for i in range(1,9)]+['S3-1','S3-2','S4-7','S4-8']+[f'S5-{i}' for i in range(1,8)]
def tests(root):
    for dp,dn,fn in os.walk(root):
        if 'node_modules' in dp: continue
        for f in fn:
            if re.search(r'\.test\.tsx?$',f): yield os.path.join(dp,f)
LIVE=re.compile(r"createLiveTestBed|createFakeTransport|@dems/shared/live/testing|live-[\w-]*bed|renderLive\w*\(|createMobileLive")
def scan(files, ids, app):
    res={}
    texts={p:open(p,encoding='utf8').read() for p in files}
    for i in ids:
        pat=re.compile(r'(?<![\w.-])'+re.escape(i)+r'(?![\w-])')
        hits=[]
        for p,t in texts.items():
            n=len(pat.findall(t))
            if n:
                rel=os.path.relpath(p,f'{R}/apps/{app}/src')
                live=bool(LIVE.search(t)) or bool(re.search(r'/testing/live-',t))
                static=bool(re.search(r'static',os.path.basename(p)))
                # names in an it/describe title?
                titled=bool(re.search(r"(?:it|describe|test)(?:\.each\([^)]*\))?\(\s*[`'\"][^`'\"\n]*"+re.escape(i)+r"(?![\w-])",t))
                hits.append({'file':rel,'count':n,'live':live,'static':static,'titled':titled})
        res[i]=hits
    return res
w=scan(list(tests(R+'/apps/core-web/src')),web_in,'core-web')
m=scan(list(tests(R+'/apps/mobile/src')),mob_in,'mobile')
json.dump({'web':w,'mobile':m},open(R+'/.cache/w87/sites2.json','w'),indent=1)
for k,v in list(w.items())+list(m.items()):
    livefiles=[h for h in v if h['live'] and not h['static']]
    print(k, 'files',len(v),'live-non-static',len(livefiles),'titled',sum(h['titled'] for h in livefiles), '|', '; '.join(f"{h['file']}({h['count']}{',T' if h['titled'] else ''}{',L' if h['live'] else ''}{',S' if h['static'] else ''})" for h in v))
