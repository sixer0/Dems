#!/bin/bash
# run.sh <phone|tablet>: starts the RF2 queue driver for that lane at the next free batch number.
L=$1; R=/private/tmp/claude-501/-Users-gsp-budi-Documents-DEMS/0d71fd65-4a51-4019-9c48-5c92d11bbdaf/scratchpad/rf2
N=""
for c in $(ls $R/$L/q 2>/dev/null | grep -E '^[0-9]{3}\.cmd$' | sort); do n=${c%.cmd}; if [ ! -f "$R/$L/q/$n.out" ]; then N=$((10#$n)); break; fi; done
if [ -z "$N" ]; then N=$(ls $R/$L/q 2>/dev/null | grep -E '^[0-9]{3}\.cmd$' | sort | tail -1 | cut -c1-3); N=$((10#${N:-0} + 1)); fi
cd /Users/gsp-budi/Documents/DEMS
RF2_LANE=$L RT_START=$N exec node --import tsx $R/rf2q.mts $R/$L/q $R/$L/assets $R/$L/frames >> $R/$L/driver.log 2>&1
