"""lag.py <rows.json> <coreStartHost HH:MM:SS.mmm> [coreStopHost]: the monitor timer against the Core's elapsed time.
For every monitor frame with a decoded timer: lag = (frame host time - Core start) - timer seconds. A drawn timer that
keeps up has lag in [0, ~1.3) s. Reported: max lag, and the intervals where lag > 2 s (the timer behind the Core),
with how many frames were written meanwhile (frames written = the screen kept changing)."""
import json, sys
def secs(t):
    h, m, s = t.split(':'); return int(h) * 3600 + int(m) * 60 + float(s)
rows = json.load(open(sys.argv[1]))[1:]
start = secs(sys.argv[2])
stop = secs(sys.argv[3]) if len(sys.argv) > 3 else None
pts = []
for r in rows:
    if r['dot'] is None or not r.get('timer') or '?' in r['timer'] or not r.get('host'):
        continue
    t = r['timer']; v = int(t[0:2]) * 3600 + int(t[2:4]) * 60 + int(t[4:6])
    h = secs(r['host'])
    if stop is not None and h > stop + 0.3:
        continue  # after the stop the timer rightly freezes at the stop's value
    pts.append((r['frame'], r['host'], h, v, round(h - start - v, 2)))
over = [p for p in pts if p[4] > 2.0]
spans = []
for p in over:
    if spans and int(p[0][2:5]) - int(spans[-1][-1][0][2:5]) <= 3:
        spans[-1].append(p)
    else:
        spans.append([p])
out = {'frames': len(pts), 'maxLag': max((p[4] for p in pts), default=None), 'maxLagAt': max(pts, key=lambda p: p[4])[:2] if pts else None,
       'framesLagOver2s': len(over),
       'spans': [{'from': s[0][1], 'to': s[-1][1], 'frames': len(s), 'timer': sorted({x[3] for x in s}), 'maxLag': max(x[4] for x in s)} for s in spans]}
print(json.dumps(out))
