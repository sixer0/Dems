#!/bin/bash
# run.sh <lane>: starts the RF-PH queue driver for a lane at the next free batch number.
LANE=$1
S=/private/tmp/claude-501/-Users-gsp-budi-Documents-DEMS/0d71fd65-4a51-4019-9c48-5c92d11bbdaf/scratchpad/rfph
mkdir -p $S/q_$LANE $S/${LANE}_assets $S/${LANE}_frames
N=""
for c in $(ls $S/q_$LANE 2>/dev/null | grep -E '^[0-9]{3}\.cmd$' | sort); do n=${c%.cmd}; if [ ! -f "$S/q_$LANE/$n.out" ]; then N=$((10#$n)); break; fi; done
if [ -z "$N" ]; then N=$(ls $S/q_$LANE 2>/dev/null | grep -E '^[0-9]{3}\.cmd$' | sort | tail -1 | cut -c1-3); N=$((10#${N:-0} + 1)); fi
cd /Users/gsp-budi/Documents/DEMS
RT_START=$N CF_LANE=$LANE exec node --import tsx $S/rfq.mts $S/q_$LANE $S/${LANE}_assets $S/${LANE}_frames >> $S/driver_$LANE.log 2>&1
