// E2E-W2 Part B: two consoles live on http://127.0.0.1:5173, two separate headless-shell processes
// with separate fresh profiles: console 1 SUPERVISOR andi.pratama, console 2 INVESTIGATOR budi.santoso.
// Each change is made in one console and watched in the other WITHOUT a reload: a watcher installed
// in the observing page (setInterval 20 ms) stores Date.now() when the change first shows. The delay
// is measured from the acting console's HTTP 2xx answer (and from the click), and the Core's own
// audit row time is read afterwards. One machine clock for all.
// Also: the SUPERVISOR reads the ACCESS_DENIED / EXPORT_DENIED rows of Part A (Security & Audit).
// Usage: TMPDIR=… node w2_part_b.mjs <assetsDir>

const shotBoth = async (change, phase, cap1, cap2) => {

  const f2 = `${nn()}_live_${change}_investigator_${phase}.png`
  await c2.screenshot(f2)
  result.screenshots.push({ file: f2, caption: cap2 })
  return [f1, f2]
}
const shotOne = async (c, slug, caption) => {
  const f = `${nn()}_${slug}.png`
  await c.screenshot(f)
  result.screenshots.push({ file: f, caption })
  return f
}
const txt = (c, tid) => c.evaluate(`document.querySelector('[data-testid="${tid}"]')?.textContent?.replace(/\\s+/g,' ').trim().slice(0,200) ?? null`)
const watch = (c, key, cond) =>
  c.evaluate(`(() => { window.__w2 = window.__w2 || {}; delete window.__w2[${JSON.stringify(key)}]; const f = () => { try { return !!(${cond}) } catch { return false } }; if (f()) { window.__w2[${JSON.stringify(key)}] = -1; return 'already' } const t = setInterval(() => { if (f()) { window.__w2[${JSON.stringify(key)}] = Date.now(); clearInterval(t) } }, 20); window.__w2t = window.__w2t || {}; window.__w2t[${JSON.stringify(key)}] = t; return 'armed' })()`)
const seen = async (c, key, timeoutMs = 15000) => {
  const until = Date.now() + timeoutMs
  while (Date.now() < until) {
    const v = await c.evaluate(`(window.__w2 || {})[${JSON.stringify(key)}] ?? null`)
    if (v !== null) return v
    await sleep(100)
  }
  return null
}
const navCount = (c) => c.allRequests.filter((r) => r.type === 'Document').length
const respOf = (c, part, method, after) => {
  const e = [...c.allRequests].reverse().find((r) => r.method === method && r.url.includes(part) && typeof r.status === 'number' && r.status < 300 && r.at >= after)
  return e ? { status: e.status, sentAt: e.at, respAt: e.respAt ?? null } : null
}
const auditAt = async (c, type, after) => {
  const r = await c.pageFetch('GET', `/api/v1/audit?type=${type}&order=DESC&limit=10`)
  const rows = r.body?.items ?? r.body?.rows ?? []
  const row = rows.find((x) => {
    const at = typeof x.at === 'number' ? x.at : Date.parse(x.at ?? x.timestamp ?? x.time)
    return at >= after - 2000
  })
  if (!row) return { status: r.status, row: null, keys: rows[0] ? Object.keys(rows[0]) : [] }
  const at = typeof row.at === 'number' ? row.at : Date.parse(row.at ?? row.timestamp ?? row.time)
  return { seq: row.seq, type: row.type, at, actorId: row.actorId ?? row.actor?.id ?? null }
}
const iso = (t) => (typeof t === 'number' && t > 0 ? new Date(t).toISOString() : t)
const record = (change, madeIn, seenIn, info) => {
  const delayFromResponse = info.seenAt && info.response?.respAt ? info.seenAt - info.response.respAt : null
  const delayFromClick = info.seenAt && info.clickAt ? info.seenAt - info.clickAt : null
  const delayFromCore = info.seenAt && info.core?.at ? info.seenAt - info.core.at : null
  const reloads = info.reloadsDuring ?? null
  const pass = info.seenAt !== null && info.seenAt > 0 && reloads === 0 && (delayFromCore ?? delayFromResponse ?? 99999) <= 2000
  const row = {
    change,
    madeIn,
    seenIn,
    clickAt: iso(info.clickAt),
    response: info.response ? { ...info.response, sentAt: iso(info.response.sentAt), respAt: iso(info.response.respAt) } : null,
    core: info.core ? { ...info.core, at: iso(info.core.at) } : null,
    seenAt: iso(info.seenAt),
    delayMsFromResponse: delayFromResponse,
    delayMsFromCoreRow: delayFromCore,
    delayMsFromClick: delayFromClick,
    observerReloadsDuringChange: reloads,
    result: pass ? 'PASS' : 'FAIL',
    screenshots: info.screenshots,
    note: info.note ?? '',
  }
  result.changes.push(row)
  c1.log(`change ${change}: ${row.result} seen ${row.seenAt} delay(resp) ${delayFromResponse} delay(core) ${delayFromCore}`)
  return row
}

  try {

    await shotOne(c2, 'live_error_investigator', `driver error: ${result.error.slice(0, 60)}`)
  } catch {}
}

// ── Sign out both consoles (their own sessions only), through the user menu ──
for (const [name, c] of [['supervisor', c1], ['investigator', c2]]) {
  try {
    await c.key('Escape', 'Escape', 27)
    await c.navigate(ORIGIN + '/dashboard', 3000)
    await c.click('user-menu')
    await c.waitUntil(`document.querySelector('[role="menu"]') !== null`, 'menu', 5000)
    await sleep(500)
    const at = await c.clickSel('[role="menu"] [role="menuitem"][data-key="signOut"]', 'sign out')
    const back = await c.waitFor('page-login', true, 15000)
    await sleep(1200)
    const after = await c.pageFetch('GET', '/api/v1/auth/session')
    const f = await shotOne(c, `${name}_signed_out_login`, `${name.toUpperCase()} console signed out (own session), Login page`)
    result.signOut[name] = { at: iso(at), loginShown: back, sessionAfter: after.status, screenshot: f }
  } catch (e) {
    result.signOut[name] = { error: String(e.message || e) }
  }
}
result.endedAt = lib.nowIso()
for (const [name, c] of [['supervisor', c1], ['investigator', c2]]) {
  result.consoles[name].inlineDataUrls = c.allRequests.filter((r) => /^(data|blob):/.test(r.url)).length
  result.consoles[name].origins = [...new Set(c.allRequests.filter((r) => !/^(data|blob):/.test(r.url)).map((r) => { try { return new URL(r.url.replace('?…', '')).origin } catch { return r.url.slice(0, 30) } }))]
  result.consoles[name].requests = c.allRequests.filter((r) => !/^(data|blob):/.test(r.url) && (r.url.includes('/api/') || r.type === 'WebSocket' || r.type === 'Document')).map((r) => `${new Date(r.at).toISOString().slice(11, 23)} ${r.method} ${r.url.replace(ORIGIN, '')} ${r.status}`)
  result.consoles[name].sockets = c.sockets.map((s) => ({ at: iso(s.at), url: s.url }))
  result.consoles[name].console = c.allConsole
}
lib.writeJson(join(ASSETS, 'w2_part_b_two_consoles.json'), result)
console.log(lib.sanitize(JSON.stringify({ error: result.error ?? null, pre: result.preconditions, changes: result.changes.map((x) => [x.change, x.result, x.delayMsFromResponse, x.delayMsFromCoreRow, x.delayMsFromClick, x.observerReloadsDuringChange]), audit: result.auditRead?.matches?.map((m) => [m.role, m.operation, m.status, m.row?.seq ?? null]), signOut: result.signOut, origins: [result.consoles.supervisor.origins, result.consoles.investigator.origins], shots: n }, null, 1)))
c1.close()
c2.close()
await lib.shutdown(b1)
await lib.shutdown(b2)
process.exit(0)
