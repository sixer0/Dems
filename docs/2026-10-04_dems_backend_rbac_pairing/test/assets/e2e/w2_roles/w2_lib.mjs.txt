// E2E-W2 raw-CDP library (Node built-in WebSocket, no install). Adapted from W86's driver
// (implementation/assets/86_cdp-main.mjs.txt). One headless-shell process per console, each with
// its own fresh profile under $TMPDIR, viewport 1440x900, origin http://127.0.0.1:5173 only.
//
// Never stored or printed: the demo password (read from README.md inside the process), a cookie
// value, a session token, a request body of a sign-in. sanitize() masks anything token-shaped.
import { spawn } from 'node:child_process'
import { mkdtempSync, readFileSync, rmSync, writeFileSync, appendFileSync } from 'node:fs'
import { join } from 'node:path'

export const ORIGIN = 'http://127.0.0.1:5173'
export const REPO = '/Users/gsp-budi/Documents/DEMS'
export const SHELL_BIN =
  '/Users/gsp-budi/Library/Caches/ms-playwright/chromium_headless_shell-1243/chrome-headless-shell-mac-arm64/chrome-headless-shell'
export const sleep = (ms) => new Promise((r) => setTimeout(r, ms))
export const nowIso = () => new Date().toISOString()

/** The demo password of `username`, from the README Quick start table. Never printed. */
export function passwordOf(username) {
  const readme = readFileSync(join(REPO, 'README.md'), 'utf8')
  const line = readme.split('\n').find((l) => l.startsWith(`| \`${username}\` |`))
  if (!line) throw new Error(`README has no Quick start row for ${username}`)
  const cells = [...line.matchAll(/`([^`]+)`/g)].map((m) => m[1])
  const pw = cells[cells.length - 1]
  if (!pw || pw === username) throw new Error('no password cell')
  return pw
}

let SECRETS = []
export function registerSecret(s) {
  if (s && !SECRETS.includes(s)) SECRETS.push(s)
}
export function sanitize(text) {
  let out = String(text)
  for (const s of SECRETS) out = out.split(s).join('***')
  out = out.replace(/\b(ses|dvc|bio|tok)_[A-Za-z0-9_-]{6,}/g, '$1_***')
  return out
}

/**
 * Starts one headless shell with a fresh profile. The devtools port is chosen by the OS
 * (`--remote-debugging-port=0`) and read from the browser's own "DevTools listening on" line, and
 * the listener's PID is checked to be this browser's: no fixed port can collide with another row's
 * browser (incident 09:42, see the report).
 */
export async function launch(name) {
  const profile = mkdtempSync(join(process.env.TMPDIR || '/tmp', `w2_${name}_`))
  const args = [
    '--headless',
    '--remote-debugging-port=0',
    `--user-data-dir=${profile}`,
    '--no-first-run',
    '--no-default-browser-check',
    '--disable-background-networking',
    '--disable-component-update',
    '--disable-sync',
    '--metrics-recording-only',
    '--window-size=1440,900',
    '--host-resolver-rules=MAP * ~NOTFOUND, EXCLUDE 127.0.0.1',
    'about:blank',
  ]
  const proc = spawn(SHELL_BIN, args, { stdio: ['ignore', 'ignore', 'pipe'], detached: false })
  let stderr = ''
  proc.stderr.on('data', (d) => (stderr += d.toString()))
  for (let i = 0; i < 100; i++) {
    const m = stderr.match(/DevTools listening on (ws:\/\/([^/]+)\/devtools\/browser\/[^\s]+)/)
    if (m) {
      const hostPort = m[2]
      const port = Number(hostPort.split(':').pop())
      const { execFileSync } = await import('node:child_process')
      let pids = ''
      try {
        pids = execFileSync('/usr/sbin/lsof', ['-nP', `-iTCP:${port}`, '-sTCP:LISTEN', '-t']).toString().trim().split('\n')
      } catch {}
      if (!pids.includes(String(proc.pid))) {
        proc.kill('SIGKILL')
        throw new Error(`port ${port}: listener pids ${pids} are not this browser ${proc.pid}`)
      }
      if (pids.length !== 1) {
        proc.kill('SIGKILL')
        throw new Error(`port ${port}: more than one listener ${pids}`)
      }
      const http = `http://${hostPort}`
      const v = await (await fetch(`${http}/json/version`)).json()
      return { proc, profile, port, http, version: v, stderrTail: () => stderr.slice(-800) }
    }
    await sleep(150)
  }
  proc.kill('SIGKILL')
  throw new Error(`headless shell ${name} printed no devtools endpoint; stderr: ${stderr.slice(-400)}`)
}

export async function shutdown(browser) {
  try {
    browser.proc.kill('SIGTERM')
  } catch {}
  await sleep(800)
  try {
    browser.proc.kill('SIGKILL')
  } catch {}
  await sleep(200)
  rmSync(browser.profile, { recursive: true, force: true })
}

const cleanUrl = (url) => String(url).replace(/\?.*$/, (q) => (q.length > 1 ? '?…' : ''))

/** Connects to the page target of the browser on `port`. */
export async function connect(browser, label, assetsDir, logFile) {
  const list = await (await fetch(`${browser.http}/json/list`)).json()
  const target = list.find((t) => t.type === 'page')
  const ws = new WebSocket(target.webSocketDebuggerUrl)
  await new Promise((r) => ws.addEventListener('open', r, { once: true }))
  let id = 0
  const pending = new Map()
  const c = {
    label,
    requests: new Map(),
    allRequests: [], // every request of the run: origin check
    sockets: [],
    wsFrames: [], // { at, type } of every received realtime frame (no payload kept)
    consoleLines: [],
    allConsole: [],
    loadFired: false,
    frameListeners: [],
  }
  const log = (line) => {
    if (logFile) appendFileSync(logFile, sanitize(`${nowIso()} [${label}] ${line}`) + '\n')
  }
  c.log = log
  ws.addEventListener('message', (event) => {
    const msg = JSON.parse(event.data)
    if (msg.id && pending.has(msg.id)) {
      const { resolve, reject, method } = pending.get(msg.id)
      pending.delete(msg.id)
      msg.error ? reject(new Error(`CDP error ${method}: ${msg.error.message}`)) : resolve(msg.result)
      return
    }
    const p = msg.params ?? {}
    switch (msg.method) {
      case 'Page.loadEventFired':
        c.loadFired = true
        break
      case 'Network.requestWillBeSent': {
        const e = { at: Date.now(), method: p.request.method, url: cleanUrl(p.request.url), status: null, type: p.type }
        c.requests.set(p.requestId, e)
        c.allRequests.push(e)
        break
      }
      case 'Network.responseReceived': {
        const e = c.requests.get(p.requestId)
        if (e) {
          e.status = p.response.status
          e.respAt = Date.now()
        }
        break
      }
      case 'Network.loadingFailed': {
        const e = c.requests.get(p.requestId)
        if (e && e.status === null) e.status = `failed: ${p.errorText}`
        break
      }
      case 'Network.webSocketCreated':
        c.sockets.push({ at: Date.now(), url: cleanUrl(p.url) })
        c.allRequests.push({ at: Date.now(), method: 'WS', url: cleanUrl(p.url), status: 'created', type: 'WebSocket' })
        break
      case 'Network.webSocketFrameReceived': {
        let type = null
        try {
          const j = JSON.parse(p.response.payloadData)
          type = j.type ?? j.kind ?? null
        } catch {}
        const f = { at: Date.now(), type }
        c.wsFrames.push(f)
        if (c.wsFrames.length > 4000) c.wsFrames.splice(0, 1000)
        break
      }
      case 'Runtime.exceptionThrown': {
        const line = `exception: ${String(p.exceptionDetails?.exception?.description ?? p.exceptionDetails?.text).slice(0, 300)}`
        c.consoleLines.push(line)
        c.allConsole.push({ at: nowIso(), line })
        break
      }
      case 'Runtime.consoleAPICalled':
        if (p.type === 'error' || p.type === 'warning') {
          const text = (p.args ?? []).map((a) => (a.value !== undefined ? String(a.value) : String(a.description ?? a.type))).join(' ')
          const line = `${p.type}: ${text.slice(0, 300)}`
          c.consoleLines.push(line)
          c.allConsole.push({ at: nowIso(), line })
        }
        break
      case 'Log.entryAdded':
        if (p.entry && (p.entry.level === 'error' || p.entry.level === 'warning')) {
          const line = `log-${p.entry.level} (${p.entry.source}): ${String(p.entry.text).slice(0, 200)} ${p.entry.url ? cleanUrl(p.entry.url) : ''}`
          c.consoleLines.push(line)
          c.allConsole.push({ at: nowIso(), line })
        }
        break
    }
  })
  c.send = (method, params = {}) =>
    new Promise((resolve, reject) => {
      const n = ++id
      pending.set(n, { resolve, reject, method })
      ws.send(JSON.stringify({ id: n, method, params }))
    })
  c.evaluate = async (expression, what = 'eval') => {
    const r = await c.send('Runtime.evaluate', { expression, awaitPromise: true, returnByValue: true })
    if (r.exceptionDetails) throw new Error(`page script failed: ${what}: ${String(r.exceptionDetails?.exception?.description ?? r.exceptionDetails.text).slice(0, 200)}`)
    return r.result.value
  }
  c.navigate = async (url, settleMs = 2500) => {
    c.loadFired = false
    await c.send('Page.navigate', { url })
    for (let i = 0; i < 150 && !c.loadFired; i++) await sleep(100)
    await sleep(settleMs)
    await c.assertOrigin('after navigate')
    return c.loadFired
  }
  c.assertOrigin = async (what) => {
    const o = await c.evaluate('location.origin', 'origin')
    if (o !== ORIGIN) throw new Error(`ORIGIN GUARD (${what}): page is at ${o}, not ${ORIGIN}; stopping`)
    return o
  }
  c.waitUntil = async (expression, what, timeoutMs = 15000) => {
    const until = Date.now() + timeoutMs
    while (Date.now() < until) {
      try {
        if ((await c.evaluate(expression, what)) === true) return true
      } catch {}
      await sleep(120)
    }
    return false
  }
  c.waitFor = (testId, present = true, timeoutMs = 15000) =>
    c.waitUntil(`(document.querySelector('[data-testid="${testId}"]') !== null) === ${present}`, `wait ${testId}`, timeoutMs)
  c.screenshot = async (name) => {
    const { data } = await c.send('Page.captureScreenshot', { format: 'png' })
    writeFileSync(join(assetsDir, name), Buffer.from(data, 'base64'))
    log(`screenshot ${name}`)
    return name
  }
  c.typeInto = async (selector, text, what) => {
    const ok = await c.evaluate(
      `(() => { const el = document.querySelector(${JSON.stringify(selector)}); if (!el) return false; el.focus(); if (el.select) el.select(); return document.activeElement === el })()`,
      `focus ${what}`,
    )
    if (!ok) throw new Error(`field ${what} not focusable`)
    await c.send('Input.insertText', { text })
  }
  c.press = (testId) =>
    c.evaluate(`(() => { const el = document.querySelector('[data-testid="${testId}"]'); if (!el) return false; el.click(); return true })()`, `press ${testId}`)
  /** A real mouse click at the centre of the first element matching `selector`. Returns the click time (ms epoch) or false. */
  c.clickSel = async (selector, what = selector) => {
    const box = await c.evaluate(
      `(() => { const el = document.querySelector(${JSON.stringify(selector)}); if (!el) return null; el.scrollIntoView({ block: 'center', inline: 'center' }); const r = el.getBoundingClientRect(); return { x: r.x + r.width / 2, y: r.y + r.height / 2, w: r.width, h: r.height } })()`,
      `box ${what}`,
    )
    if (!box || box.w === 0) return false
    await sleep(120)
    await c.send('Input.dispatchMouseEvent', { type: 'mouseMoved', x: box.x, y: box.y })
    await sleep(40)
    const at = Date.now()
    await c.send('Input.dispatchMouseEvent', { type: 'mousePressed', x: box.x, y: box.y, button: 'left', clickCount: 1 })
    await sleep(50)
    await c.send('Input.dispatchMouseEvent', { type: 'mouseReleased', x: box.x, y: box.y, button: 'left', clickCount: 1 })
    return at
  }
  c.click = (testId) => c.clickSel(`[data-testid="${testId}"]`, testId)
  c.key = async (key, code = key, keyCode = 0) => {
    await c.send('Input.dispatchKeyEvent', { type: 'keyDown', key, code, windowsVirtualKeyCode: keyCode })
    await c.send('Input.dispatchKeyEvent', { type: 'keyUp', key, code, windowsVirtualKeyCode: keyCode })
  }
  c.takeNet = () => {
    const api = [...c.requests.values()].filter((e) => e.url.includes('/api/v1/')).map((e) => `${e.method} ${e.url.replace(ORIGIN, '')} ${e.status}`)
    c.requests = new Map()
    const cons = c.consoleLines
    c.consoleLines = []
    return { api, console: cons }
  }
  /** fetch() from the page context, same origin, the page's own cookie. Body summarised, never the sign-in. */
  c.pageFetch = async (method, path, body) => {
    const expr = `(async () => {
      const init = { method: ${JSON.stringify(method)}, credentials: 'same-origin', headers: {} }
      ${body !== undefined ? `init.headers['content-type'] = 'application/json'; init.body = ${JSON.stringify(JSON.stringify(body))};` : ''}
      const t0 = Date.now()
      const r = await fetch(${JSON.stringify(path)}, init)
      let j = null; const txt = await r.text(); try { j = JSON.parse(txt) } catch {}
      return { status: r.status, requestId: r.headers.get('x-request-id'), body: j, at: t0 }
    })()`
    return c.evaluate(expr, `fetch ${method} ${path}`)
  }
  await c.send('Network.enable')
  await c.send('Page.enable')
  await c.send('Runtime.enable')
  await c.send('Log.enable')
  await c.send('Emulation.setDeviceMetricsOverride', { width: 1440, height: 900, deviceScaleFactor: 1, mobile: false })
  c.close = () => ws.close()
  return c
}

/** Signs in through the Login form. The password never leaves this process except into the field. */
export async function signIn(c, username) {
  const pw = passwordOf(username)
  registerSecret(pw)
  await c.navigate(`${ORIGIN}/login`, 2500)
  const shown = await c.waitFor('page-login', true, 20000)
  if (!shown) throw new Error('login page not shown')
  const mode = await c.evaluate(`document.querySelector('[data-testid="login-mode-selector"]')?.getAttribute('data-mode') ?? null`)
  if (mode !== 'LIVE') throw new Error(`login mode is ${mode}, not LIVE`)
  await c.assertOrigin('before typing the sign-in')
  await c.typeInto('[data-testid="login-username-field"] input', username, 'username')
  await c.typeInto('[data-testid="login-password-field"] input', pw, 'password')
  await c.assertOrigin('before submitting the sign-in')
  const at = Date.now()
  await c.press('login-submit')
  const ok = await c.waitFor('page-dashboard', true, 25000)
  await sleep(2500)
  const shell = await c.evaluate(
    `(() => ({ href: location.origin + location.pathname, coreMode: document.querySelector('[data-core-mode]')?.getAttribute('data-core-mode') ?? null, userMenu: document.querySelector('[data-testid="user-menu"]')?.getAttribute('aria-label') ?? null, scriptCookies: document.cookie === '' ? 0 : document.cookie.split(';').length }))()`,
  )
  c.log(`signIn ${username}: dashboard=${ok} ${JSON.stringify(shell)}`)
  return { username, at, dashboard: ok, shell }
}

/** The visible navigation entries of the sidebar. */
export const navFacts = (c) =>
  c.evaluate(`(() => {
    const items = [...document.querySelectorAll('.dems-nav-list li[data-id]')].filter((li) => li.offsetParent !== null || li.getClientRects().length > 0)
    return {
      entries: items.map((li) => li.getAttribute('data-id')),
      adminGroup: document.querySelector('[data-testid="nav-admin-group"]') !== null,
      current: [...document.querySelectorAll('.dems-nav-list a[aria-current="page"]')].map((a) => a.closest('li')?.getAttribute('data-id')),
    }
  })()`)

/**
 * Every interactive control inside `rootSel` (default: main content): kind, label, test ID, enabled state,
 * disabled reason nearby. Text is trimmed and capped; no input value is read.
 */
export const controlFacts = (c, rootSel = 'main') =>
  c.evaluate(`(() => {
    const root = document.querySelector(${JSON.stringify(rootSel)}) || document.body
    const els = [...root.querySelectorAll('button, a[href], [role="button"], [role="menuitem"], [role="switch"], [role="tab"], [role="checkbox"], [role="radio"], input, select, textarea, [role="combobox"]')]
    const seen = new Set()
    const out = []
    for (const el of els) {
      if (!(el.offsetParent !== null || el.getClientRects().length > 0)) continue
      const tid = el.getAttribute('data-testid') || el.closest('[data-testid]')?.getAttribute('data-testid') || null
      const label = (el.getAttribute('aria-label') || el.textContent || el.getAttribute('placeholder') || el.getAttribute('name') || '').replace(/\\s+/g, ' ').trim().slice(0, 70)
      const disabled = el.disabled === true || el.getAttribute('aria-disabled') === 'true' || el.hasAttribute('data-disabled') || el.getAttribute('data-disabled') === 'true'
      const kind = el.getAttribute('role') || el.tagName.toLowerCase()
      const key = kind + '|' + tid + '|' + label
      if (seen.has(key)) continue
      seen.add(key)
      out.push({ kind, tid, label, enabled: !disabled })
    }
    return out
  })()`)

/** Texts of "Requires the … role" / "Not permitted" style reasons on screen. */
export const reasonFacts = (c) =>
  c.evaluate(`(() => {
    const t = document.body.innerText
    const m = t.match(/(Requires the [A-Za-z /]+ role[^\\n]*|Not permitted[^\\n]*|Live mode only[^\\n]*|needs the [A-Za-z ]+ role[^\\n]*|cannot [a-z ]+ sessions[^\\n]*)/g) || []
    return [...new Set(m.map((s) => s.slice(0, 120)))]
  })()`)

export const pageId = (c) =>
  c.evaluate(`(() => [...document.querySelectorAll('[data-testid^="page-"]')].map((e) => e.getAttribute('data-testid')))()`)

export const toastFacts = (c) =>
  c.evaluate(`(() => [...document.querySelectorAll('[data-sonner-toast], [role="status"], [role="alert"], .toast, [data-toast]')].map((e) => e.textContent.replace(/\\s+/g, ' ').trim().slice(0, 160)).filter(Boolean))()`)

export function writeJson(file, obj) {
  writeFileSync(file, sanitize(JSON.stringify(obj, null, 2)) + '\n')
}
