// E2E-P client: writes the next numbered batch for e2e_q.mts and waits for its .out.
//   node send.mjs <queueDir> <timeoutSeconds> <<'EOF' ... EOF
import { existsSync, readFileSync, readdirSync, writeFileSync } from 'node:fs';
const [Q, timeout = '240'] = process.argv.slice(2);
const body = readFileSync(0, 'utf8');
const taken = readdirSync(Q).filter((f) => /^\d{3}\.cmd$/.test(f)).map((f) => Number(f.slice(0, 3)));
const id = String((taken.length ? Math.max(...taken) : 0) + 1).padStart(3, '0');
writeFileSync(`${Q}/${id}.cmd`, body);
const until = Date.now() + Number(timeout) * 1000;
const tick = () => {
  if (existsSync(`${Q}/${id}.out`)) {
    setTimeout(() => { process.stdout.write(`== ${id}\n${readFileSync(`${Q}/${id}.out`, 'utf8')}`); }, 100);
    return;
  }
  if (Date.now() > until) { console.log(`== ${id} no answer within ${timeout} s`); process.exit(3); }
  setTimeout(tick, 200);
};
tick();
