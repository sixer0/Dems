// E2E-S step 2 (part a): the web console in Live mode, SUPERVISOR andi.pratama signs in THROUGH THE FORM
// (password from the project's seed constant, never printed), opens Paired Devices from the sidebar.
//   node --import tsx 02a_web_signin.mjs
import { DEMO_OPERATOR_PASSWORD, OPERATORS } from '/Users/gsp-budi/Documents/DEMS/packages/shared/src/sim/dataset.ts'
import { ORIGIN, connect, log, pdFacts, sleep, stamp, writeJson } from './lib.mjs'

const andi = OPERATORS.find((o) => o.username === 'andi.pratama')
if (!andi || andi.role !== 'SUPERVISOR' || typeof DEMO_OPERATOR_PASSWORD !== 'string') throw new Error('seed: andi.pratama is not the SUPERVISOR')
const page = await connect()
const version = await page.send('Browser.getVersion')
log(`browser ${version.product}; headless ${/Headless/i.test(version.userAgent)}; operator ${andi.username} (${andi.id}, ${andi.role})`)
await page.navigate(`${ORIGIN}/login`)
const coreMode = await page.evaluate(`document.querySelector('[data-testid="page-login"]')?.getAttribute('data-core-mode') ?? null`, 'core mode')
log(`login page at ${await page.href()}; data-core-mode=${coreMode}`)
await page.typeInto('[data-testid="login-username-field"] input', andi.username, 'username')
await page.typeInto('[data-testid="login-password-field"] input', DEMO_OPERATOR_PASSWORD, 'password')
await page.press('login-submit')
const dash = await page.waitFor('page-dashboard', true, 25000)
await sleep(2500)
log(`dashboard shown: ${dash}; shell mode: ${await page.text('shell-mode')}`)
const linked = await page.evaluate(`(() => { const a = document.querySelector('a[href="/admin/devices"]'); if (!a) return false; a.click(); return true })()`, 'sidebar')
const shown = await page.waitFor('pd-issue-panel', true, 20000)
await page.waitAny(['pd-empty', 'pd-devices-table', 'pd-error'], 15000)
await sleep(2500)
const facts = await pdFacts(page)
const shot = await page.screenshot('05_web_resignin_paired_devices.png')
const net = page.takeNetwork()
log(`sidebar link: ${linked}; page shown: ${shown}; ${JSON.stringify(facts)}`)
log(`api calls: ${net.api.map((a) => `${a.method} ${a.path} ${a.status}`).join(' | ')}`)
log(`origins: ${net.origins.join(', ')}; sockets: ${JSON.stringify(net.sockets)}; console: ${net.console.length}`)
writeJson('05_web_resignin.json', { at: stamp(), browser: version.product, coreMode, dashboard: dash, sidebarLink: linked, pageShown: shown, facts, screenshot: shot, network: net })
page.close()
