// E2E-S final scan of every output file: (1) the demo password (from the seed constant; never printed), counted
// in every file including images and video (latin1); (2) digit-bounded 6-digit numbers in the text files, each
// hit printed with 30 characters of context so it can be explained; binary files get a count only.
//   node --import tsx 10_secret_scan.mjs <file or folder>...
import { readFileSync, readdirSync, statSync } from 'node:fs'
import { join } from 'node:path'
import { DEMO_OPERATOR_PASSWORD } from '/Users/gsp-budi/Documents/DEMS/packages/shared/src/sim/dataset.ts'

const files = []
for (const p of process.argv.slice(2)) {
  if (statSync(p).isDirectory()) for (const n of readdirSync(p).sort()) files.push(join(p, n))
  else files.push(p)
}
const binary = /\.(png|mp4)$/i
let passwordTotal = 0
const textHits = []
const binaryCounts = []
for (const f of files) {
  const t = readFileSync(f).toString('latin1')
  const pw = t.split(DEMO_OPERATOR_PASSWORD).length - 1
  passwordTotal += pw
  if (pw) console.log(`PASSWORD FOUND in ${f}: ${pw}`)
  const re = /(?<![0-9])[0-9]{6}(?![0-9])/g
  let m, n = 0
  while ((m = re.exec(t)) !== null) {
    n++
    if (!binary.test(f)) textHits.push({ file: f.split('/').pop(), match: m[0], before: t.slice(Math.max(0, m.index - 16), m.index), after: t.slice(m.index + 6, m.index + 14), context: t.slice(Math.max(0, m.index - 30), m.index + 36).replace(/\s+/g, ' ') })
  }
  if (binary.test(f)) binaryCounts.push(`${f.split('/').pop()}:${n}`)
}
console.log(`files scanned: ${files.length}; password occurrences: ${passwordTotal}`)
console.log(`digit-bounded 6-digit numbers in text files: ${textHits.length}`)
const byKind = {}
for (const h of textHits) {
  const before = h.before, after = h.after
  const kind = /AUD-$/.test(before) ? 'audit row ID (AUD-NNNNNN)'
    : /PCD-$/.test(before) ? 'pairing code ID (PCD-NNNNNN; an ID, not the code)'
    : /^ bytes/.test(after) || /"bytes": ?$/.test(before) || /\($/.test(before) && /^ bytes/.test(after) ? 'byte size of a capture file'
    : /\[1006\/$/.test(before) || /\[1006\/[0-9]{6}\.$/.test(before) ? 'Chromium log timestamp (HHMMSS / microseconds)'
    : /hermes-stable:$/.test(before) ? 'JS bundle line number in a stack trace'
    : /[0-9a-f]-?$/i.test(before) && /^-?[0-9a-f]/i.test(after) || /[0-9a-f]{2}-$/i.test(before) || /^-[0-9a-f]{4}/i.test(after) ? 'fragment of a request UUID'
    : /pts:? *$/.test(before) ? 'video timestamp' : 'OTHER'
  byKind[kind] = (byKind[kind] ?? 0) + 1
  if (kind === 'OTHER') console.log(`  OTHER ${h.file}: "${h.context}"`)
}
console.log(`classified: ${JSON.stringify(byKind, null, 1)}`)
console.log(`binary files (raw byte matches inside compressed data, not text): ${binaryCounts.join(' ')}`)
