]
detected = 0
for label, file, old, new, tests in CASES:
    path = f"{S}/{file}"
    shutil.copy2(path, path + '.orig')
    text = open(path).read()
    assert text.count(old) == 1, (label, text.count(old))
    open(path, 'w').write(text.replace(old, new))
    try:
        out = subprocess.run([f"{R}/node_modules/.bin/vitest", 'run', '--root', S, '--project', 'unit', *tests], capture_output=True, text=True).stdout
    finally:
        shutil.move(path + '.orig', path)
    summary = ' '.join(re.findall(r"Tests\s+.*", out))
    failed = [re.sub(r"^\s*FAIL\s+\|unit\|\s*", '', l)[:170] for l in out.split('\n') if l.strip().startswith('FAIL')]
    red = 'failed' in summary
    detected += red
    print(f"{label}: {'DETECTED' if red else 'NOT DETECTED'} — {re.sub(r'  +', ' ', summary)}")
    for f in failed: print(f"     {f}")
print(f"{detected} of {len(CASES)} detected")
