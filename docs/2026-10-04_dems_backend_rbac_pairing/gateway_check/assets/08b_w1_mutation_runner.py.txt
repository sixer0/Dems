def dump():
    with open(os.path.join(HERE, 'mutations.json'), 'w', encoding='utf8') as handle:
        json.dump([{'id': mu['id'], 'edits': mu['edits']} for mu in M], handle, indent=1)

def check():
    dump()
    bad = 0
    for mu in M:
        for edit in mu['edits']:
            text = open(os.path.join(ROOT, edit['file']), encoding='utf8').read()
            for find, replace in edit['pairs']:
                count = text.count(find)
                if count != 1:
                    bad += 1
                    print(f"{mu['id']}: a text occurs {count} times in {edit['file']}: {find[:80]!r}")
                text = text.replace(find, replace, 1)
    ids = [mu['id'] for mu in M]
    assert len(ids) == len(set(ids)), 'duplicate id'
    print(f"{len(M)} mutations; texts that do not occur exactly once: {bad}")
    return bad

def digests(files):
    return {f: hashlib.sha256(open(os.path.join(ROOT, f), 'rb').read()).hexdigest()[:16] for f in sorted(set(files))}

def vitest(mid, tests, config=CONFIG):
    env = dict(os.environ)
    env['W1_APPLIED_DIR'] = APPLIED
    if mid: env['W1_MUT'] = mid
    else: env.pop('W1_MUT', None)
    started = time.time()
    done = subprocess.run(['npx', 'vitest', 'run', '--config', config, '--maxWorkers=2', *tests], cwd=ROOT, env=env, capture_output=True, text=True)
    out = done.stdout + done.stderr
    summary = ' | '.join(line.strip() for line in out.splitlines() if line.strip().startswith(('Test Files', 'Tests ')))
    first = next((line.strip() for line in out.splitlines() if line.strip().startswith(('×', 'FAIL'))), '')
    return done.returncode, summary, first[:180], time.time() - started, out

def run(ids, baseline):
    if check(): sys.exit(2)
    chosen = [mu for mu in M if not ids or mu['id'] in ids]
    log = open(os.path.join(HERE, 'mutations.log'), 'a', encoding='utf8')
    def say(line):
        print(line, flush=True); log.write(line + '\n'); log.flush()
    tree_files = sorted({edit['file'] for mu in chosen for edit in mu['edits']} | set(EXTRA_WATCH))
    before = digests(tree_files)
    say(f"## {time.strftime('%Y-%m-%d %H:%M:%S WIB')} batch {[mu['id'] for mu in chosen]}")
    if baseline:
        for config in sorted({mu['config'] for mu in chosen}):
            tests = sorted({t for mu in chosen if mu['config'] == config for t in mu['tests']})
            code, summary, first, secs, _ = vitest(None, tests, config)
            say(f"baseline (no mutation) {config} exit {code} {summary} ({secs:.0f} s) over {len(tests)} files")
            if code != 0: sys.exit(3)
    detected, survived, missing, unexpected, controls_ok, controls_bad = [], [], [], [], [], []
    for mu in chosen:
        markers = [os.path.join(APPLIED, mu['id'] + '__' + re.sub(r'[^A-Za-z0-9.-]', '_', edit['file'])) for edit in mu['edits']]
        for marker in markers:
            if os.path.exists(marker): os.remove(marker)
        code, summary, first, secs, out = vitest(mu['id'], mu['tests'], mu['config'])
        applied = all(os.path.exists(marker) for marker in markers)
        if not applied: missing.append(mu['id']); verdict = 'NOT APPLIED'
        elif mu['control']:
            if code == 0: controls_ok.append(mu['id']); verdict = 'control ok'
            else: controls_bad.append(mu['id']); verdict = 'CONTROL RED'
        elif code != 0 and (mu['expect'] is None or mu['expect'] in out): detected.append(mu['id']); verdict = 'detected'
        elif code != 0: unexpected.append(mu['id']); verdict = 'RED, OTHER'
        else: survived.append(mu['id']); verdict = 'SURVIVED'
        seen = '' if mu['expect'] is None else (' | expected text seen' if mu['expect'] in out else ' | expected text NOT seen')
        say(f"{mu['id']} {verdict:11} {mu['what']} | {summary} | {first}{seen} ({secs:.0f} s)")
        with open(os.path.join(HERE, f"out_{mu['id']}.txt"), 'w', encoding='utf8') as handle: handle.write(out)
        for marker in markers:
            if os.path.exists(marker): os.remove(marker)
    after = digests(tree_files)
    say(f"## {time.strftime('%Y-%m-%d %H:%M:%S WIB')} {len(chosen)} mutations: {len(detected)} detected, {len(survived)} survived {survived}, {len(unexpected)} red for another reason {unexpected}, {len(missing)} not applied {missing}; controls green {controls_ok}, controls red {controls_bad}")
    say(f"## the mutated files (and watched files) in the tree, before and after: {'identical' if before == after else 'CHANGED'}")
    for f, h in after.items(): say(f"   {h}  {f}")
