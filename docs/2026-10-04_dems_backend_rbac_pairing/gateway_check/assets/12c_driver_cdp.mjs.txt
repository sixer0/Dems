// CF-W evidence driver (copied from FX-W 11b_driver_cdp.mjs): raw CDP over an OS-assigned port, fresh profile in $TMPDIR, origin 127.0.0.1:5178 only.
// The demo password is read from README (Quick start table) inside this process and never printed or written.
import { spawn, execFileSync } from 'node:child_process'
import { existsSync, mkdtempSync, readFileSync, rmSync, writeFileSync } from 'node:fs'
import { join } from 'node:path'

export const ORIGIN = 'http://127.0.0.1:5178'
const SHELL = `${process.env.HOME}/Library/Caches/ms-playwright/chromium_headless_shell-1243/chrome-headless-shell-mac-arm64/chrome-headless-shell`
const README = '/Users/gsp-budi/Documents/DEMS/README.md'

const sleep = (ms) => new Promise((resolve) => setTimeout(resolve, ms))

/** The Supervisor's demo password from the README Quick start table (never logged). */
export function readPassword() {
  const row = readFileSync(README, 'utf8').split('\n').find((line) => /^\|\s*`andi\.pratama`\s*\|/.test(line))
  if (!row) throw new Error('README row for andi.pratama not found')
  const cells = row.split('|').map((cell) => cell.trim())
  const value = cells[4]?.replace(/`/g, '')
  if (!value) throw new Error('README password cell empty')
  return value
}

export async function launch(log) {
  const profile = mkdtempSync(join(process.env.TMPDIR ?? '/tmp', 'cfw_profile_'))
  const child = spawn(
    SHELL,
    [
      '--headless',
      '--remote-debugging-port=0',
      `--user-data-dir=${profile}`,
      '--no-first-run',
      '--no-default-browser-check',
      '--disable-background-networking',
      '--disable-component-update',
      '--disable-sync',
      '--hide-scrollbars',
      '--mute-audio',
      '--force-color-profile=srgb',
      'about:blank',
    ],
    { stdio: ['ignore', 'ignore', 'ignore'] },
  )
  // Never leave the browser behind: whatever ends this process ends the browser it spawned.
  process.on('exit', () => {
    try {
      child.kill('SIGKILL')
    } catch {
      /* gone */
    }
  })
  const portFile = join(profile, 'DevToolsActivePort')
  for (let i = 0; i < 100 && !existsSync(portFile); i += 1) await sleep(100)
  if (!existsSync(portFile)) throw new Error('DevToolsActivePort not written')
  const [portLine, wsPath] = readFileSync(portFile, 'utf8').trim().split('\n')
  const port = Number(portLine)
  // Ownership check: the listening socket on that port belongs to the process this driver spawned.
  const owner = execFileSync('lsof', ['-nP', '-a', '-p', String(child.pid), `-iTCP:${port}`, '-sTCP:LISTEN'], { encoding: 'utf8' })
  if (!owner.includes(String(child.pid))) throw new Error('CDP port is not owned by the spawned browser')
  log(`browser pid ${child.pid}, CDP port ${port} (OS-assigned), owner verified`)
  const cdp = await Cdp.connect(`ws://127.0.0.1:${port}${wsPath}`)
  return {
    cdp,
    async close() {
      try {
        await cdp.send('Browser.close')
      } catch {
        /* closing */
      }
      await sleep(500)
      try {
        child.kill('SIGKILL')
      } catch {
        /* gone */
      }
      rmSync(profile, { recursive: true, force: true })
      log('browser closed, profile removed')
    },
  }
}

export class Cdp {
  static connect(url) {
    return new Promise((resolve, reject) => {
      const ws = new WebSocket(url)
      ws.onopen = () => resolve(new Cdp(ws))
      ws.onerror = (event) => reject(new Error(`ws error ${event.message ?? ''}`))
    })
  }
  constructor(ws) {
    this.ws = ws
    this.id = 0
    this.pending = new Map()
    this.listeners = new Set()
    ws.onmessage = (event) => {
      const message = JSON.parse(event.data)
      if (message.id !== undefined) {
        const entry = this.pending.get(message.id)
        if (!entry) return
        this.pending.delete(message.id)
        if (message.error) entry.reject(new Error(`${entry.method}: ${message.error.message}`))
        else entry.resolve(message.result)
      } else {
        for (const listener of this.listeners) listener(message)
      }
    }
  }
  send(method, params = {}, sessionId) {
    const id = ++this.id
    return new Promise((resolve, reject) => {
      this.pending.set(id, { resolve, reject, method })
      this.ws.send(JSON.stringify({ id, method, params, ...(sessionId ? { sessionId } : {}) }))
    })
  }
  on(listener) {
    this.listeners.add(listener)
    return () => this.listeners.delete(listener)
  }
}

/** One page target with helpers. Every request to another origin than 127.0.0.1:5178 is failed. */
export async function openPage(cdp, log) {
  const { targetId } = await cdp.send('Target.createTarget', { url: 'about:blank' })
  const { sessionId } = await cdp.send('Target.attachToTarget', { targetId, flatten: true })
  const s = (method, params) => cdp.send(method, params, sessionId)
  const exceptions = []
  const blocked = []
  cdp.on((message) => {
    if (message.sessionId !== sessionId) return
    if (message.method === 'Runtime.exceptionThrown') exceptions.push(message.params.exceptionDetails?.exception?.description?.split('\n')[0] ?? 'exception')
    if (message.method === 'Fetch.requestPaused' && message.params.responseStatusCode !== undefined) {
      // Response stage (CF-W): only GET /api/v1/evidence/{id} answers pause here; the page's hook may
      // rewrite the answer in THIS browser only; otherwise it continues unchanged.
      const hook = page.responseHook
      if (hook) hook(message.params, s).catch(() => s('Fetch.continueRequest', { requestId: message.params.requestId }).catch(() => undefined))
      else s('Fetch.continueRequest', { requestId: message.params.requestId }).catch(() => undefined)
      return
    }
    if (message.method === 'Fetch.requestPaused') {
      const url = message.params.request.url
      let allowed = false
      try {
        const parsed = new URL(url)
        allowed = parsed.origin === ORIGIN || parsed.protocol === 'data:' || parsed.protocol === 'blob:'
      } catch {
        allowed = false
      }
      if (allowed) s('Fetch.continueRequest', { requestId: message.params.requestId }).catch(() => undefined)
      else {
        blocked.push(url.slice(0, 80))
        s('Fetch.failRequest', { requestId: message.params.requestId, errorReason: 'BlockedByClient' }).catch(() => undefined)
      }
    }
  })
  await s('Page.enable')
  await s('Runtime.enable')
  await s('Fetch.enable', { patterns: [{ urlPattern: '*', requestStage: 'Request' }, { urlPattern: `${ORIGIN}/api/v1/evidence/EV-*`, requestStage: 'Response' }] })
  const page = {
    responseHook: null,
    sessionId,
    exceptions,
    blocked,
    send: s,
    async viewport(width, height, mobile = false) {
      await s('Emulation.setDeviceMetricsOverride', { width, height, deviceScaleFactor: 1, mobile })
      log(`viewport ${width}x${height}`)
    },
    async goto(path) {
      await s('Page.navigate', { url: `${ORIGIN}${path}` })
      await page.waitFor(() => document.readyState === 'complete', 20000)
    },
    async eval(fn, ...args) {
      const expression = `(${fn.toString()})(...${JSON.stringify(args)})`
      const result = await s('Runtime.evaluate', { expression, awaitPromise: true, returnByValue: true })
      if (result.exceptionDetails) throw new Error(`eval failed: ${result.exceptionDetails.exception?.description ?? result.exceptionDetails.text}`)
      return result.result.value
    },
    async waitFor(fn, timeout = 15000, ...args) {
      const start = Date.now()
      for (;;) {
        try {
          if (await page.eval(fn, ...args)) return true
        } catch {
          /* page navigating */
        }
        if (Date.now() - start > timeout) throw new Error(`timeout waiting for ${fn.toString().slice(0, 120)}`)
        await sleep(200)
      }
    },
    async shot(file, clip) {
      const params = { format: 'png', captureBeyondViewport: false }
      if (clip) params.clip = { ...clip, scale: 1 }
      const { data } = await s('Page.captureScreenshot', params)
      writeFileSync(file, Buffer.from(data, 'base64'))
      log(`screenshot ${file.split('/').pop()}`)
    },
    async clickSelector(selector) {
      const box = await page.eval((sel) => {
        const element = document.querySelector(sel)
        if (!element) return null
        element.scrollIntoView({ block: 'center', inline: 'center' })
        const rect = element.getBoundingClientRect()
        return { x: rect.left + rect.width / 2, y: rect.top + rect.height / 2 }
      }, selector)
      if (!box) throw new Error(`no element ${selector}`)
      await page.clickAt(box.x, box.y)
    },
    async clickAt(x, y) {
      await s('Input.dispatchMouseEvent', { type: 'mouseMoved', x, y })
      await s('Input.dispatchMouseEvent', { type: 'mousePressed', x, y, button: 'left', clickCount: 1 })
      await s('Input.dispatchMouseEvent', { type: 'mouseReleased', x, y, button: 'left', clickCount: 1 })
    },
    async key(key, code = key) {
      await s('Input.dispatchKeyEvent', { type: 'keyDown', key, code, windowsVirtualKeyCode: key === 'Escape' ? 27 : key === 'Enter' ? 13 : 0 })
      await s('Input.dispatchKeyEvent', { type: 'keyUp', key, code, windowsVirtualKeyCode: key === 'Escape' ? 27 : key === 'Enter' ? 13 : 0 })
    },
    async type(text) {
      await s('Input.insertText', { text })
    },
  }
  return page
}

/** Signs andi.pratama in through the Login form in Live mode. */
export async function signIn(page, log) {
  await page.goto('/login')
  await page.waitFor(() => !!document.querySelector('[data-testid="login-username-field"] input'), 20000)
  await page.waitFor(() => {
    const button = document.querySelector('[data-testid="login-submit"]')
    return !!button && !button.hasAttribute('disabled')
  }, 20000)
  await page.clickSelector('[data-testid="login-username-field"] input')
  await page.type('andi.pratama')
  await page.clickSelector('[data-testid="login-password-field"] input')
  await page.type(readPassword())
  await page.clickSelector('[data-testid="login-submit"]')
  await page.waitFor(() => location.pathname.startsWith('/dashboard'), 20000)
  log('signed in as andi.pratama (SUPERVISOR); landed on /dashboard')
}

export async function signOut(page, log) {
  const status = await page.eval(async () => {
    const response = await fetch('/api/v1/auth/sign-out', { method: 'POST', headers: { 'content-type': 'application/json', accept: 'application/json' }, body: '{}' })
    return response.status
  })
  log(`sign-out POST /api/v1/auth/sign-out -> ${status}`)
}
