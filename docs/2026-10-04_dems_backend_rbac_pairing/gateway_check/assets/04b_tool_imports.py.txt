#!/usr/bin/env python3
"""Lists relative imports of the core-server sources: deep imports across units, and the unit graph."""
import os, re, sys, json
root = sys.argv[1]
mode = sys.argv[2] if len(sys.argv) > 2 else 'deep'
IMPORT = re.compile(r"""(?:^|\n)\s*(?:import|export)\s+(type\s+)?(?:[^'";]*?\s+from\s+)?['"](\.{1,2}/[^'"]*)['"]""")
def unit_of(path):
    p = path.split('/')
    if p[0] != 'src': return p[0] + ('/' + p[1] if len(p) > 2 else '')
    if len(p) == 2: return 'src/' + p[1]
    if p[1] in ('plugins', 'modules', 'db'):
        return '/'.join(p[:3]) if len(p) > 3 else '/'.join(p[:3])
    return '/'.join(p[:2])
def is_test(path):
    return '__tests__/' in path or path.endswith('.test.ts') or path.startswith('test/')
files = []
for base in ('src', 'test'):
    for d, _, names in os.walk(os.path.join(root, base)):
        if 'node_modules' in d or 'selfcheck' in d: continue
        for n in names:
            if n.endswith('.ts'): files.append(os.path.relpath(os.path.join(d, n), root))
out = []
for f in sorted(files):
    text = open(os.path.join(root, f)).read()
    for m in IMPORT.finditer(text):
        spec = m.group(2)
        line = text.count('\n', 0, m.start(2)) + 1
        target = os.path.normpath(os.path.join(os.path.dirname(f), spec))
        if os.path.isdir(os.path.join(root, target)): resolved, via_index = target + '/index.ts', True
        elif os.path.isfile(os.path.join(root, target + '.ts')): resolved, via_index = target + '.ts', False
        else: resolved, via_index = target + ' (MISSING)', False
        out.append(dict(file=f, line=line, spec=spec, resolved=resolved, via_index=via_index, type_only=bool(m.group(1)), test=is_test(f), ttest=is_test(resolved)))
if mode == 'deep':
    for o in out:
        if o['test']: continue
        a, b = unit_of(o['file']), unit_of(o['resolved'])
        if a == b: continue
        unit_dir = b
        is_index = o['resolved'] == unit_dir + '/index.ts' or o['resolved'] == unit_dir + '.ts' or unit_dir.count('/') == 1 and o['resolved'].count('/') == 1
        if not is_index:
            print(f"{o['file']}:{o['line']} -> {o['resolved']}{' (type)' if o['type_only'] else ''}")
elif mode == 'json':
    json.dump(out, sys.stdout)
elif mode == 'testsupport':
    for o in out:
        if o['ttest'] and not o['resolved'].endswith('.test.ts'):
            a = o['file']
            if os.path.dirname(a) != os.path.dirname(o['resolved']):
                print(f"{o['file']}:{o['line']} -> {o['resolved']}")
elif mode == 'prodtotest':
    for o in out:
        if not o['test'] and o['ttest']: print(f"{o['file']}:{o['line']} -> {o['resolved']}")
