// W1 mutation runner (W0's pattern, `.cache/w0/mut/vitest.mutation.config.mts`; rule M15: a mutation
// never changes a file of the workspace tree). The app's own Vite / Vitest configuration, plus ONE plugin
// that replaces texts in the named modules while they are transformed, in memory. Which mutation: the environment
// variable W1_MUT (an id of mutations.json); unset: no mutation (the baseline). A mutation holds edits,
// each one module and one or more [find, replace] pairs; each find must occur exactly once in that module.
import { readFileSync, writeFileSync } from 'node:fs'
import { fileURLToPath, URL } from 'node:url'
import { mergeConfig } from 'vite'
import appConfig from '../../../apps/core-web/vite.config.ts'

const here = fileURLToPath(new URL('./', import.meta.url))
const web = fileURLToPath(new URL('../../../apps/core-web/', import.meta.url))
const table = JSON.parse(readFileSync(`${here}mutations.json`, 'utf8'))
const id = process.env.W1_MUT ?? ''
const mutation = id === '' ? null : table.find((entry: { id: string }) => entry.id === id)
if (id !== '' && !mutation) throw new Error(`W1 mutation "${id}" is not in mutations.json`)

function mutationPlugin() {
  return {
    name: 'w1-mutation',
    enforce: 'pre' as const,
    transform(code: string, moduleId: string) {
      if (!mutation) return null
      const file = moduleId.split('?')[0] ?? ''
      const edit = (mutation.edits as { file: string; pairs: [string, string][] }[]).find((entry) => file.endsWith(entry.file))
      if (!edit) return null
      let next = code
      for (const [find, replace] of edit.pairs) {
        const count = next.split(find).length - 1
        if (count !== 1) throw new Error(`W1 mutation ${mutation.id}: the text to replace occurs ${count} times in ${edit.file}`)
        next = next.replace(find, () => replace)
      }
      // The marker that this edit really ran, in the directory the runner names (under TMPDIR).
      writeFileSync(`${process.env.W1_APPLIED_DIR}/${mutation.id}__${edit.file.replace(/[^A-Za-z0-9.-]/g, '_')}`, 'applied\n')
      return { code: next, map: null }
    },
  }
}

const base = typeof appConfig === 'function' ? await appConfig({ command: 'serve', mode: 'test', isPreview: false, isSsrBuild: false }) : appConfig

// The mutation plugin stands FIRST, so it sees the source text as it is on disk.
const { plugins = [], ...rest } = base
export default {
  ...mergeConfig(rest, { root: web, cacheDir: `${here}vite-cache`, test: { root: web } }),
  plugins: [mutationPlugin(), ...plugins],
}
