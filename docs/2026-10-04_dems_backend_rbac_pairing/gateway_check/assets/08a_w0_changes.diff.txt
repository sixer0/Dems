+/**
+ * The one way to start a Core command (or a query refresh) from an effect (plan row W0, item 1;
+ * finding F69-1, decision D158 DC-2).
+ *
+ * The console renders under React StrictMode (`main.tsx`), and Live mode exists only in dev builds,
+ * where StrictMode runs every effect twice on mount (set-up, clean-up, set-up). A command sent from
+ * the set-up is therefore sent TWICE in every Live session; in Simulated mode the simulator answers
+ * inside the first set-up, so a guard kept in a ref is already cleared when the second one runs.
+ *
+ * `deferEffectStart(start)` schedules `start` one task later and returns the clean-up that cancels
+ * it. StrictMode's clean-up runs before that task, so only the second set-up's start remains: one
+ * command. A production build (no double run) starts one task later than a synchronous call, which
+ * no operator can see.
+ *
+ * Use it as the effect's return value, after the effect's own conditions:
+ *
+ * ```ts
+ * useEffect(() => {
+ *   if (!wanted) return undefined
+ *   return deferEffectStart(() => {
+ *     settle(() => actions.verify(id), { fail: (error) => { if (presentCoreError(error) !== null) return; ... } })
+ *   })
+ * }, [wanted, id])
+ * ```
+ *
+ * A change of the effect's dependencies within that task cancels the pending start and schedules
+ * the new one, so a guard that must hold for the command itself (an in-flight marker) is checked and
+ * set inside `start`. `start` handles its own failures (`settle`'s `fail`, the presenter first):
+ * nothing here catches. `src/test/live-strict-mode-starts.test.tsx` fails the workspace when a file
+ * under `features/` starts a command from an effect without returning this helper.
+ */
+export function deferEffectStart(start: () => void): () => void {
+  const handle = setTimeout(start, 0)
+  return () => clearTimeout(handle)
+}

+ * | Phase | From | What the control shows | A press |
+ * |-------|------|------------------------|---------|
+ * | idle | — | its own look | runs the command |
+ * | busy | the press | no visual change; `aria-busy="true"` on the element of `ref` (when the caller attaches it) | is ignored: one command per control (`run` returns `false`, nothing is sent) |

+ * - `run(start, key?)`: `start` gets `release` and starts the command itself, so the call site keeps
+ *   rule 1's literal shape `settle(() => actions.x(…), { ok, fail, done: release })` (the static
+ *   rules of the page rows look for it), or runs several settles in order and calls `release` once
+ *   at the end (the export wizard).

+ * `aria-busy` PROP never reaches the DOM. `ref` writes the attribute on the pressed control itself

+ * removes it. A caller that marks a container instead reads `busy`.

+  onWait?: () => void

+export interface PendingControl {
+  /** From the press until the command settled: further presses are ignored. */
+  readonly busy: boolean
+  /** Busy for longer than the reveal time: the control shows the kit's pending state. */
+  readonly pending: boolean
+  /** What the command in flight is about (`key` of `run`), or `null`. */
+  readonly key: string | null
+  /**
+   * Starts one command unless this control has one in flight. `start` gets `release`, which must run
+   * once when the command settled (pass it as `settle`'s `done`). Returns whether it started.
+   */
+  readonly run: (start: (release: () => void) => void, key?: string) => boolean
+  /** `settle(action, handlers)` on this control, unless a command of it is out. Returns whether it was sent. */
+  readonly settle: <T>(action: () => T | PromiseLike<T>, handlers: PendingHandlers<T>) => boolean
+  /** For the pressed control's element: carries `aria-busy="true"` while a command is in flight. */
+  readonly ref: RefObject<HTMLButtonElement | null>
+}

+/**
+ * Whether the pending look of a busy control is shown: only after it has been busy for
+ * `LIVE_PENDING_REVEAL_MS`, scaled by the motion policy as every other duration (LB-1). A command
+ * that settles sooner never changes the look.
+ */
+export function usePendingReveal(busy: boolean): boolean {
+  const { timeScale } = useMotionPolicy()
+  const delayMs = LIVE_PENDING_REVEAL_MS * timeScale
+  const [revealed, setRevealed] = useState(false)
+  useEffect(() => {
+    if (!busy) return undefined
+    const timer = setTimeout(() => setRevealed(true), delayMs)
+    return () => {
+      clearTimeout(timer)
+      setRevealed(false)
+    }
+  }, [busy, delayMs])
+  return busy && revealed
+}
+
+export function usePendingControl(): PendingControl {
+  const ref = useRef<HTMLButtonElement | null>(null)
+  const inFlight = useRef(false)
+  const [busy, setBusy] = useState(false)
+  const [key, setKey] = useState<string | null>(null)
+  const pending = usePendingReveal(busy)
+
+  useLayoutEffect(() => {
+    const element = ref.current
+    if (element === null) return
+    if (busy) element.setAttribute('aria-busy', 'true')
+    else element.removeAttribute('aria-busy')
+  }, [busy])
+
+  const begin = useCallback((start: (release: () => void) => void, actionKey: string | undefined, onWait: (() => void) | undefined): boolean => {
+    if (inFlight.current) return false
+    inFlight.current = true
+    let released = false
+    let waiting = false
+    const release = (): void => {
+      if (released) return
+      released = true
+      inFlight.current = false
+      if (waiting) {
+        setBusy(false)
+        setKey(null)
+      }
+    }
+    try {
+      start(release)
+    } catch (error) {
+      release()
+      throw error
+    }
+    // Settled already (Simulated mode, or a command refused before it was sent): nothing to show.
+    if (!released) {
+      waiting = true
+      setBusy(true)
+      setKey(actionKey ?? null)
+      onWait?.()
+    }
+    return true
+  }, [])
+
+  const run = useCallback((start: (release: () => void) => void, actionKey?: string): boolean => begin(start, actionKey, undefined), [begin])
+
+  const settle = useCallback(
+    <T,>(action: () => T | PromiseLike<T>, handlers: PendingHandlers<T>): boolean =>
+      begin(
+        (release) => {
+          settleResult(action, {
+            ...(handlers.ok ? { ok: handlers.ok } : {}),
+            fail: handlers.fail,
+            done: () => {
+              release()
+              handlers.done?.()
+            },
+          })
+        },
+        undefined,
+        handlers.onWait,
+      ),
+    [begin],
+  )
+
+  return { busy, pending, key, run, settle, ref }

+/** The text from `open` (an index just after an opening parenthesis) to its closing one. */
+function balanced(text: string, open: number): string {
+  let depth = 1
+  let index = open
+  while (index < text.length && depth > 0) {
+    const char = text[index]
+    if (char === '(') depth += 1
+    else if (char === ')') depth -= 1
+    index += 1
+  }
+  return text.slice(open, index - 1)
+}

+/** The callback text with every `deferEffectStart(...)` argument removed. */
+function outsideHelper(body: string): string {
+  let rest = body
+  let at = rest.search(/\bdeferEffectStart\s*\(/)

+    const open = rest.indexOf('(', at) + 1
+    const inner = balanced(rest, open)
+    rest = `${rest.slice(0, at)}HELPER${rest.slice(open + inner.length + 1)}`
+    at = rest.search(/\bdeferEffectStart\s*\(/)
+  }
+  return rest

+    const line = text.slice(0, match.index).split('\n').length
+    const outside = outsideHelper(body).match(CORE_CALL)
+    if (outside) {
+      found.push({ line, token: outside[0].replace(/\s+/g, ''), problem: 'outside the helper' })
+      continue
+    }

+    if (inside && !/(?:\breturn\s+|=>\s*)deferEffectStart\s*\(/.test(body)) {
+      found.push({ line, token: inside[0].replace(/\s+/g, ''), problem: 'the helper is not returned' })
+    }
+  }
