import sys
S = sys.argv[1]
def read(path): return open(f"{S}/{path}").read()
def write(path, text): open(f"{S}/{path}", 'w').write(text)
def sub(path, old, new, count=1):
    t = read(path)
    if t.count(old) != count: raise SystemExit(f"{path}: expected {count} occurrence(s) of {old!r}, found {t.count(old)}")
    write(path, t.replace(old, new))
def cut(path, start, end):
    """Removes the text from `start` up to (not including) `end` and returns it."""
    t = read(path)
    if t.count(start) != 1 or t.count(end) < 1: raise SystemExit(f"{path}: cut markers not found once: {start!r} / {end!r}")
    i = t.index(start); j = t.index(end, i)
    write(path, t[:i] + t[j:])
    return t[i:j]
