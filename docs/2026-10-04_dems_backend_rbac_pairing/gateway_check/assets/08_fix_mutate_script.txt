def run(tests: list[str]) -> tuple[int, int]:
    """Runs the named files on the scratch tree; returns (failed, total) tests."""
    proc = subprocess.run(
        ['npx', 'vitest', 'run', *tests, '--maxWorkers=2'],
        cwd=str(WEB), capture_output=True, text=True, timeout=1_800,
    )
    out = proc.stdout + proc.stderr
    lines = [ln for ln in out.splitlines() if re.match(r'\s*Tests\s+\d', ln)]
    if not lines:
        raise RuntimeError('no runner summary:\n' + out[-3000:])
    summary = lines[-1]
    failed = sum(int(m) for m in re.findall(r'(\d+) failed', summary))
    passed = sum(int(m) for m in re.findall(r'(\d+) passed', summary))
    return failed, failed + passed


def main() -> int:
    wanted = set(sys.argv[1:])
    detected = 0
    rows = []
    for mid, rel, old, new, tests in MUTATIONS:
        if wanted and mid.split()[0] not in wanted:
            continue
        path = WEB / rel
        text = path.read_text()
        if text.count(old) != 1:
            rows.append((mid, f'SKIPPED (anchor found {text.count(old)} times)'))
            continue
        path.write_text(text.replace(old, new, 1))
        try:
            failed, total = run(tests)
        finally:
            path.write_text(text)
        ok = failed > 0
        detected += 1 if ok else 0
        rows.append((mid, f'{"DETECTED" if ok else "SURVIVED"} ({failed} of {total} tests fail)'))
        print(f'{mid:78s} {"DETECTED" if ok else "SURVIVED"} ({failed}/{total})', flush=True)
    print(f'\n{detected} of {len(rows)} mutations detected.')
    for mid, verdict in rows:
        print(f'  {verdict:28s} {mid}')
    return 0


if __name__ == '__main__':
    sys.exit(main())
