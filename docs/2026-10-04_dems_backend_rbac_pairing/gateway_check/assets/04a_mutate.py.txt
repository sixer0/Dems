def run(project, tests):
    command = ['node', 'node_modules/vitest/vitest.mjs', 'run', '--root', str(ROOT), '--project', project, '--maxWorkers=2', *tests]
    env = None
    if project == 'db':
        import os
        env = dict(os.environ, DEMS_TEST_STEP='60a')
    done = subprocess.run(command, cwd=REPO, capture_output=True, text=True, env=env)
    out = done.stdout + done.stderr
    tests_line = re.findall(r'^\s+Tests\s+(.*)$', out, flags=re.M)
    files_line = re.findall(r'^\s+Test Files\s+(.*)$', out, flags=re.M)
    failed = re.search(r'(\d+) failed', tests_line[-1]) if tests_line else None
    return done.returncode, (int(failed.group(1)) if failed else 0), (files_line[-1].strip() if files_line else 'no summary'), (tests_line[-1].strip() if tests_line else 'no summary')


def main():
    project = sys.argv[1]
    wanted = sys.argv[2:]
    detected = 0
    total = 0
    for name, path, old, new, where, tests in MUTATIONS:
        if where != project:
            continue
        if wanted and name.split(' ')[0] not in wanted:
            continue
        file = ROOT / path
        original = file.read_text()
        if original.count(old) != 1:
            print(f'{name}: NOT APPLIED (the text occurs {original.count(old)} times)', flush=True)
            total += 1
            continue
        file.write_text(original.replace(old, new))
        try:
            code, failed, files, summary = run(project, tests)
        finally:
            file.write_text(original)
        total += 1
        verdict = 'DETECTED' if code != 0 and (failed > 0 or 'failed' in files) else 'NOT DETECTED'
        if verdict == 'DETECTED':
            detected += 1
        print(f'{name}: {verdict} — exit {code}; files {files}; tests {summary}', flush=True)
    print(f'{detected} of {total} mutations detected ({project})', flush=True)


main()
