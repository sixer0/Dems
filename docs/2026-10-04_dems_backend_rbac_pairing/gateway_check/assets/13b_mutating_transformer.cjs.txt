// CF2-B mutation check (copy of CF-M2's transformer): a jest transformer that changes texts of source files IN MEMORY and hands the
// result to the app's own transformer (babel-jest with jest-expo's options). Nothing in the tree is
// edited (the method of FX-M2, gateway_check/assets/11d_fxm2b_mutating_transformer.cjs.txt).
// Spec file named by CF2B_MUT_SPEC: { "id", "files": [{ "file", "edits": [{ "from": "<occurs exactly once>", "to" }] } | { "file", "replaceWith" }] }
const path = require('node:path')
const fs = require('node:fs')
const crypto = require('node:crypto')
const ROOT = '/Users/gsp-budi/Documents/DEMS'
const APP = path.join(ROOT, 'apps/mobile')
const babelJest = require(path.join(ROOT, 'node_modules/babel-jest'))
const { resolveBabelOptions } = require(path.join(ROOT, 'node_modules/jest-expo/src/resolveBabelOptions'))
const inner = (babelJest.default ?? babelJest).createTransformer(resolveBabelOptions(APP))

const specPath = process.env.CF2B_MUT_SPEC
const spec = specPath ? JSON.parse(fs.readFileSync(specPath, 'utf8')) : null
const parts = spec ? spec.files : []
const targets = new Map(parts.map((part) => [path.join(APP, part.file), part]))
const specHash = spec ? crypto.createHash('sha256').update(JSON.stringify(spec)).digest('hex') : 'none'

function mutate(source, filename) {
  const part = targets.get(filename)
  if (!part) return source
  if (part.replaceWith) return fs.readFileSync(part.replaceWith, 'utf8')
  let text = source
  for (const edit of part.edits) {
    const count = text.split(edit.from).length - 1
    if (count !== 1) throw new Error(`CF2B mutation ${spec.id}: a text occurs ${count} times in ${part.file} (expected exactly 1)`)
    text = text.replace(edit.from, () => edit.to)
  }
  return text
}

module.exports = {
  canInstrument: false,
  process(source, filename, options) {
    return inner.process(mutate(source, filename), filename, options)
  },
  getCacheKey(source, filename, options) {
    const key = inner.getCacheKey(source, filename, options)
    return targets.has(filename) ? crypto.createHash('sha256').update(key).update(specHash).digest('hex') : key
  },
}
