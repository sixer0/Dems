Each mutant changes texts of ONE production file IN MEMORY (the jest transformer beside this file) and runs
the test files mapped to it. DETECTED = jest exits non-zero. Before a mutant runs, each `from` text is checked
to occur exactly once in the file (otherwise: NOT APPLIED, never counted as detected). The SHA-256 of every
mutated file is compared before and after the whole run. 'control' runs the unmutated tests once.

  python3 mutate.py [id ...]
"""

def sha(path):
    with open(path, 'rb') as f:
        return hashlib.sha256(f.read()).hexdigest()


def run(tests, spec):
    env = dict(os.environ)
    if spec is None:
        env.pop('CFM1_MUT_SPEC', None)
    else:
        with open(SPEC, 'w') as f:
            json.dump(spec, f)
        env['CFM1_MUT_SPEC'] = SPEC
    t = time.time()
    proc = subprocess.run(['npx', 'jest', '--config', CONFIG, '--maxWorkers=2', '--silent', *tests], cwd=APP, env=env, capture_output=True, text=True)
    out = proc.stdout + proc.stderr
    summary = [l.strip() for l in out.splitlines() if l.strip().startswith('Tests:')]
    applied_error = 'CF-M1 mutation' in out and 'occurs' in out
    return proc.returncode, (summary[-1] if summary else '(no summary)'), round(time.time() - t, 1), applied_error


def main():
    wanted = sys.argv[1:]
    files = sorted({m['file'] for m in MUTANTS})
    before = {f: sha(os.path.join(APP, f)) for f in files}
    results = []
    if not wanted or 'control' in wanted:
        code, summary, secs, _ = run(ALL, None)
        print(f"control (unmutated, {len(ALL)} files): exit {code}; {summary}; {secs} s", flush=True)
        results.append(('control', code, summary))
    for m in MUTANTS:
        if wanted and m['id'].split(' ')[0] not in wanted and 'all' not in wanted:
            continue
        text = open(os.path.join(APP, m['file'])).read()
        bad = [x['from'][:60] for x in m['edits'] if text.count(x['from']) != 1]
        if bad:
            print(f"{m['id']}: NOT APPLIED (text not found exactly once: {bad})", flush=True)
            results.append((m['id'], None, 'not applied'))
            continue
        code, summary, secs, applied_error = run(m['tests'], {'id': m['id'], 'file': m['file'], 'edits': m['edits']})
        verdict = 'NOT APPLIED' if applied_error else ('DETECTED' if code != 0 else 'MISSED')
        print(f"{m['id']} [{m['file']}] -> {verdict} (jest exit {code}; {summary}; {secs} s)", flush=True)
        results.append((m['id'], code, verdict))
    after = {f: sha(os.path.join(APP, f)) for f in files}
    print('tree unchanged by the run:', all(before[f] == after[f] for f in files), flush=True)
    for f in files:
        print(f'  {before[f][:16]} {f}')


if __name__ == '__main__':
    main()
