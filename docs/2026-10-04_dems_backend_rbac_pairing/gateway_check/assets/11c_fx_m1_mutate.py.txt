`from` text is checked to occur exactly once in the file (otherwise: NOT APPLIED, never counted as
detected). The SHA-256 of every mutated file is compared before and after the whole run.

  python3 mutate.py [id ...]      (no id: all; 'control' runs the unmutated tests once)
"""

def sha(path):
    with open(APP + '/' + path, 'rb') as fh:
        return hashlib.sha256(fh.read()).hexdigest()

def run(tests, spec):
    env = dict(os.environ)
    if spec is None:
        env.pop('FXM1_MUT_SPEC', None)
    else:
        with open(SPEC, 'w') as fh:
            json.dump(spec, fh)
        env['FXM1_MUT_SPEC'] = SPEC
    started = time.time()
    proc = subprocess.run(['npx', 'jest', '--config', CONFIG, '--maxWorkers=2', '--silent', *tests], cwd=APP, env=env, capture_output=True, text=True)
    out = proc.stdout + proc.stderr
    summary = [line.strip() for line in out.splitlines() if line.strip().startswith('Tests:')]
    failing = sorted({line.strip()[2:].strip() for line in out.splitlines() if line.strip().startswith('● ') and '›' in line})
    return proc.returncode, (summary[-1] if summary else '(no summary)'), failing, round(time.time() - started)

def load():
    return os.getloadavg()[0]

def main():
    ids = sys.argv[1:]
    files = sorted({m[1] for m in M})
    before = {f: sha(f) for f in files}
    print(f'# FX-M1 mutation check, {time.strftime("%Y-%m-%d %H:%M:%S")}; the tree is not edited')
    if not ids or 'control' in ids:
        code, summary, failing, secs = run(ALL, None)
        print(f'control (no mutation): exit={code} {summary} ({secs} s)')
    detected = missed = skipped = 0
    for mid, path, what, tests, edits in M:
        if ids and mid not in ids:
            continue
        while load() > 14:
            print(f'   (load {load():.1f} > 14: waiting)', flush=True)
            time.sleep(20)
        text = open(APP + '/' + path).read()
        counts = [text.count(ed['from']) for ed in edits]
        if any(c != 1 for c in counts):
            skipped += 1
            print(f'{mid} NOT APPLIED ({path}: occurrences {counts}) — {what}')
            continue
        code, summary, failing, secs = run(tests, {'id': mid, 'file': path, 'edits': edits})
        ok = code != 0
        detected += ok
        missed += not ok
        print(f'{mid} {"DETECTED" if ok else "NOT DETECTED"} exit={code} {summary} ({secs} s) — {what} [{path}]', flush=True)
        for name in failing[:4]:
            print(f'     fails: {name}')
    after = {f: sha(f) for f in files}
    print(f'## {detected} detected, {missed} not detected, {skipped} not applied')
    print('## the mutated files in the tree, before and after:', 'identical' if before == after else 'CHANGED')
    for f in files:
        print(f'   {after[f][:16]}  {f}')
    print(f'# finished {time.strftime("%H:%M:%S")}')

main()
