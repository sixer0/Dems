#!/bin/sh
# Focus pairing-code slot 0 on the Pair screen of ONE device (bounds from a UI dump of Pair; allowed) and check the
# keypad is up. No digits typed here.   sh prep_pair.sh <serial> <prefix>
set -u
S=$1; P=$2
case "$S" in emulator-5554|emulator-5556|emulator-5558) ;; *) echo "serial refused"; exit 64;; esac
OUT=/Users/gsp-budi/Documents/DEMS/docs/2026-10-04_dems_backend_rbac_pairing/verification/assets/08_performance_realtime
APP=com.dems.demo.tablet
a() { adb -s "$S" "$@"; }
now() { python3 -c 'import time; t=time.time(); print(time.strftime("%H:%M:%S", time.localtime(t)) + ".%03d" % int((t % 1) * 1000))'; }
kg() { a shell dumpsys window 2>/dev/null | tr -d '\r' | grep -o 'isKeyguardShowing=[a-z]*' | head -n 1; }
echo "== prep pair $S — host $(now); $(kg); load $(uptime | sed 's/.*averages: //')"
case "$(kg)" in *true*) echo "KEYGUARD: stop"; exit 77;; esac
for try in 1 2 3; do
  a shell uiautomator dump /sdcard/w98_ui.xml >/dev/null 2>&1; a shell cat /sdcard/w98_ui.xml > "$OUT/${P}_ui_pair.xml" 2>/dev/null; a shell rm -f /sdcard/w98_ui.xml
  grep -q 'resource-id="pairing-code-slot-0"' "$OUT/${P}_ui_pair.xml" && break; sleep 2
done
echo "   UI dump ${P}_ui_pair.xml: screen ids [$(grep -o 'resource-id="screen-[a-z-]*"' "$OUT/${P}_ui_pair.xml" | sed 's/resource-id=//; s/"//g' | sort -u | tr '\n' ' ')]"
case "$(grep -o 'resource-id="screen-[a-z-]*"' "$OUT/${P}_ui_pair.xml")" in *screen-pair*) ;; *) echo "-- not on Pair: nothing tapped"; exit 3;; esac
B=$(grep -o '<node[^>]*resource-id="pairing-code-slot-0"[^>]*>' "$OUT/${P}_ui_pair.xml" | grep -o 'bounds="[^"]*"' | head -n 1)
X=$(echo "$B" | sed -E 's/bounds="\[([0-9]+),([0-9]+)\]\[([0-9]+),([0-9]+)\]"/\1 \3/' | awk '{print int(($1+$2)/2)}')
Y=$(echo "$B" | sed -E 's/bounds="\[([0-9]+),([0-9]+)\]\[([0-9]+),([0-9]+)\]"/\2 \4/' | awk '{print int(($1+$2)/2)}')
echo "   pairing-code-slot-0 $B -> tap $X $Y at host $(now)"
a shell input tap "$X" "$Y"; sleep 2
IME=$(a shell dumpsys input_method | tr -d '\r' | grep -o 'mInputShown=[a-z]*' | head -n 1)
if [ "$IME" != "mInputShown=true" ]; then echo "   $IME after the first tap; tap again"; a shell input tap "$X" "$Y"; sleep 2; IME=$(a shell dumpsys input_method | tr -d '\r' | grep -o 'mInputShown=[a-z]*' | head -n 1); fi
a exec-out screencap -p > "$OUT/${P}_phone_code_field_focused.png"
echo "   screencap ${P}_phone_code_field_focused.png at host $(now) (empty slots); IME $IME; $(kg)"
