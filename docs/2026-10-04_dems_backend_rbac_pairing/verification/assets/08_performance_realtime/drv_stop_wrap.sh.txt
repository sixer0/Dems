#!/bin/sh
# Web stop while the TABLET (emulator-5556) shows the monitor: 864x540 --bugreport screen recording + streamed
# ReactNativeJS log on the tablet around web_stop.mjs. No setting, lock or reverse rule touched.
#   sh stop_wrap.sh <prefix> "<reason>"
set -u
OUT=/Users/gsp-budi/Documents/DEMS/docs/2026-10-04_dems_backend_rbac_pairing/verification/assets/08_performance_realtime
SP=/private/tmp/claude-501/-Users-gsp-budi-Documents-DEMS/0d71fd65-4a51-4019-9c48-5c92d11bbdaf/scratchpad/w98
P=$1; R=$2
a() { adb -s emulator-5556 "$@"; }
case "$(a shell dumpsys window | tr -d '\r' | grep -o 'isKeyguardShowing=[a-z]*' | head -n 1)" in *true*) echo "KEYGUARD: stop"; exit 77;; esac
B=$(python3 -c 'import time; print(time.time())'); D=$(a shell date +%s.%N | tr -d '\r'); A=$(python3 -c 'import time; print(time.time())')
T0=$(a shell "date '+%m-%d %H:%M:%S.000'" | tr -d '\r')
echo "== $P tablet web stop — host $(date '+%H:%M:%S'); device-minus-host $(python3 -c "print(round(($D - ($B + $A) / 2) * 1000))") ms (bracket $(python3 -c "print(round(($A - $B) * 1000))") ms); load $(uptime | sed 's/.*averages: //')"
a shell screenrecord --bugreport --size 864x540 --time-limit 40 /sdcard/w98_$P.mp4 &
REC=$!
a logcat -v epoch -T "$T0" -s ReactNativeJS:V > "$OUT/${P}_tablet_js_stream.log" 2>&1 &
LOGP=$!
sleep 4
(cd /Users/gsp-budi/Documents/DEMS && CTX=a node --import tsx "$SP/web_stop.mjs" "$R" "$P")
wait $REC; sleep 1
kill $LOGP 2>/dev/null
a pull /sdcard/w98_$P.mp4 "$OUT/${P}_tablet_web_stop.mp4" 2>&1 | tail -n 1
a shell rm -f /sdcard/w98_$P.mp4
echo "-- end host $(date '+%H:%M:%S'); load $(uptime | sed 's/.*averages: //'); JS stream lines $(wc -l < "$OUT/${P}_tablet_js_stream.log" | tr -d ' '); token-shaped lines (must be 0): $(grep -c -E 'dvc_|ses_|bio_' "$OUT/${P}_tablet_js_stream.log")"
