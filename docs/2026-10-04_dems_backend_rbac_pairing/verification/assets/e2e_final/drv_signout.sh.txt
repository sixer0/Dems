#!/bin/sh
# Sign the operator out from More on ONE device; expect Login. UI dump only on More. 540 px recording around the tap.
#   sh signout.sh <serial> <prefix>
set -u
S=$1; P=$2
case "$S" in emulator-5554|emulator-5556|emulator-5558) ;; *) echo "serial refused"; exit 64;; esac
OUT=/Users/gsp-budi/Documents/DEMS/docs/2026-10-04_dems_backend_rbac_pairing/verification/assets/e2e_final
APP=com.dems.demo.tablet
SIZE=540x1212; [ "$S" = emulator-5556 ] && SIZE=864x540
a() { adb -s "$S" "$@"; }
now() { python3 -c 'import time; t=time.time(); print(time.strftime("%H:%M:%S", time.localtime(t)) + ".%03d" % int((t % 1) * 1000))'; }
kg() { a shell dumpsys window 2>/dev/null | tr -d '\r' | grep -o 'isKeyguardShowing=[a-z]*' | head -n 1; }
echo "== sign out $S — host $(now); $(kg); load $(uptime | sed 's/.*averages: //')"
case "$(kg)" in *true*) echo "KEYGUARD: stop"; exit 77;; esac
a shell "am start -W -a android.intent.action.VIEW -d 'demsdemo:///more' $APP" 2>&1 | tr -d '\r' | grep -E "Status|Error" | sed 's/^/   /'
sleep 5
for try in 1 2 3; do
  a shell uiautomator dump /sdcard/e2f_ui.xml >/dev/null 2>&1; a shell cat /sdcard/e2f_ui.xml > "$OUT/${P}_ui_more.xml" 2>/dev/null; a shell rm -f /sdcard/e2f_ui.xml
  grep -q 'resource-id="more-sign-out"' "$OUT/${P}_ui_more.xml" && break; sleep 2
done
IDS=$(grep -o 'resource-id="screen-[a-z-]*"' "$OUT/${P}_ui_more.xml" | sed 's/resource-id=//; s/"//g' | sort -u | tr '\n' ' ')
echo "   UI dump ${P}_ui_more.xml: screen ids [$IDS]"
case "$IDS" in *screen-more*) ;; *) echo "-- More not confirmed: NOTHING tapped"; exit 3;; esac
a exec-out screencap -p > "$OUT/${P}_more.png"
B=$(grep -o '<node[^>]*resource-id="more-sign-out"[^>]*>' "$OUT/${P}_ui_more.xml" | grep -o 'bounds="[^"]*"' | head -n 1)
X=$(echo "$B" | sed -E 's/bounds="\[([0-9]+),([0-9]+)\]\[([0-9]+),([0-9]+)\]"/\1 \3/' | awk '{print int(($1+$2)/2)}')
Y=$(echo "$B" | sed -E 's/bounds="\[([0-9]+),([0-9]+)\]\[([0-9]+),([0-9]+)\]"/\2 \4/' | awk '{print int(($1+$2)/2)}')
[ -z "$X" ] && { echo "-- no Sign Out bounds: NOTHING tapped"; exit 3; }
T0=$(a shell "date '+%m-%d %H:%M:%S.000'" | tr -d '\r')
a shell screenrecord --bugreport --size $SIZE --time-limit 12 /sdcard/e2f_signout.mp4 &
REC=$!
sleep 2
echo "-- $(kg); tap more-sign-out $X $Y at host $(now)"
a shell input tap "$X" "$Y"
wait $REC; sleep 1
a pull /sdcard/e2f_signout.mp4 "$OUT/${P}_sign_out_screenrecord.mp4" 2>&1 | tail -n 1
a shell rm -f /sdcard/e2f_signout.mp4
a exec-out screencap -p > "$OUT/${P}_after_sign_out.png"; echo "   screencap ${P}_after_sign_out.png at host $(now)"
echo "-- app requests since T0:"
a logcat -d -v epoch -T "$T0" -s ReactNativeJS:V 2>/dev/null | grep 'DEMS-LIVE' | sed -E 's/^ *([0-9.]+) .*\[DEMS-LIVE\] /   \1 /; s/ [0-9a-f]{8}-[0-9a-f-]{27}$//'
echo "-- prefs: $(a shell "run-as $APP sh -c 'sqlite3 databases/RKStorage \"select value from catalystLocalStorage where key = \\\"dems.mobile.prefs.v1\\\"\"'" | tr -d '\r')"
echo "-- secure-store entry NAMES: $(a shell "run-as $APP sh -c 'grep -o \"name=\\\"[^\\\"]*\\\"\" shared_prefs/SecureStore.xml'" | tr -d '\r' | tr '\n' ' ')"
echo "-- credential/token lines since T0 (must be 0): $(a logcat -d -T "$T0" -s ReactNativeJS:V 2>/dev/null | grep -c -E 'dvc_|ses_|bio_')"
echo "-- end host $(now); $(kg)"
