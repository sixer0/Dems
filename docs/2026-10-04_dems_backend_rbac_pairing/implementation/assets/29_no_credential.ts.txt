// Evidence script of plan row W29 (not part of the project): the documents, the evidence files
// and the sources of this row hold no credential value. The two role passwords are read through
// the project's own readers only to be searched for; only COUNTS are printed.
import { readdirSync, readFileSync, statSync } from 'node:fs';
import { join } from 'node:path';

const ROOT = '/Users/gsp-budi/Documents/DEMS';
const { readOwnerConnection } = await import(`${ROOT}/apps/core-server/src/db/commands/owner-connection.ts`);
const { loadConfig } = await import(`${ROOT}/apps/core-server/src/plugins/config/index.ts`);
const { DEMO_OPERATOR_PASSWORD, DEMO_PAIRING_CODE } = await import(`${ROOT}/packages/shared/src/sim/index.ts`);

const needles: Array<[string, string]> = [
  ['the owner role password', readOwnerConnection().password.reveal()],
  ['the runtime role password', loadConfig({ env: {} }).database.password.reveal()],
  ['the demo operator password', DEMO_OPERATOR_PASSWORD],
];

function walk(directory: string, accept: (file: string) => boolean, out: string[] = []): string[] {
  for (const name of readdirSync(directory)) {
    const full = join(directory, name);
    if (statSync(full).isDirectory()) walk(full, accept, out);
    else if (accept(full)) out.push(full);
  }
  return out;
}

const docs = `${ROOT}/docs/2026-10-04_dems_backend_rbac_pairing`;
const files = [
  `${docs}/implementation/29_db_reset_routine.md`,
  `${docs}/delegation_progress/database_specialist_step_51.md`,
  ...walk(`${docs}/implementation/assets`, (file) => /\/29_[^/]+$/.test(file)),
  ...walk(`${ROOT}/apps/core-server/src/db/reset`, () => true),
];
let bytes = 0;
const hits: Record<string, number> = Object.fromEntries(needles.map(([label]) => [label, 0]));
let passwordKeyWithValue = 0;
for (const file of files) {
  const text = readFileSync(file, 'utf8');
  bytes += text.length;
  for (const [label, value] of needles) {
    if (value.length < 8) throw new Error(`${label} is too short to search for`);
    for (let index = text.indexOf(value); index !== -1; index = text.indexOf(value, index + 1)) hits[label] += 1;
  }
  // A connection-file line with a value: KEY=value for one of the two password keys.
  passwordKeyWithValue += (text.match(/DEMS_PG_(?:OWNER|APP)_PASSWORD=(?!w29-not-a-credential|\$\{)\S+/g) ?? []).length;
}
console.log(`files searched: ${files.length} (${bytes} characters): 2 documents, the 29_* evidence files, src/db/reset/**`);
for (const [label] of needles) console.log(`occurrences of ${label}: ${hits[label]}`);
console.log(`connection-file password lines with a value (other than the test's own marker): ${passwordKeyWithValue}`);
console.log(`the pairing code constant has ${String(DEMO_PAIRING_CODE).length} characters and is not searched as a bare number (it would match timestamps); the command's output is searched for it in cli.db.test.ts`);
