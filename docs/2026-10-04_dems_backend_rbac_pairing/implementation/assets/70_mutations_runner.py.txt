def run(tests):
    proc = subprocess.run(
        ['npx', 'vitest', 'run', *tests, '--maxWorkers=2', '--reporter=json', '--outputFile=.cache-mut-result.json'],
        cwd=WEB, capture_output=True, text=True, timeout=420,
    )
    out = WEB / '.cache-mut-result.json'
    if not out.exists():
        return None, proc.returncode, (proc.stdout + proc.stderr)[-1500:]
    data = json.loads(out.read_text())
    out.unlink()
    return (data.get('numFailedTests', 0), data.get('numTotalTests', 0), data.get('numFailedTestSuites', 0)), proc.returncode, ''


def main():
    wanted = set(sys.argv[2:])
    sets = {}
    print(f'scratch root: {ROOT}', flush=True)
    for mid, rel, old, new, tests in MUTATIONS:
        if wanted and mid.split()[0] not in wanted:
            continue
        key = tuple(tests)
        if key not in sets:
            base, code, tail = run(list(tests))
            sets[key] = base
            print(f'BASELINE {list(tests)}: failed/total/failedSuites = {base} (exit {code}) {tail}', flush=True)
        path = WEB / rel
        text = path.read_text()
        if text.count(old) != 1:
            print(f'{mid}: NOT APPLIED (pattern count {text.count(old)})', flush=True)
            continue
        path.write_text(text.replace(old, new))
        try:
            res, code, tail = run(list(tests))
        finally:
            path.write_text(text)
        base = sets[key]
        if res is None or base is None:
            verdict = f'NO RESULT (exit {code}) {tail}'
        else:
            delta = res[0] - base[0]
            suites = res[2] - base[2]
            verdict = 'DETECTED' if (delta > 0 or suites > 0) else 'SURVIVED'
            verdict = f'{verdict} (failed {res[0]} vs baseline {base[0]}, +{delta}; failed suites +{suites})'
        print(f'{mid}: {verdict}', flush=True)


main()
