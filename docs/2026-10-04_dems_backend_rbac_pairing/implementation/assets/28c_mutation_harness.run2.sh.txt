#!/bin/sh
# As run.sh, for a re-run on the final state: the unit project first; the db project only for a
# mutation that no unit test detected. Usage: run2.sh <scratch dir> id [id ...]
M="$1"; shift
ROOT=/Users/gsp-budi/Documents/DEMS
cd "$ROOT/apps/core-server" || exit 2
summary() { grep -E "^ +Tests |the text to replace is not in|no mutation " "$1" | sed -n '1p' | sed -e 's/^ *//' -e 's/  */ /g'; }
for id in "$@"; do
  note=$(python3 -c "import json,sys; print([x['note'] for x in json.load(open('$M/mutations.json')) if x['id']=='$id'][0])")
  W28_MUTATION="$id" "$ROOT/node_modules/.bin/vitest" run --config "$M/vitest.config.mjs" --project unit --maxWorkers=2 src/plugins/permission-guard src/modules/rbac > "$M/out2.unit.txt" 2>&1
  u=$(summary "$M/out2.unit.txt")
  case "$u" in
    *failed*) d="not run (a unit test detects it)" ;;
    *) W28_MUTATION="$id" DEMS_TEST_STEP=50b "$ROOT/node_modules/.bin/vitest" run --config "$M/vitest.config.mjs" --project db --maxWorkers=2 src/plugins/permission-guard src/modules/rbac > "$M/out2.db.txt" 2>&1
       d=$(summary "$M/out2.db.txt") ;;
  esac
  echo "$id | $note | unit: ${u:-NO-SUMMARY} | db: ${d:-NO-SUMMARY}"
done
