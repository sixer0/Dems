const mode = process.argv[2] === 'db' ? 'db' : 'unit';
const only = process.argv[3] === undefined ? null : new Set(process.argv[3].split(','));
const sha = (text) => createHash('sha256').update(text).digest('hex');

function replaceOnce(text, find, replace, id) {
  const first = text.indexOf(find);
  if (first < 0) throw new Error(`${id}: the text to change was not found`);
  if (text.indexOf(find, first + 1) >= 0) throw new Error(`${id}: the text to change occurs more than once`);
  return text.slice(0, first) + replace + text.slice(first + find.length);
}

function runTests() {
  const args = mode === 'db'
    ? ['run', 'test:db', '-w', '@dems/core-server', '--', '--maxWorkers=2', 'src/modules/storage', 'src/modules/network-time']
    : ['run', 'test', '-w', '@dems/core-server', '--', '--maxWorkers=2', 'src/modules/storage', 'src/modules/network-time'];
  const result = spawnSync('npm', args, { cwd: ROOT, encoding: 'utf8', env: { ...process.env, DEMS_TEST_STEP: '59' }, maxBuffer: 64 * 1024 * 1024 });
  const output = `${result.stdout}\n${result.stderr}`;
  const tests = /Tests\s+(?:(\d+) failed \| )?(\d+) passed/.exec(output);
  const files = [...new Set([...output.matchAll(/FAIL\s+\|(?:unit|db)\|\s+(\S+?)(?:\s|$)/g)].map((match) => match[1].replace('src/modules/', '')))];
  return { status: result.status, failed: Number(tests?.[1] ?? 0), passed: Number(tests?.[2] ?? 0), files, output };
}

const stamp = () => new Date().toTimeString().slice(0, 8);
console.log(`[w37 mutation] mode ${mode}; start ${stamp()}`);
const baseline = runTests();
console.log(`[w37 mutation] baseline: exit ${baseline.status}, ${baseline.passed} passed, ${baseline.failed} failed`);
if (baseline.status !== 0) {
  console.log(baseline.output.slice(-3000));
  process.exit(2);
}

let detected = 0;
let survived = 0;
for (const mutation of MUTATIONS) {
  if (only !== null && !only.has(mutation.id)) continue;
  const original = readFileSync(mutation.file, 'utf8');
  const hash = sha(original);
  let mutated = replaceOnce(original, mutation.find, mutation.replace, mutation.id);
  for (const more of mutation.also ?? []) mutated = replaceOnce(mutated, more.find, more.replace, mutation.id);
  let outcome;
  try {
    writeFileSync(mutation.file, mutated);
    outcome = runTests();
  } finally {
    writeFileSync(mutation.file, original);
  }
  if (sha(readFileSync(mutation.file, 'utf8')) !== hash) throw new Error(`${mutation.id}: the file was not restored`);
  const killed = outcome.status !== 0;
  if (killed) detected += 1;
  else survived += 1;
  const where = mutation.file.replace(`${SRC}/`, '');
  console.log(`[w37 mutation] ${mutation.id} ${killed ? 'DETECTED' : 'SURVIVED'} ${where}: ${mutation.what} -> ${outcome.failed} failed in ${outcome.files.join(', ') || '(no test file named)'}`);
  if (killed && outcome.failed === 0) console.log(outcome.output.split('\n').filter((line) => /Error|error/.test(line)).slice(0, 4).join('\n'));
}
console.log(`[w37 mutation] end ${stamp()}: ${detected} detected, ${survived} survived of ${detected + survived}`);
process.exit(survived === 0 ? 0 : 1);
