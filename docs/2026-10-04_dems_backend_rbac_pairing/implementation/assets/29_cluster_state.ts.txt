// Evidence script of plan row W29 (not part of the project): the state of the project cluster as
// `dems_owner` sees it from the maintenance database. It reads only. It uses the project's own
// reader of the owner connection file, so no credential is printed, passed or copied.
import { createOwnerPool, readOwnerConnection } from '/Users/gsp-budi/Documents/DEMS/apps/core-server/src/db/commands/owner-connection';

const owner = readOwnerConnection();
const pool = createOwnerPool(owner, 'postgres', 'dems-w29-evidence');
try {
  const who = await pool.query<{ role: string; superuser: boolean; address: string; port: number; started: string; version: string }>(
    "SELECT current_user::text AS role, r.rolsuper AS superuser, host(inet_server_addr()) AS address, inet_server_port() AS port, " +
      "to_char(pg_postmaster_start_time(), 'YYYY-MM-DD HH24:MI:SS TZ') AS started, current_setting('server_version') AS version " +
      'FROM pg_roles r WHERE r.rolname = current_user',
  );
  const w = who.rows[0];
  console.log(`session        ${w?.role} (superuser ${String(w?.superuser)}) on ${w?.address}:${w?.port}, PostgreSQL ${w?.version}, postmaster started ${w?.started}`);

  const databases = await pool.query<{ name: string; owner: string; template: boolean; connections: boolean; encoding: string; collate: string }>(
    'SELECT d.datname::text AS name, pg_get_userbyid(d.datdba)::text AS owner, d.datistemplate AS template, d.datallowconn AS connections, ' +
      'pg_encoding_to_char(d.encoding)::text AS encoding, d.datcollate::text AS collate FROM pg_database d ORDER BY d.datname',
  );
  console.log(`pg_database    ${databases.rows.length} row(s)`);
  for (const d of databases.rows) {
    console.log(`  ${d.name.padEnd(48)} owner ${d.owner.padEnd(11)} template ${String(d.template).padEnd(5)} allowconn ${String(d.connections).padEnd(5)} ${d.encoding} ${d.collate}`);
  }

  const roles = await pool.query<{ rolname: string; rolsuper: boolean; rolcreatedb: boolean; rolcreaterole: boolean; rolcanlogin: boolean }>(
    "SELECT rolname::text AS rolname, rolsuper, rolcreatedb, rolcreaterole, rolcanlogin FROM pg_roles WHERE rolname NOT LIKE 'pg\\_%' ORDER BY rolname",
  );
  console.log(`pg_roles       ${roles.rows.length} role(s) outside pg_*`);
  for (const r of roles.rows) {
    console.log(`  ${r.rolname.padEnd(12)} superuser ${String(r.rolsuper).padEnd(5)} createdb ${String(r.rolcreatedb).padEnd(5)} createrole ${String(r.rolcreaterole).padEnd(5)} login ${String(r.rolcanlogin)}`);
  }

  const settings = await pool.query<{ settings: string; digest: string; overrides: string }>(
    "SELECT count(*)::text AS settings, md5(string_agg(name || '=' || coalesce(setting, ''), E'\\n' ORDER BY name)) AS digest, " +
      '(SELECT count(*)::text FROM pg_db_role_setting) AS overrides FROM pg_settings',
  );
  const s = settings.rows[0];
  console.log(`pg_settings    ${s?.settings} visible setting(s), digest ${s?.digest}; pg_db_role_setting ${s?.overrides} row(s)`);

  const sessions = await pool.query<{ database: string; role: string; application: string; sessions: string }>(
    "SELECT datname::text AS database, usename::text AS role, application_name AS application, count(*)::text AS sessions " +
      'FROM pg_stat_activity WHERE datname IS NOT NULL AND usename IS NOT NULL GROUP BY 1, 2, 3 ORDER BY 1, 2, 3',
  );
  console.log(`client sessions ${sessions.rows.length} group(s)`);
  for (const x of sessions.rows) console.log(`  ${x.database.padEnd(48)} ${x.role.padEnd(11)} ${x.application.padEnd(24)} ${x.sessions}`);

  const locks = await pool.query<{ classid: string; objid: string; granted: boolean }>(
    "SELECT classid::text AS classid, objid::text AS objid, granted FROM pg_locks WHERE locktype = 'advisory' ORDER BY 1, 2",
  );
  console.log(`advisory locks ${locks.rows.length}${locks.rows.map((l) => ` (${l.classid}, ${l.objid}${l.granted ? '' : ' waiting'})`).join('')}`);
} finally {
  await pool.end();
}
