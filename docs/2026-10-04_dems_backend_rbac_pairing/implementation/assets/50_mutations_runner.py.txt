def read(file):
    with open(os.path.join(SRC, file), encoding='utf8') as handle:
        return handle.read()


def check():
    bad = []
    ids = set()
    for entry in M:
        if entry['id'] in ids:
            bad.append(f"{entry['id']}: duplicate id")
        ids.add(entry['id'])
        count = read(entry['file']).count(entry['find'])
        if count != 1:
            bad.append(f"{entry['id']} ({entry['file']}): the text occurs {count} times")
        if entry['find'] == entry['replace']:
            bad.append(f"{entry['id']}: replacement equals the text")
        if entry['file'] not in TESTS:
            bad.append(f"{entry['id']}: no tests for {entry['file']}")
    return bad


def vitest(files, mutation=None):
    env = dict(os.environ)
    env.pop('W50_MUT', None)
    env['W50_APPLIED_DIR'] = APPLIED
    if mutation:
        env['W50_MUT'] = mutation
    started = time.time()
    # One worker: the row's files run one after another, so the check adds little to the load (P9).
    done = subprocess.run(['npx', 'vitest', 'run', '--config', CONFIG, '--maxWorkers=1', *files], cwd=ROOT, env=env, capture_output=True, text=True)
    text = done.stdout + done.stderr
    summary = ' | '.join(line.strip() for line in text.splitlines() if line.strip().startswith(('Test Files', 'Tests ')))
    return done.returncode, summary, time.time() - started, text


def failed_tests(text):
    """The names of the tests that failed, as the default reporter prints them (lines that start with the cross)."""
    names = []
    for line in text.splitlines():
        stripped = line.strip()
        if stripped.startswith('\u00d7 '):
            name = stripped[2:]
            # the trailing duration
            parts = name.rsplit(' ', 1)
            if len(parts) == 2 and parts[1].endswith('ms'):
                name = parts[0]
            names.append(name)
    return names


def timed_out(text):
    return 'Test timed out' in text or 'Timed out in waitFor' in text


def main():
    mode = sys.argv[1] if len(sys.argv) > 1 else 'check'
    bad = check()
    print(f'{len(M)} mutations in {len(set(e["file"] for e in M))} modules; texts that do not occur exactly once: {len(bad)}')
    for line in bad:
        print('  ' + line)
    if bad or mode == 'check':
        sys.exit(1 if bad else 0)
    with open(os.path.join(HERE, 'mutations.json'), 'w', encoding='utf8') as handle:
        json.dump(M, handle, indent=1)
    wanted = sys.argv[2:]
    print(time.strftime('## %Y-%m-%d %H:%M:%S %Z') + ' baseline: the nine test files of the row, no mutation')
    code, summary, seconds, text = vitest(ALL)
    print(f'baseline exit={code} {summary} ({seconds:.1f} s)')
    if code != 0:
        print(text[-3000:])
        sys.exit(2)
    survived, invalid = [], []
    for entry in M:
        if wanted and entry['id'] not in wanted:
            continue
        marker = os.path.join(APPLIED, entry['id'])
        if os.path.exists(marker):
            os.remove(marker)
        code, summary, seconds, text = vitest(TESTS[entry['file']], entry['id'])
        applied = os.path.exists(marker)
        verdict = 'DETECTED' if code != 0 and applied else ('NOT APPLIED' if not applied else 'SURVIVED')
        if verdict == 'SURVIVED':
            survived.append(entry['id'])
        if verdict == 'NOT APPLIED':
            invalid.append(entry['id'])
        names = failed_tests(text)
        first = names[0][:150] if names else '-'
        note = ' TIMEOUT-IN-OUTPUT' if timed_out(text) else ''
        print(f"{entry['id']} {verdict:11s} exit={code} {entry['file']}: {entry['what']} [{summary}] failed tests: {len(names)}; first: {first}{note} ({seconds:.1f} s)", flush=True)
    total = len([e for e in M if not wanted or e['id'] in wanted])
    print(time.strftime('## %Y-%m-%d %H:%M:%S %Z') + f' {total} mutations: {total - len(survived) - len(invalid)} detected, {len(survived)} survived {survived}, {len(invalid)} not applied {invalid}')


main()
