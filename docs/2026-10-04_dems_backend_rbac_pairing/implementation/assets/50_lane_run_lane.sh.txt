#!/bin/sh
# W50 lane check (lane 3): the web console against the REAL Core server through the dev proxy.
# Run from the workspace root, inside the lane lock:
#   /usr/bin/lockf -k -t 900 .locks/core-server.lane3.lock /bin/sh .cache/w50/lane/run_lane.sh
# It starts two processes of its own (the lane backend on 127.0.0.1:4423, Vite on 127.0.0.1:5178)
# and a headless Chromium (devtools on 127.0.0.1:9398), and stops all three at the end. It never
# touches a listener it did not start. No credential is printed.
set -u
K=3; PORT=4423; VITE=5178; DEV=9398
OUT=.cache/w50/lane/out
ASSETS=docs/2026-10-04_dems_backend_rbac_pairing/implementation/assets
CHROME="$HOME/Library/Caches/ms-playwright/chromium_headless_shell-1243/chrome-headless-shell-mac-arm64/chrome-headless-shell"
# The browser profile lives under TMPDIR and is removed there; without TMPDIR the browser step is not run.
[ -n "${TMPDIR:-}" ] || { echo "ABORT: TMPDIR is not set"; exit 6; }
PROFILE="${TMPDIR%/}/w50-chrome-profile"
mkdir -p "$OUT" "$ASSETS"

stamp() { date '+%H:%M:%S'; }
say() { echo; echo "== $(stamp) $*"; }
listening() { lsof -nP -iTCP:"$1" -sTCP:LISTEN >/dev/null 2>&1; }

BACK_PID=""; VITE_PID=""; CHROME_PID=""; STARTED_BACK=0; STARTED_VITE=0; STARTED_CHROME=0

stop_all() {
  say "stop what this check started"
  if [ "$STARTED_CHROME" = 1 ]; then
    kill "$CHROME_PID" 2>/dev/null; wait "$CHROME_PID" 2>/dev/null
    P=$(lsof -nP -tiTCP:$DEV -sTCP:LISTEN 2>/dev/null); [ -n "$P" ] && kill $P 2>/dev/null
    echo "chrome stopped"
  fi
  if [ "$STARTED_VITE" = 1 ]; then
    kill "$VITE_PID" 2>/dev/null; wait "$VITE_PID" 2>/dev/null; echo "vite: npm exit status $?"
    i=0; while listening $VITE && [ $i -lt 20 ]; do i=$((i + 1)); sleep 0.25; done
    P=$(lsof -nP -tiTCP:$VITE -sTCP:LISTEN 2>/dev/null); [ -n "$P" ] && { echo "vite: fallback kill of the listener $P"; kill $P 2>/dev/null; }
    : > .locks/core-server.lane$K.vite.pid   # emptied, not removed: this row runs no rm outside TMPDIR
  fi
  if [ "$STARTED_BACK" = 1 ]; then
    kill "$BACK_PID" 2>/dev/null; wait "$BACK_PID" 2>/dev/null; echo "backend: npm exit status $?"
    i=0; while listening $PORT && [ $i -lt 40 ]; do i=$((i + 1)); sleep 0.25; done
    P=$(lsof -nP -tiTCP:$PORT -sTCP:LISTEN 2>/dev/null); [ -n "$P" ] && { echo "backend: fallback kill of the listener $P"; kill $P 2>/dev/null; sleep 1; }
    : > .locks/core-server.lane$K.pid
  fi
  [ -d "$PROFILE" ] && rm -rf "$PROFILE"
  say "listeners left on $PORT, $VITE, $DEV (expect none):"
  lsof -nP -iTCP:$PORT -iTCP:$VITE -iTCP:$DEV -sTCP:LISTEN; echo "lsof exit $? (1 with no output: no listener)"
}
trap stop_all EXIT

say "0. the three ports must be free before anything is started (nothing is stopped that this check did not start)"
for p in $PORT $VITE $DEV; do
  if listening $p; then echo "port $p is in use:"; lsof -nP -iTCP:$p -sTCP:LISTEN; echo "ABORT: not this check's listener"; exit 3; fi
done
echo "free: $PORT $VITE $DEV"
date '+%Y-%m-%d %H:%M:%S %Z'; uptime

say "1. lane database before the check: npm run db:status -w @dems/core-server -- --database dems_lane_$K"
npm run db:status -w @dems/core-server -- --database dems_lane_$K > "$OUT/db_status_before.txt" 2>&1; echo "db:status exit $?"
grep -E "audit_log|sessions|pairing|operators|audit head|seeded_at|last_reset" "$OUT/db_status_before.txt" | head -20

say "2. start the lane backend: DEMS_CORE_PORT=$PORT npm run start -w @dems/core-server (demo flag OFF)"
attempt=1
while :; do
  env -u DEMS_DEMO_CONTROLS DEMS_CORE_PORT=$PORT npm run start -w @dems/core-server > .locks/core-server.lane$K.log 2>&1 &
  BACK_PID=$!; STARTED_BACK=1; echo $BACK_PID > .locks/core-server.lane$K.pid
  i=0; code=000
  while [ $i -lt 240 ]; do
    code=$(curl -s -o /dev/null -w '%{http_code}' http://127.0.0.1:$PORT/api/v1/health 2>/dev/null)
    [ "$code" = "200" ] && break
    kill -0 "$BACK_PID" 2>/dev/null || break
    i=$((i + 1)); sleep 0.25
  done
  if [ "$code" = "200" ]; then echo "backend up after $i quarter seconds (attempt $attempt), npm pid $BACK_PID"; break; fi
  echo "backend did not come up (attempt $attempt, last status $code). Last lines of its log:"; tail -15 .locks/core-server.lane$K.log
  kill "$BACK_PID" 2>/dev/null; wait "$BACK_PID" 2>/dev/null
  P=$(lsof -nP -tiTCP:$PORT -sTCP:LISTEN 2>/dev/null); [ -n "$P" ] && kill $P 2>/dev/null
  STARTED_BACK=0
  if [ $attempt -ge 3 ]; then echo "ABORT: the backend did not boot in three attempts"; exit 4; fi
  attempt=$((attempt + 1)); echo "waiting 60 s before attempt $attempt"; sleep 60
done
lsof -nP -iTCP:$PORT -sTCP:LISTEN
echo "-- direct health on the backend:"; curl -s http://127.0.0.1:$PORT/api/v1/health; echo

say "3. start Vite for lane $K: DEMS_CORE_PROXY_PORT=$PORT VITE_DEMS_CORE_MODE=live npm run dev -w @dems/core-web -- --host 127.0.0.1 --port $VITE --strictPort"
DEMS_CORE_PROXY_PORT=$PORT VITE_DEMS_CORE_MODE=live npm run dev -w @dems/core-web -- --host 127.0.0.1 --port $VITE --strictPort > .locks/core-server.lane$K.vite.log 2>&1 &
VITE_PID=$!; STARTED_VITE=1; echo $VITE_PID > .locks/core-server.lane$K.vite.pid
i=0; until listening $VITE || [ $i -ge 120 ]; do i=$((i + 1)); sleep 0.25; done
echo "waited $i quarter seconds"; lsof -nP -iTCP:$VITE -sTCP:LISTEN || { echo "ABORT: Vite does not listen"; tail -20 .locks/core-server.lane$K.vite.log; exit 5; }

say "4. AC-C-33: curl -si http://127.0.0.1:$VITE/api/v1/health"
curl -si http://127.0.0.1:$VITE/api/v1/health | tee "$OUT/health.txt"; echo
echo "-- access-control-allow- headers in that answer: $(grep -ci '^access-control-allow-' "$OUT/health.txt")"
echo "-- service: $(grep -o '"service":"[^"]*"' "$OUT/health.txt")"

say "5. AC-C-34 (script half): the socket THROUGH the dev server, Node's global WebSocket"
echo "-- a. no Origin header (what a script sends):"
node .cache/w50/lane/ws-check.mjs ws://127.0.0.1:$VITE/ws; echo "ws-check exit $?"
echo "-- b. the console's own Origin:"
node .cache/w50/lane/ws-check.mjs ws://127.0.0.1:$VITE/ws http://127.0.0.1:$VITE; echo "ws-check exit $?"

say "6. AC-C-102: a sign-in POST through the dev server. The body is syntactically valid, names no seeded operator, and holds an obviously wrong password"
BODY='{"username":"origin.check","password":"obviously-wrong-password"}'
for ORIGIN in "https://evil.example" "http://localhost:5173"; do
  echo "-- Origin: $ORIGIN"
  curl -si -X POST -H 'content-type: application/json' -H "Origin: $ORIGIN" --data "$BODY" http://127.0.0.1:$VITE/api/v1/auth/sign-in | grep -iv '^set-cookie' ; echo
done
echo "-- control, the console's own Origin (http://127.0.0.1:$VITE): the origin stage passes, and the wrong password is refused as such"
curl -si -X POST -H 'content-type: application/json' -H "Origin: http://127.0.0.1:$VITE" --data "$BODY" http://127.0.0.1:$VITE/api/v1/auth/sign-in | grep -iv '^set-cookie'; echo

say "7. the console in a real browser, both modes (headless Chromium over CDP)"
if [ -x "$CHROME" ]; then
  rm -rf "$PROFILE"
  "$CHROME" --remote-debugging-port=$DEV --remote-debugging-address=127.0.0.1 --user-data-dir="$PROFILE" --no-first-run --disable-background-networking --disable-component-update --disable-sync --disable-default-apps --disable-domain-reliability --host-resolver-rules="MAP * ~NOTFOUND, EXCLUDE 127.0.0.1, EXCLUDE localhost" --window-size=1440,900 about:blank > "$OUT/chrome.log" 2>&1 &
  CHROME_PID=$!; STARTED_CHROME=1; echo "chrome pid $CHROME_PID"
  i=0; until curl -s http://127.0.0.1:$DEV/json/version >/dev/null 2>&1 || [ $i -ge 60 ]; do i=$((i + 1)); sleep 0.25; done
  node --import tsx .cache/w50/lane/cdp-driver.mjs $DEV $VITE "$OUT/browser.json" "$ASSETS" > "$OUT/browser.log" 2>&1; echo "cdp-driver exit $?"
  tail -3 "$OUT/browser.log"
else
  echo "NOT DONE: no headless Chromium at the expected path"
fi

say "8. what Vite printed (hot update lines; no page error)"
grep -n "hmr update\|page reload\|error\|Error" .locks/core-server.lane$K.vite.log | head -12 || true
echo "-- lines in Vite's log: $(wc -l < .locks/core-server.lane$K.vite.log)"

say "9. lane database after the check"
npm run db:status -w @dems/core-server -- --database dems_lane_$K > "$OUT/db_status_after.txt" 2>&1; echo "db:status exit $?"
echo "-- row counts that differ (before / after):"
diff "$OUT/db_status_before.txt" "$OUT/db_status_after.txt" | grep -E '^[<>]' | head -30
echo "-- the backend's log: lines $(wc -l < .locks/core-server.lane$K.log); lines with level error or fatal: $(grep -c '"level":\(50\|60\)\|"level":"\(error\|fatal\)"' .locks/core-server.lane$K.log)"
