#!/bin/sh
# Runs mutations against the row's own tests. Prints one line each.
# Usage: run.sh <scratch dir> <unit|db|both> [id ...]   (no id: every mutation of mutations.json)
# Vitest writes into a file that is read afterwards. No directory is changed: Vitest gets the
# workspace as its root. No file of the repository is written.
M="$1"; MODE="$2"; shift; shift
ROOT=/Users/gsp-budi/Documents/DEMS
WS="$ROOT/apps/core-server"
if [ "$#" -eq 0 ]; then set -- $(python3 -c "import json; print(' '.join(x['id'] for x in json.load(open('$M/mutations.json'))))"); fi
summary() { grep -E "^ +Tests |the text to replace is |no mutation " "$1" | sed -n '1p' | sed -e 's/^ *//' -e 's/  */ /g'; }
for id in "$@"; do
  note=$(python3 -c "import json,sys; print([x['note'] for x in json.load(open('$M/mutations.json')) if x['id']=='$id'][0])")
  u="-"; d="-"
  if [ "$MODE" = unit ] || [ "$MODE" = both ]; then
    W40_MUTATION="$id" "$ROOT/node_modules/.bin/vitest" run --config "$M/vitest.config.mjs" --root "$WS" --maxWorkers=2 --project unit src/modules/system src/modules/alerts > "$M/out.unit.txt" 2>&1
    u=$(summary "$M/out.unit.txt"); u="${u:-NO-SUMMARY}"
  fi
  if [ "$MODE" = db ] || [ "$MODE" = both ]; then
    W40_MUTATION="$id" DEMS_TEST_STEP=62 "$ROOT/node_modules/.bin/vitest" run --config "$M/vitest.config.mjs" --root "$WS" --maxWorkers=2 --project db src/modules/system src/modules/alerts > "$M/out.db.txt" 2>&1
    d=$(summary "$M/out.db.txt"); d="${d:-NO-SUMMARY}"
  fi
  echo "$id | $note | unit: $u | db: $d"
done
