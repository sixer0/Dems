def sha(path):
    return hashlib.sha256(open(path, 'rb').read()).hexdigest()


FAST = {
    LM: [D + 'pairing-live-model.test.ts', D + 'pairing-model.test.ts'],
    PM: [D + 'pairing-live-model.test.ts', D + 'pairing-model.test.ts'],
    I18: [D + 'pairing-live-static.test.ts'],
    HS: [D + 'stack-check-service-name.test.tsx'],
}


def run(spec, tests=None):
    tests = tests or TESTS
    env = dict(os.environ)
    if spec is None:
        env.pop('W57_MUT_SPEC', None)
    else:
        json.dump(spec, open(SPEC, 'w'))
        env['W57_MUT_SPEC'] = SPEC
    started = time.time()
    done = subprocess.run(['npm', 'exec', '-w', '@dems/mobile', '--', 'jest', '--config', CONFIG, '--maxWorkers=2', '--silent', '--forceExit', *tests],
                          cwd=ROOT, env=env, capture_output=True, text=True)
    out = done.stdout + done.stderr
    failed = [line.strip() for line in out.splitlines() if line.strip().startswith('●') and '›' in line]
    summary = next((line.strip() for line in out.splitlines() if line.startswith('Tests:')), '(no summary)')
    broken = 'W57 mutation' in out and 'occurs' in out
    return done.returncode, summary, failed, broken, time.time() - started, out


def main():
    wanted = set(sys.argv[1:])
    todo = [m for m in M if not wanted or m[0] in wanted]
    files = sorted({m[1] for m in M})
    before = {f: sha(APP + '/' + f) for f in files}
    print(f"## W57 mutation check, {time.strftime('%Y-%m-%d %H:%M:%S')}; {len(TESTS)} test files per run; the tree is not edited")
    code, summary, failed, _, took, out = run(None)
    print(f"control (no mutation): exit={code} {summary} ({took:.0f} s)")
    if code != 0:
        print(out[-3000:])
        sys.exit(2)
    detected = missed = errors = 0
    for mid, file, old, new, what in todo:
        spec = {'id': mid, 'file': file, 'from': old, 'to': new}
        # The file's own unit tests first (seconds); the whole folder only when they do not notice.
        code, summary, failed, broken, took, out = run(spec, FAST.get(file))
        if code == 0 and not broken and file in FAST:
            code, summary, failed, broken, more, out = run(spec)
            took += more
        if broken:
            errors += 1
            print(f"{mid} ERROR ({file}): the text to change was not found exactly once")
            continue
        if summary == '(no summary)' and code == 0:
            errors += 1
            print(f"{mid} ERROR ({file}): jest gave no result\n{out[-1500:]}")
            continue
        if code != 0:
            detected += 1
            first = failed[0][2:140] if failed else 'the file does not run'
            # A test that only ran out of time under load is no detection: say so.
            late = ' [A TEST TIMED OUT]' if 'Exceeded timeout' in out else ''
            print(f"{mid} DETECTED  {file.split('/')[-1]}: {what} -> {len(failed)} failing; first: {first} ({took:.0f} s){late}")
        else:
            missed += 1
            print(f"{mid} NOT DETECTED  {file.split('/')[-1]}: {what} ({summary}; {took:.0f} s)")
        sys.stdout.flush()
    after = {f: sha(APP + '/' + f) for f in files}
    print(f"## {detected} detected, {missed} not detected, {errors} errors, of {len(todo)}")
    print('## the mutated files in the tree, before and after: ' + ('identical' if before == after else 'CHANGED'))
    for f in files:
        print(f"   {before[f][:16]} {'same' if before[f] == after[f] else 'CHANGED'} {f}")
    if os.path.exists(SPEC):
        os.remove(SPEC)


main()
