});
jest.mock('./use-monitor', () => {
  const actual = jest.requireActual<Record<string, unknown>>('./use-monitor');
  // eslint-disable-next-line @typescript-eslint/no-require-imports
  const runtime = require('@/runtime');
  return {
    ...actual,
    useMonitorReadOnly: () => {
      const banner = runtime.useConnectionBanner();
      const link = runtime.useCoreLink();
      return banner.kind === 'offline' || link.state === 'OFFLINE';
    },
  };
});

let bed: LiveMonitorBed | null = null;
beforeEach(async () => { await resetLiveMonitorBed(); });
afterEach(() => { bed?.stop(); bed = null; });
const rt = () => { const r = (globalThis as any).__DEMS_DEBUG__?.runtime?.(); return `${r?.coreLink?.state}/${r?.coreLink?.heartbeat} banner=${r?.banner}`; };

it('monitor with the step-79 read-only rule, real wait during the outage', async () => {
  const b = await renderLiveMonitor();
  bed = b;
  await b.openMonitor();
  b.setReachable(false);
  await b.flush();

  await act(async () => { await new Promise((r) => setTimeout(r, 1100)); });
  b.setReachable(true);
  await b.elapse(4_000, 250);
  console.log('OLDRULE+NOA11YINTERVAL', rt());
