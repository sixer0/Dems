#!/usr/bin/env python3
"""W81 mutation runner (rule M15): each mutant is applied in memory by the Vite plugin of
vitest.mutation.config.mts; detected = Vitest exit != 0. The SHA-256 of every mutated file is taken
before and after each batch. Usage: runner.py <first index> <last index> [--baseline]"""
import hashlib, json, os, subprocess, sys, tempfile, time
ROOT = os.getcwd()
HERE = os.path.join(ROOT, '.cache/w81/mut')
LOG = os.path.join(ROOT, 'docs/2026-10-04_dems_backend_rbac_pairing/implementation/assets/81_mutations.log')
table = json.load(open(os.path.join(HERE, 'mutations.json')))
first, last = int(sys.argv[1]), int(sys.argv[2])
baseline = '--baseline' in sys.argv
batch = table[first:last + 1]
sha = lambda path: hashlib.sha256(open(path, 'rb').read()).hexdigest()
files = sorted({os.path.join(ROOT, 'apps/core-web/src', m['file']) for m in batch})
before = {f: sha(f) for f in files}
applied_dir = tempfile.mkdtemp(prefix='w81mut-', dir=os.environ.get('TMPDIR'))
def run(tests, mut):
    env = dict(os.environ, W81_MUT=mut, W81_APPLIED_DIR=applied_dir)
    cmd = ['node_modules/.bin/vitest', 'run', '--config', '.cache/w81/mut/vitest.mutation.config.mts', '--maxWorkers=2', *tests]
    t0 = time.time()
    p = subprocess.run(cmd, env=env, capture_output=True, text=True)
    tail = [l for l in (p.stdout + p.stderr).splitlines() if 'Tests ' in l or 'Test Files' in l]
    return p.returncode, round(time.time() - t0), ' | '.join(s.strip() for s in tail)
out = open(LOG, 'a')
out.write(f"\n## {time.strftime('%Y-%m-%d %H:%M:%S')} WIB batch {first}..{last} ({len(batch)} mutants)\n")
if baseline:
    tests = sorted({t for m in table for t in m['tests']})
    code, secs, tail = run(tests, '')
    out.write(f"BASELINE exit={code} {secs}s {tail}\n"); out.flush()
results = []
for m in batch:
    code, secs, tail = run(m['tests'], m['id'])
    applied = os.path.exists(os.path.join(applied_dir, m['id']))
    verdict = 'detected' if code != 0 and applied else ('NOT APPLIED' if not applied else 'SURVIVED')
    results.append((m['id'], verdict))
    out.write(f"{m['id']} {verdict:9s} exit={code} {secs:>3}s {m['file']} :: {m['what']} :: {tail}\n"); out.flush()
after = {f: sha(f) for f in files}
same = all(before[f] == after[f] for f in files)
out.write(f"files mutated in memory: {len(files)}; identical before and after: {same}\n")
det = sum(1 for _, v in results if v == 'detected')
out.write(f"## batch result: {det} detected of {len(results)}; survived {[i for i, v in results if v == 'SURVIVED']}; not applied {[i for i, v in results if v == 'NOT APPLIED']}\n")
out.close()
print(open(LOG).read().split('## ')[-1])
