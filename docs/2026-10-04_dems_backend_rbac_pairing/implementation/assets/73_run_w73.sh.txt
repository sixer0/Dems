#!/bin/sh
# W73 device check, lane 1 (phone, dems_lane_1 on 127.0.0.1:4421). NO lane reset, no scenario load, no fault
# injection: the phone is paired as DEV-0001 (D137) and that must survive. The one server-side event of the
# sitting is a graceful stop and start of the lane backend this script started (the driver's `restart`), while
# a recording the app started runs, so the server raises the power-failure fault of 01_specs §8.6 itself.
# Run INSIDE the lane lock, outside the sandbox:
#   /usr/bin/lockf -k -t 300 .locks/core-server.lane1.lock /bin/sh .cache/w73/run_w73.sh
set -u
ROOT=/Users/gsp-budi/Documents/DEMS; H="$ROOT/.cache/w73"; OUT="$H/out"; PORT=4421; METRO=${METRO_PORT:-8083}
mkdir -p "$OUT"
say() { printf '\n== %s %s\n' "$(date '+%H:%M:%S')" "$*"; }
listener() { lsof -nP -iTCP:"$1" -sTCP:LISTEN -t 2>/dev/null | head -n 1; }
counts() { npm --prefix "$ROOT" run db:status -w @dems/core-server -- --database dems_lane_1 2>&1 | grep -E "^ +(paired_devices|device_credentials|active_session|session_history|sessions|evidence|alerts|audit_log) +[0-9]+|db:status:" | sed 's/^ */   /'; }

say "0. before (no reset): ports free?; dems_lane_1 counts"
[ -z "$(listener $PORT)" ] || { echo "port $PORT busy"; exit 70; }
[ -z "$(listener "$METRO")" ] || { echo "Metro $METRO busy"; exit 70; }
counts | tee "$OUT/counts_before.txt"

say "1. start the lane-1 backend"
: >"$OUT/server.log"
DEMS_CORE_PORT=$PORT npm --prefix "$ROOT" run start:server >"$OUT/server.log" 2>&1 &
NPM_PID=$!; i=0
until [ -n "$(listener $PORT)" ] || ! kill -0 "$NPM_PID" 2>/dev/null || [ "$i" -ge 160 ]; do i=$((i + 1)); sleep 0.25; done
SERVER_PID=$(listener $PORT); [ -n "$SERVER_PID" ] || { echo "backend did not boot"; tail -n 20 "$OUT/server.log"; exit 70; }
echo "$SERVER_PID" >"$ROOT/.locks/core-server.lane1.pid"
echo "npm PID $NPM_PID, listener PID $SERVER_PID"
grep -E 'configuration loaded' "$OUT/server.log" | grep -o '"port":[0-9]*,"database":"[a-z_0-9]*","demoControls":[a-z]*' | sed 's/^/   /'
curl -s "http://127.0.0.1:$PORT/api/v1/health"; echo

say "2. device session: phone, Metro $METRO, tcp:4420 -> tcp:$PORT (cap 600 s)"
METRO_PORT=$METRO DEMS_CORE_HOST_PORT=$PORT DEVICE_SESSION_MAX_SECONDS=600 LOCK_WAIT=${LOCK_WAIT:-300} \
  "$ROOT/apps/mobile/scripts/device-session.sh" phone "/bin/sh '$H/device_w73.sh' '$OUT'"
DEVICE_EXIT=$?; echo "device session exit=$DEVICE_EXIT"

say "3. backend requests during the sitting (both server processes)"
cat "$OUT/server.log" "$OUT/server_restarted.log" 2>/dev/null >"$OUT/server_all.log"
grep '"msg":"request completed"' "$OUT/server_all.log" | sed -E 's/.*"method":"([^"]*)","route":"([^"]*)","status":([0-9]*).*"principal":([^,}]*).*/\1 \2 \3 principal=\4/' | sed -E 's/principal="?(operator|device|null)[^ ]*/principal=\1/' | sort | uniq -c | sed 's/^/   /'
echo "-- start and stop lines:"; grep -E '"msg":"(configuration loaded|server listening|shutdown complete|shutting down|engine started|restart fault)' "$OUT/server_all.log" | sed -E 's/.*"time":([0-9]+).*"msg":"([^"]*)".*/   \1 \2/' | head -n 20
grep -oE '"epoch":[0-9]+' "$OUT/server_all.log" | sort | uniq -c | sed 's/^/   epoch lines: /' | head -n 5
echo "-- bearer/cookie/credential lines in the server logs (must be 0): $(grep -c -E 'dvc_|ses_|bio_|[Bb]earer |[Cc]ookie' "$OUT/server_all.log")"

say "4. stop the backend (whichever process listens now); nothing left"
LAST=$(listener $PORT); [ -n "$LAST" ] && kill "$LAST" 2>/dev/null
i=0; while [ -n "$(listener $PORT)" ] && [ "$i" -lt 80 ]; do i=$((i + 1)); sleep 0.25; done
kill "$NPM_PID" 2>/dev/null; wait "$NPM_PID" 2>/dev/null
: >"$ROOT/.locks/core-server.lane1.pid"
echo "-- shutdown complete lines: $(grep -c 'shutdown complete' "$OUT/server_all.log" 2>/dev/null) (before the last stop) + last: $(tail -n 5 "$OUT/server_restarted.log" "$OUT/server.log" 2>/dev/null | grep -c 'shutdown complete')"
lsof -nP -iTCP:$PORT -iTCP:"$METRO" -iTCP:8081 -sTCP:LISTEN; echo "lsof exit $? (1 = none)"
adb -s emulator-5554 reverse --list; echo "reverse rules listed above (none expected)"
counts | tee "$OUT/counts_after.txt"
say "end"; exit "$DEVICE_EXIT"
