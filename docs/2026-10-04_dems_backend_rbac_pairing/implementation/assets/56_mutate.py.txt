M = [m for m in M if m[0] != 'P31']

def sha(path):
    with open(path, 'rb') as f:
        return hashlib.sha256(f.read()).hexdigest()

def main():
    wanted = set(sys.argv[1:])
    todo = [m for m in M if not wanted or m[0] in wanted]
    files = sorted({m[1] for m in M})
    before = {f: sha(APP + '/' + f) for f in files}
    print('# W56 mutation check, %s; %d test files per run; the tree is not edited' % (time.strftime('%Y-%m-%d %H:%M:%S'), len(TESTS)), flush=True)
    detected, survived, broken = [], [], []
    for (mid, file, old, new, what) in todo:
        source = open(APP + '/' + file).read()
        count = source.count(old)
        if count != 1:
            broken.append(mid)
            print('%s  NOT APPLIED: the text occurs %d times in %s   (%s)' % (mid, count, file, what), flush=True)
            continue
        json.dump({'id': mid, 'file': file, 'from': old, 'to': new}, open(SPEC, 'w'))
        env = dict(os.environ, W56_MUT_SPEC=SPEC, CI='1')
        started = time.time()
        run = subprocess.run(['node', ROOT + '/node_modules/jest/bin/jest.js', '--config', CONFIG, '--maxWorkers=2', '--silent', *TESTS],
                             cwd=APP, env=env, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)
        tail = [line for line in run.stdout.splitlines() if line.startswith('Tests:')]
        failed_files = sorted({line.split()[1].replace('src/auth/', '') for line in run.stdout.splitlines() if line.startswith('FAIL ')})
        verdict = 'detected' if run.returncode != 0 else 'NOT DETECTED'
        (detected if run.returncode != 0 else survived).append(mid)
        print('%s  %-12s %4.0fs  %s  | %s  | %s' % (mid, verdict, time.time() - started, what, (tail[-1].strip() if tail else 'no summary line'), ', '.join(failed_files)), flush=True)
    after = {f: sha(APP + '/' + f) for f in files}
    print('## %d detected, %d not detected, %d not applied, of %d' % (len(detected), len(survived), len(broken), len(todo)), flush=True)
    if survived: print('## not detected: ' + ' '.join(survived), flush=True)
    if broken: print('## not applied: ' + ' '.join(broken), flush=True)
    print('## the mutated files in the tree, before and after: ' + ('identical' if before == after else 'CHANGED'), flush=True)
    for f in files: print('   %s  %s' % (after[f][:16], f), flush=True)
    print('# finished %s' % time.strftime('%H:%M:%S'), flush=True)
    return 0 if not survived and not broken and before == after else 1

if __name__ == '__main__':
    sys.exit(main())
