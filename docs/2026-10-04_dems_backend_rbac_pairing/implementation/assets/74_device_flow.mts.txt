// W74 device check: the Evidence screen in LIVE mode (the server's verdict, the settled site S1-5)
// and in SIMULATED mode, on the phone of lane 1. Every tap goes to a node found by its test ID in a
// `uiautomator dump` (React Native puts a testID into `resource-id`), never to a guessed pixel.
//   node --import tsx device_flow.mts <serial> <out> <T0> <live|simulated>
// Rules (P19, D125): the demo password is read from the shared dataset constant and never printed;
// it is only ever the argument of one `adb shell input text` into the APP's own password field.
// Nothing is typed unless the app has the focus and no keyguard shows (exit 77 otherwise). No
// keyguard, fingerprint or device setting is touched.
import { execFileSync } from 'node:child_process';
import { writeFileSync } from 'node:fs';
import { DEMO_OPERATOR_PASSWORD, OPERATORS } from '../../packages/shared/src/sim/dataset';

const [serial, out, t0, phase] = process.argv.slice(2);
if (!serial || !out || !t0 || (phase !== 'live' && phase !== 'simulated')) {
  console.error('usage: device_flow.mts <serial> <out> <T0> <live|simulated>');
  process.exit(64);
}
const BIG = { encoding: 'utf8' as const, maxBuffer: 256 * 1024 * 1024 };
const adb = (...a: string[]): string => execFileSync('adb', ['-s', serial, ...a], BIG);
const sleep = (ms: number) => new Promise((r) => setTimeout(r, ms));
const jsLog = (): string => adb('logcat', '-d', '-T', t0, '-s', 'ReactNativeJS:V');
const say = (text: string) => console.log(`-- ${new Date().toTimeString().slice(0, 8)} ${text}`);

function focused(): boolean {
  const w = adb('shell', 'dumpsys', 'window');
  return /isKeyguardShowing=false/.test(w) && /mCurrentFocus=.*com\.dems\.demo\.tablet/.test(w);
}

interface Box { l: number; t: number; r: number; b: number }
interface Dump { ids: Map<string, Box>; height: number }
function dump(): Dump {
  adb('shell', 'uiautomator', 'dump', '/sdcard/w74.xml');
  const xml = adb('exec-out', 'cat', '/sdcard/w74.xml');
  const ids = new Map<string, Box>();
  let height = 0;
  for (const [node] of xml.matchAll(/<node\b[^>]*>/g)) {
    const bounds = / bounds="\[(-?\d+),(-?\d+)\]\[(-?\d+),(-?\d+)\]"/.exec(node);
    if (!bounds) continue;
    const box = { l: Number(bounds[1]), t: Number(bounds[2]), r: Number(bounds[3]), b: Number(bounds[4]) };
    height = Math.max(height, box.b);
    const id = / resource-id="([^"]*)"/.exec(node)?.[1]?.replace(/^.*:id\//, '');
    if (id && !ids.has(id)) ids.set(id, box);
  }
  return { ids, height };
}
const visible = (d: Dump, box: Box | undefined): box is Box => box !== undefined && box.b > box.t && box.t >= 0 && box.b <= d.height && box.r > box.l;

/** Waits for a node by test ID (scrolling the content up when asked), then taps its centre. */
async function tapId(id: string, { tries = 12, scroll = false }: { tries?: number; scroll?: boolean } = {}): Promise<void> {
  for (let attempt = 0; attempt < tries; attempt += 1) {
    const d = dump();
    const box = d.ids.get(id);
    if (visible(d, box) && box.b < d.height - 40) {
      if (!focused()) throw Object.assign(new Error('app not focused / keyguard'), { code: 77 });
      adb('shell', 'input', 'tap', String(Math.round((box.l + box.r) / 2)), String(Math.round((box.t + box.b) / 2)));
      say(`tapped ${id}`);
      return;
    }
    if (scroll) adb('shell', 'input', 'swipe', '540', '1700', '540', '900', '400');
    await sleep(1000);
  }
  throw new Error(`no visible node ${id}`);
}
async function waitId(id: string, tries = 20): Promise<boolean> {
  for (let attempt = 0; attempt < tries; attempt += 1) {
    if (dump().ids.has(id)) return true;
    await sleep(1000);
  }
  return false;
}
function shot(name: string): void {
  const png = execFileSync('adb', ['-s', serial, 'exec-out', 'screencap', '-p'], { maxBuffer: BIG.maxBuffer });
  writeFileSync(`${out}/74_${name}.png`, png);
  say(`screencap 74_${name}.png (${png.length} bytes)`);
}
function coreLines(pattern: RegExp): string[] {
  return jsLog().split('\n').map((l) => /DEMS-LIVE\] (.*)$/.exec(l)?.[1] ?? '').filter((l) => pattern.test(l)).map((l) => l.replace(/ [0-9a-f]{8}-[0-9a-f-]{27}$/, ''));
}

/** Signs in on the app's own Login form (the username and the demo password; nothing printed). */
async function signIn(username: string): Promise<void> {
  if (!focused()) throw Object.assign(new Error('app not focused / keyguard'), { code: 77 });
  // The emulator's one-time handwriting coach may take the first focus: tap the field, tap the
  // coach's Cancel (empty space when no coach is up; W60's measured position, phone portrait), tap
  // the field again (W60 session F).
  await tapId('login-username');
  await sleep(1200);
  adb('shell', 'input', 'tap', '540', '2338');
  await sleep(800);
  await tapId('login-username');
  await sleep(600);
  if (!focused()) throw Object.assign(new Error('app not focused / keyguard'), { code: 77 });
  adb('shell', 'input', 'text', username);
  say(`typed the operator username (${username})`);
  await tapId('login-password');
  await sleep(500);
  if (!focused()) throw Object.assign(new Error('app not focused / keyguard'), { code: 77 });
  adb('shell', 'input', 'text', DEMO_OPERATOR_PASSWORD); // never printed
  say('typed the demo password (not shown)');
  await sleep(400);
  adb('shell', 'input', 'keyevent', '66');
  say("pressed the keypad's go key");
}

async function live(): Promise<number> {
  const investigator = OPERATORS.find((o) => o.role === 'INVESTIGATOR');
  if (!investigator) throw new Error('no INVESTIGATOR in the dataset');
  if (!(await waitId('login-username', 40))) say('no Login form: already signed in?');
  else {
    shot('phone_live_login');
    await signIn(investigator.username);
    for (let waited = 0; waited < 30 && coreLines(/POST \/api\/v1\/auth\/sign-in/).length === 0; waited += 1) await sleep(1000);
    say(`the app's sign-in: ${coreLines(/POST \/api\/v1\/auth\/sign-in/).join(' | ') || 'NOT SEEN'}`);
    await sleep(4000);
  }
  // A sealed session of lane 1 with an unacknowledged summary may open first (controller note 3).
  if (dump().ids.has('summary-done')) {
    say('the summary of an earlier session opened first: acknowledging it with the app\'s own Done');
    shot('phone_live_summary_first');
    await tapId('summary-done');
    await sleep(3000);
  }
  await tapId('nav-item-evidence');
  await sleep(3000);
  const rows = [...dump().ids.keys()].filter((id) => /^evidence-row-EV-\d{8}-\d{4}$/.test(id));
  say(`evidence rows on screen: ${rows.length}`);
  shot('phone_live_list');
  const row = rows.find((id) => id.endsWith('-0001')) ?? rows[0];
  if (!row) throw new Error('no evidence row');
  const evidenceId = row.replace('evidence-row-', '');
  await tapId(row);
  await sleep(2500);
  await tapId('evidence-tab-integrity', { scroll: true });
  await sleep(1500);
  await tapId('evidence-verify', { scroll: true });
  await sleep(3500);
  shot('phone_live_verdict');
  say(`verify: ${coreLines(new RegExp(`POST /api/v1/evidence/${evidenceId}/verify`)).join(' | ') || 'NOT SEEN'}`);
  // Site S1-5: an INVESTIGATOR's export is refused by the Core and audited.
  await tapId('evidence-action-export');
  await sleep(4000);
  shot('phone_live_export_denied');
  say(`export probe: ${coreLines(new RegExp(`POST /api/v1/evidence/${evidenceId}/export`)).join(' | ') || 'NOT SEEN'}`);
  say(`media reference: ${coreLines(new RegExp(`GET /api/v1/evidence/${evidenceId}`)).join(' | ') || 'NOT SEEN'}`);
  const log = jsLog();
  say(`credential/token lines in the JS log (must be 0): ${log.split('\n').filter((l) => /dvc_|ses_|bio_/.test(l)).length}`);
  say(`demo operations in the JS log (must be 0): ${coreLines(/\/api\/v1\/demo\//).length}`);
  return 0;
}

async function simulated(): Promise<number> {
  const investigator = OPERATORS.find((o) => o.role === 'INVESTIGATOR');
  if (!investigator) throw new Error('no INVESTIGATOR in the dataset');
  if (await waitId('login-username', 30)) {
    await signIn(investigator.username);
    await sleep(5000);
  }
  if (dump().ids.has('summary-done')) {
    await tapId('summary-done');
    await sleep(2000);
  }
  await tapId('nav-item-evidence');
  await sleep(3000);
  shot('phone_simulated_list');
  const rows = [...dump().ids.keys()].filter((id) => /^evidence-row-EV-\d{8}-\d{4}$/.test(id));
  const row = rows.find((id) => id.endsWith('-0001')) ?? rows[0];
  if (row) {
    await tapId(row);
    await sleep(2500);
    await tapId('evidence-tab-integrity', { scroll: true });
    await sleep(1500);
    shot('phone_simulated_integrity');
  }
  say(`Core requests in SIMULATED mode (must be 0): ${jsLog().split('\n').filter((l) => l.includes('DEMS-LIVE]')).length}`);
  return 0;
}

(phase === 'live' ? live() : simulated()).then(
  (status) => process.exit(status),
  (error: unknown) => {
    const code = (error as { code?: unknown }).code === 77 ? 77 : 1;
    console.error(`-- failed: ${error instanceof Error ? error.message : 'unknown'}`);
    process.exit(code);
  },
);
