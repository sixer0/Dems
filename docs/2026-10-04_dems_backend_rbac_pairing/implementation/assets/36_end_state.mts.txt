// Scratch script of plan row W36 (outside the repository): what is left in the PostgreSQL
// instance. It connects as dems_owner through the test helper's own reader of the connection
// file and prints names and counts only.
import pg from '/Users/gsp-budi/Documents/DEMS/node_modules/pg/lib/index.js';
import { readOwnerCredentials } from '/Users/gsp-budi/Documents/DEMS/apps/core-server/test/helpers/db/credentials';

const owner = readOwnerCredentials();
const client = new pg.Client({ host: owner.host, port: owner.port, database: owner.database, user: owner.user, password: owner.password.reveal(), ssl: false });
await client.connect();
const one = async (text: string, values: unknown[] = []): Promise<Record<string, unknown>[]> => (await client.query(text, values)).rows;
const session = (await one("SELECT current_user AS role, current_setting('server_version') AS version, current_setting('port') AS port, pg_postmaster_start_time()::text AS started"))[0];
console.log(`session: role=${session?.role} server=${session?.version} port=${session?.port} postmaster started ${session?.started}`);
const mine = await one("SELECT datname FROM pg_database WHERE datname LIKE 'dems\\_test\\_step\\_58\\_%' ORDER BY 1");
console.log(`databases with this row's prefix (dems_test_step_58_*): ${mine.length}${mine.length > 0 ? ` (${mine.map((row) => row.datname).join(', ')})` : ''}`);
const tests = await one("SELECT count(*)::int AS n FROM pg_database WHERE datname LIKE 'dems\\_test\\_%'");
console.log(`test databases of all rows at this moment: ${tests[0]?.n}`);
const project = await one("SELECT datname FROM pg_database WHERE datname LIKE 'dems\\_%' AND datname NOT LIKE 'dems\\_test\\_%' ORDER BY 1");
console.log(`project databases: ${project.map((row) => row.datname).join(', ')}`);
const open = await one("SELECT count(*)::int AS n FROM pg_stat_activity WHERE datname LIKE 'dems\\_test\\_step\\_58\\_%'");
console.log(`sessions on this row's databases still open: ${open[0]?.n}`);
await client.end();
