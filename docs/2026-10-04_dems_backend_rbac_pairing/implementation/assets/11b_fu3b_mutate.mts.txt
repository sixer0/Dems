function edit(file, from, to) {
  const text = readFileSync(file, 'utf8');
  if (!text.includes(from)) throw new Error(`anchor not found in ${file}: ${from.slice(0, 60)}`);
  writeFileSync(file, text.replace(from, to));
}

function restoreAll(files) {
  for (const [file, text] of files) writeFileSync(file, text);
}

function snapshot(files) {
  return files.map((file) => [file, readFileSync(file, 'utf8')]);
}

function run(project, files) {
  try {
    const out = execFileSync('npx', ['vitest', 'run', '--project', project, '--maxWorkers=1', '--root', 'apps/core-server', ...files], {
      cwd: ROOT,
      encoding: 'utf8',
      env: { ...process.env, DEMS_TEST_STEP: 'w11b3m' },
      timeout: 600_000,
    });
    return { failed: 0, text: out };
  } catch (error) {
    const text = `${error.stdout ?? ''}${error.stderr ?? ''}`;
    const match = /Tests\s+(\d+) failed \| (\d+) passed \| (\d+) skipped|Tests\s+(\d+) failed\s+\((\d+)\)|Tests\s+(\d+) passed \((\d+)\)/.exec(text);
    return { failed: -1, text, summary: summaryOf(text) };
  }
}

function summaryOf(text) {
  const failed = /Tests\s+(\d+) failed/.exec(text);
  const total = /Tests\s+[^\n]*\((\d+)\)/.exec(text);
  return { failed: failed ? Number(failed[1]) : 0, total: total ? Number(total[1]) : 0 };
}
