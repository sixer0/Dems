method of M0 / W56 / W78) and runs the test files of its group. A mutation is DETECTED when jest exits
non-zero. Nothing in the tree is written: the hashes of the mutated files are compared before and after.

def sha(path):
    with open(path, 'rb') as f:
        return hashlib.sha256(f.read()).hexdigest()

def jest(env, tests):
    # The app's babel resolves `@/` from its own folder: the jest subprocess gets cwd=apps/mobile (as M0's / W78's runner); no shell cd.
    return subprocess.run(['node', ROOT + '/node_modules/jest/bin/jest.js', '--config', CONFIG, '--maxWorkers=2', '--silent', *[APP + '/' + t for t in tests]],
                          cwd=APP, env=env, stdout=subprocess.PIPE, stderr=subprocess.STDOUT, text=True)

def summary(out):
    tail = [line.strip() for line in out.splitlines() if line.startswith('Tests:')]
    files = sorted({line.split()[1].split('/')[-1] for line in out.splitlines() if line.startswith('FAIL ')})
    first = [re.sub(r'\s+', ' ', line.strip().lstrip('●').strip()) for line in out.splitlines() if line.strip().startswith('● ') and 'Console' not in line]
    return (tail[-1] if tail else 'no summary line'), files, first[:1]

def main():
    group = sys.argv[1]
    wanted = set(sys.argv[2:])
    tests = GROUP_TESTS[group]
    spec_path = HERE + '/spec_' + group + '.json'
    files = sorted({m[1] for m in M[group]})
    before = {f: sha(APP + '/' + f) for f in files}
    print('# W79 mutation check, group %s, %s; %d test files per run; the tree is not edited' % (group, time.strftime('%Y-%m-%d %H:%M:%S'), len(tests)), flush=True)
    if not wanted or 'control' in wanted:
        started = time.time()
        run = jest(dict(os.environ, CI='1'), tests)
        tail, failed, _ = summary(run.stdout)
        print('control (no mutation): exit=%d %s (%.0f s)' % (run.returncode, tail, time.time() - started), flush=True)
        if run.returncode != 0:
            print(run.stdout[-3000:])
            print('## the control run is not green: stop', flush=True)
            return 2
    todo = [m for m in M[group] if not wanted or m[0] in wanted]
    detected, survived, broken = [], [], []
    for (mid, file, what, edits) in todo:
        text = open(APP + '/' + file).read()
        ok = True
        for edit in edits:
            if text.count(edit['from']) != 1:
                ok = False
                break
            text = text.replace(edit['from'], edit['to'], 1)
        if not ok:
            broken.append(mid)
            print('%s  NOT APPLIED: a text does not occur exactly once in %s   (%s)' % (mid, file, what), flush=True)
            continue
        json.dump({'id': mid, 'file': file, 'edits': edits}, open(spec_path, 'w'))
        started = time.time()
        run = jest(dict(os.environ, W79_MUT_SPEC=spec_path, CI='1'), tests)
        tail, failed, first = summary(run.stdout)
        verdict = 'detected' if run.returncode != 0 else 'NOT DETECTED'
        (detected if run.returncode != 0 else survived).append(mid)
        print('%-4s %-12s %s (%.0f s) — %s; failing: %s; %s' % (mid, verdict, file.split('/')[-1], time.time() - started, what, ', '.join(failed) or '-', tail), flush=True)
        if first:
            print('       first failure: %s' % first[0][:200], flush=True)
    after = {f: sha(APP + '/' + f) for f in files}
    print('## %d detected, %d not detected, %d not applied, of %d' % (len(detected), len(survived), len(broken), len(todo)), flush=True)
    print('## the mutated files in the tree, before and after: %s' % ('identical' if before == after else 'CHANGED'), flush=True)
    return 0 if not survived and not broken and before == after else 1

if __name__ == '__main__':
    sys.exit(main())
