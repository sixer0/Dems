for mu in M:
    mu['tests'] = BY_PREFIX[mu['id'][0]]

def check():
    bad = 0
    for mu in M + [RELAX]:
        with open(os.path.join(ROOT, mu['file']), encoding='utf8') as handle:
            text = handle.read()
        count = text.count(mu['find'])
        if count != 1:
            bad += 1
            print(f"{mu['id']}: the text occurs {count} times in {mu['file']}")
    ids = [mu['id'] for mu in M]
    assert len(ids) == len(set(ids)), 'duplicate id'
    print(f"{len(M)} mutations and 1 relaxation; texts that do not occur exactly once: {bad}")
    return bad

def vitest(mid, tests):
    env = dict(os.environ)
    env['W55_APPLIED_DIR'] = APPLIED
    if mid:
        env['W55_MUT'] = mid
    else:
        env.pop('W55_MUT', None)
    started = time.time()
    done = subprocess.run(['npx', 'vitest', 'run', '--config', CONFIG, '--maxWorkers=2', *tests], cwd=ROOT, env=env, capture_output=True, text=True)
    out = done.stdout + done.stderr
    summary = ' | '.join(line.strip() for line in out.splitlines() if line.strip().startswith(('Test Files', 'Tests ')))
    first = next((line.strip() for line in out.splitlines() if line.strip().startswith(('×', 'FAIL'))), '')
    return done.returncode, summary, first[:230], time.time() - started

def stamp():
    return time.strftime('%Y-%m-%d %H:%M:%S WIB')

def run(ids):
    json.dump(M + [RELAX], open(os.path.join(HERE, 'mutations.json'), 'w'), indent=1)
    if check():
        sys.exit(2)
    code, summary, _first, took = vitest('', ALL)
    print(f"## {stamp()} baseline: the six test files of the row, no mutation\nbaseline exit={code} {summary} ({took:.1f} s)", flush=True)
    if code != 0:
        sys.exit(3)
    chosen = [mu for mu in M if not ids or mu['id'] in ids]
    survived, not_applied = [], []
    for mu in chosen:
        marker = os.path.join(APPLIED, mu['id'])
        if os.path.exists(marker):
            os.remove(marker)
        code, summary, first, took = vitest(mu['id'], mu['tests'])
        applied = os.path.exists(marker)
        if not applied:
            not_applied.append(mu['id'])
            verdict = 'NOT APPLIED'
        elif code != 0:
            verdict = 'DETECTED   '
        else:
            verdict = 'SURVIVED   '
            survived.append(mu['id'])
        print(f"{mu['id']} {verdict} exit={code} {mu['file'].split('/')[-1]}: {mu['what']} [{summary}] first: {first}", flush=True)
    print(f"## {stamp()} {len(chosen)} mutations: {len(chosen) - len(survived) - len(not_applied)} detected, {len(survived)} survived {survived}, {len(not_applied)} not applied {not_applied}", flush=True)

def relax():
    json.dump(M + [RELAX], open(os.path.join(HERE, 'mutations.json'), 'w'), indent=1)
    marker = os.path.join(APPLIED, RELAX['id'])
    if os.path.exists(marker):
        os.remove(marker)
    code, summary, first, took = vitest('', RELAX['tests'])
    print(f"## {stamp()} W50's start-up test file as it is on disk: exit={code} {summary} first: {first}", flush=True)
    code, summary, first, took = vitest(RELAX['id'], RELAX['tests'])
    print(f"## {stamp()} the same file with line 398 relaxed IN MEMORY (applied: {os.path.exists(marker)}): exit={code} {summary} first: {first}", flush=True)

if __name__ == '__main__':
    mode = sys.argv[1] if len(sys.argv) > 1 else 'check'
    if mode == 'check':
        json.dump(M + [RELAX], open(os.path.join(HERE, 'mutations.json'), 'w'), indent=1)
        sys.exit(1 if check() else 0)
    elif mode == 'relax':
        relax()
    else:
        run(sys.argv[2:])
