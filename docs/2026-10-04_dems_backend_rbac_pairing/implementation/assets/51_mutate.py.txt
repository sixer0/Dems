def sha(path):
    return hashlib.sha256(open(path, 'rb').read()).hexdigest()

def run(spec):
    env = dict(os.environ)
    if spec is None:
        env.pop('W51_MUT_SPEC', None)
    else:
        json.dump(spec, open(SPEC, 'w'))
        env['W51_MUT_SPEC'] = SPEC
    started = time.time()
    done = subprocess.run(['npm', 'exec', '-w', '@dems/mobile', '--', 'jest', '--config', CONFIG, '--maxWorkers=2', '--silent', '--forceExit', *TESTS],
                          cwd=ROOT, env=env, capture_output=True, text=True)
    out = done.stdout + done.stderr
    failed = [line.strip() for line in out.splitlines() if line.strip().startswith('●') and '›' in line]
    summary = next((line.strip() for line in out.splitlines() if line.startswith('Tests:')), '(no summary)')
    broken = 'W51 mutation' in out and 'occurs' in out
    return done.returncode, summary, failed, broken, time.time() - started, out

def main():
    wanted = set(sys.argv[1:])
    files = sorted({m[1] for m in M})
    before = {f: sha(APP + '/' + f) for f in files}
    print(f"## W51 mutation check, {time.strftime('%Y-%m-%d %H:%M:%S')}; {len(TESTS)} test files per run; the tree is not edited")
    if not wanted:
        code, summary, failed, _, took, out = run(None)
        print(f"control (no mutation): exit={code} {summary} ({took:.0f} s)")
        if code != 0:
            print(out[-3000:])
            sys.exit(2)
    detected = survived = 0
    for mid, file, old, new, what in M:
        if wanted and mid not in wanted:
            continue
        code, summary, failed, broken, took, out = run({'id': mid, 'file': file, 'from': old, 'to': new})
        if broken:
            print(f"{mid} NOT APPLIED ({file}): the text does not occur exactly once")
            survived += 1
            continue
        if code != 0 and not failed and 'Tests:' not in out:
            print(f"{mid} ERROR ({file}): jest gave no result\n{out[-1500:]}")
            survived += 1
            continue
        verdict = 'DETECTED' if code != 0 else 'SURVIVED'
        detected += code != 0
        survived += code == 0
        first = failed[0][:150] if failed else ''
        print(f"{mid} {verdict} ({took:.0f} s) {file.split('/')[-1]}: {what} | {summary} | {first}")
        sys.stdout.flush()
    after = {f: sha(APP + '/' + f) for f in files}
    print(f"## {detected} detected, {survived} not detected, of {detected + survived}")
    print("## the mutated files in the tree, before and after: " + ("identical" if before == after else "CHANGED"))
    for f in files:
        print(f"   {before[f][:16]} {after[f][:16]} {f}")

main()
