// W72 device check: a queue-driven UI driver (W67's pattern, `implementation/assets/67_drive_q.mts.txt`).
// The agent writes numbered command files (`<out>/q/NNN.cmd`, one command per line) after reading the
// screencap of the previous step; the driver runs them in order, writes `NNN.out`, a screencap
// `NNN.png` (unless the batch says `noshot`), and one line `ACK NNN …` to `<out>/q/events.log`.
// Exits with `done <code>`, or 6 when no command arrives for 90 s.
//
// Device rules (D164 DN-7): `ids` (uiautomator) runs ONLY with a declared screen of the allow-list
// (`ids session-new`, `ids sessions`): never on Login, the Summary or Home. Credential rules:
// `password` types the demo password from the shared dataset constant (never printed, logged or
// written), and ONLY when the IME reports a password field as the focused editor. After `password`,
// no screencap is taken until a sign-in request was answered 200 (`waitlog signin`); if the sign-in
// is not answered 200, the app is force-stopped (memory and fields cleared) before anything else.
// Nothing is typed or tapped unless the app has the focus and no keyguard shows. No keyguard, PIN,
// fingerprint or device setting is ever touched.
import { execFileSync } from 'node:child_process';
import { appendFileSync, existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs';
import { DEMO_OPERATOR_PASSWORD, OPERATORS } from '../../packages/shared/src/sim/dataset';

const [serial, out, t0] = process.argv.slice(2);
if (!serial || !out || !t0) {
  console.error('usage: drive_q.mts <serial> <out> <T0>');
  process.exit(64);
}
const APP = 'com.dems.demo.tablet';
const Q = `${out}/q`;
mkdirSync(Q, { recursive: true });
const BIG = { encoding: 'utf8' as const, maxBuffer: 256 * 1024 * 1024 };
const adb = (...args: string[]): string => execFileSync('adb', ['-s', serial, ...args], BIG);
const sleep = (ms: number) => new Promise((resolve) => setTimeout(resolve, ms));
const stamp = () => new Date().toTimeString().slice(0, 8);
const event = (line: string) => {
  appendFileSync(`${Q}/events.log`, `${line}\n`);
  console.log(`-- ${stamp()} ${line}`);
};
const jsLog = (): string => adb('logcat', '-d', '-T', t0, '-s', 'ReactNativeJS:V');
const investigator = OPERATORS.find((operator) => operator.role === 'INVESTIGATOR');

function focused(): boolean {
  const window = adb('shell', 'dumpsys', 'window');
  return /isKeyguardShowing=false/.test(window) && /mCurrentFocus=.*com\.dems\.demo\.tablet/.test(window);
}
function editorInputType(): number | null {
  const dump = adb('shell', 'dumpsys', 'input_method');
  const types = [...dump.matchAll(/inputType=0x([0-9a-f]+)/g)].map((m) => Number.parseInt(m[1]!, 16));
  return types.length > 0 ? types[types.length - 1]! : null;
}
const isPasswordType = (type: number | null): boolean => type !== null && (type & 0xf) === 0x1 && [0x80, 0x90, 0xe0].includes(type & 0xff0);

const PATTERNS: Record<string, RegExp> = {
  signin: /DEMS-LIVE\] POST \/api\/v1\/auth\/sign-in (\d{3})/,
  create: /DEMS-LIVE\] POST \/api\/v1\/session (\d{3})/,
  precheck: /DEMS-LIVE\] POST \/api\/v1\/session\/\S+\/precheck (\d{3})/,
  pack: /DEMS-LIVE\] POST \/api\/v1\/session\/\S+\/precheck\/acknowledge (\d{3})/,
  start: /DEMS-LIVE\] POST \/api\/v1\/session\/\S+\/start (\d{3})/,
  stop: /DEMS-LIVE\] POST \/api\/v1\/session\/\S+\/stop (\d{3})/,
  ack: /DEMS-LIVE\] POST \/api\/v1\/session\/\S+\/summary\/acknowledge (\d{3})/,
};
const seen = new Set<string>();
let passwordPending = false;
const IDS_ALLOWED = new Set(['session-new', 'sessions']);

async function waitLog(name: string, seconds: number): Promise<string> {
  const pattern = PATTERNS[name];
  if (!pattern) return `unknown pattern ${name}`;
  for (let waited = 0; waited < seconds * 4; waited += 1) {
    const line = jsLog()
      .split('\n')
      .find((l) => pattern.test(l) && !seen.has(l));
    if (line !== undefined) {
      seen.add(line);
      const status = pattern.exec(line)?.[1] ?? '?';
      if (name === 'signin') {
        if (status === '200') passwordPending = false;
        else return `signin ${status}`;
      }
      return `${name} ${status} (${line.replace(/^.*DEMS-LIVE\] /, '').replace(/ [0-9a-f]{8}-[0-9a-f-]{27}\s*$/, '')})`;
    }
    await sleep(250);
  }
  return `${name} NOT SEEN in ${seconds} s`;
}

function shot(path: string): string {
  if (passwordPending) return 'screencap refused: a password was typed and the sign-in is not answered 200';
  const png = execFileSync('adb', ['-s', serial, 'exec-out', 'screencap', '-p'], { maxBuffer: BIG.maxBuffer });
  writeFileSync(path, png);
  return `screencap ${path.split('/').pop()} (${png.length} bytes)`;
}

async function run(line: string): Promise<string> {
  const [verb, ...rest] = line.trim().split(/\s+/);
  switch (verb) {
    case 'tap':
      if (!focused()) return 'NOT FOCUSED: tap skipped';
      adb('shell', 'input', 'tap', rest[0]!, rest[1]!);
      return `tap ${rest[0]},${rest[1]}`;
    case 'text': {
      if (!focused()) return 'NOT FOCUSED: text skipped';
      const type = editorInputType();
      if (isPasswordType(type)) return 'REFUSED: the focused editor is a password field';
      adb('shell', 'input', 'text', rest.join('%s'));
      return `text typed (${rest.join(' ')}); editor inputType 0x${(type ?? 0).toString(16)}`;
    }
    case 'username': {
      if (!focused() || !investigator) return 'NOT FOCUSED: username skipped';
      const type = editorInputType();
      if (isPasswordType(type)) return 'REFUSED: the focused editor is a password field';
      adb('shell', 'input', 'text', investigator.username);
      return `username typed (${investigator.username}); editor inputType 0x${(type ?? 0).toString(16)}`;
    }
    case 'password': {
      if (!focused()) return 'NOT FOCUSED: password skipped';
      const type = editorInputType();
      if (!isPasswordType(type)) return `REFUSED: the focused editor is not a password field (inputType 0x${(type ?? 0).toString(16)}); nothing typed`;
      passwordPending = true;
      adb('shell', 'input', 'text', DEMO_OPERATOR_PASSWORD); // never printed
      return `password typed (not shown); editor inputType 0x${(type ?? 0).toString(16)}`;
    }
    case 'key':
      if (!focused()) return 'NOT FOCUSED: key skipped';
      adb('shell', 'input', 'keyevent', rest[0]!);
      return `key ${rest[0]}`;
    case 'link':
      adb('shell', 'am', 'start', '-W', '-a', 'android.intent.action.VIEW', '-d', rest[0]!, APP);
      return `deep link ${rest[0]}`;
    case 'waitlog': {
      const result = await waitLog(rest[0]!, Number(rest[1] ?? '20'));
      if (rest[0] === 'signin' && passwordPending) {
        adb('shell', 'am', 'force-stop', APP);
        passwordPending = false;
        return `${result}; the app was force-stopped (fields cleared) before any screencap`;
      }
      return result;
    }
    case 'sleep':
      await sleep(Number(rest[0] ?? '1000'));
      return `slept ${rest[0]} ms`;
    case 'shot':
      return shot(`${out}/72_phone_${rest[0]}.png`);
    case 'focus':
      return `focused ${focused()}; editor inputType 0x${(editorInputType() ?? 0).toString(16)}`;
    case 'ids': {
      const screenName = rest[0] ?? '';
      if (!IDS_ALLOWED.has(screenName)) return `ids REFUSED: declare an allowed screen (${[...IDS_ALLOWED].join(', ')}); never Login, Summary or Home`;
      try {
        execFileSync('adb', ['-s', serial, 'shell', 'uiautomator', 'dump', '/sdcard/w72.xml'], { ...BIG, timeout: 7000 });
        const xml = adb('shell', 'cat', '/sdcard/w72.xml');
        const found = [...xml.matchAll(/<node\b[^>]*>/g)]
          .map((m) => m[0])
          .map((n) => ({ id: (/ resource-id="([^"]*)"/.exec(n)?.[1] ?? '').replace(/^.*:id\//, ''), b: / bounds="([^"]*)"/.exec(n)?.[1] ?? '' }))
          .filter((n) => /^(session-new|session-row|sessions|screen)-/.test(n.id));
        return `ids: ${found.map((n) => `${n.id}${n.b}`).join(' ')}`;
      } catch {
        return 'ids: uiautomator could not dump within 7 s';
      }
    }
    default:
      return `unknown command: ${line}`;
  }
}

async function main(): Promise<number> {
  let next = 1;
  for (;;) {
    const id = String(next).padStart(3, '0');
    const file = `${Q}/${id}.cmd`;
    let waited = 0;
    while (!existsSync(file) && waited < 90_000) {
      await sleep(250);
      waited += 250;
    }
    if (!existsSync(file)) {
      event(`EXIT 6 no command ${id} within 90 s`);
      return 6;
    }
    await sleep(150);
    const lines = readFileSync(file, 'utf8').split('\n').map((l) => l.trim()).filter((l) => l !== '' && !l.startsWith('#'));
    const results: string[] = [];
    let noshot = false;
    for (const line of lines) {
      if (line === 'noshot') {
        noshot = true;
        continue;
      }
      const done = /^done\s+(\d+)/.exec(line);
      if (done) {
        writeFileSync(`${Q}/${id}.out`, `${results.join('\n')}\ndone ${done[1]}\n`);
        event(`ACK ${id} done ${done[1]}`);
        return Number(done[1]);
      }
      try {
        results.push(await run(line));
      } catch (error) {
        results.push(`ERROR in "${line.split(/\s+/)[0]}": ${error instanceof Error ? error.message.split('\n')[0] : 'unknown'}`);
      }
    }
    if (!noshot) {
      try {
        results.push(shot(`${Q}/${id}.png`));
      } catch (error) {
        results.push(`screencap failed: ${error instanceof Error ? error.message.split('\n')[0] : 'unknown'}`);
      }
    }
    writeFileSync(`${Q}/${id}.out`, `${results.join('\n')}\n`);
    event(`ACK ${id} | ${results.join(' | ')}`);
    next += 1;
  }
}

main().then(
  (status) => process.exit(status),
  (error: unknown) => {
    event(`EXIT 1 failed: ${error instanceof Error ? error.message : 'unknown'}`);
    process.exit(1);
  },
);
