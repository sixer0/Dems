#!/bin/sh
# W82 device commands, run by device-session.sh on the phone; lane-1 backend on 4421 via tcp:4420.
# One sitting: SIMULATED -> (Settings segment LIVE: the §3.3 dialog, the simulator's sign-out) -> LIVE Login
# -> LIVE sign-in -> Settings in LIVE (the third segment, the biometric switch's wording, the Demo controls
# panel shown, the Profile sheet) -> (segment SIMULATED: the dialog, the Core's sign-out) -> SIMULATED Login.
# Ends in SIMULATED (D124). No demo control, no biometric "Enable", nothing on a keyguard. Removes the one
# dump file this row creates (/sdcard/w82.xml). Exit 77 when locked: reported, never worked around.
set -u
OUT=$1; H="$ROOT/.cache/w82"; A="adb -s $SERIAL"
echo "-- serial $SERIAL ($AVD_NAME), Metro $METRO_PORT"
unlocked() {
  KG=$($A shell dumpsys window 2>/dev/null | tr -d '\r' | grep -o 'isKeyguardShowing=[a-z]*' | head -n 1)
  ST=$($A shell dumpsys user 2>/dev/null | grep -m 1 "State:" | tr -d '\r' | sed 's/^ *//')
  case "$KG/$ST" in isKeyguardShowing=false/*RUNNING_UNLOCKED*) return 0 ;; *) echo "-- LOCKED ($KG; $ST) at: $1 - stop, report."; return 1 ;; esac
}
launch() { echo "-- am start -W -d '$1'"; $A shell "am start -W -a android.intent.action.VIEW -d '$1' $APP_ID" 2>&1 | tr -d '\r' | grep -E "Status|LaunchState|Error|TotalTime" | sed 's/^/   /'; }
simauth() { $A shell "run-as $APP_ID sh -c 'sqlite3 databases/RKStorage \"select value from catalystLocalStorage where key = \\\"dems.sim.v1\\\"\"'" 2>/dev/null | tr -d '\r' | grep -o '"auth":{"status":"[A-Z_]*"' | head -n 1; }
prefs() { $A shell "run-as $APP_ID sh -c 'sqlite3 databases/RKStorage \"select value from catalystLocalStorage where key = \\\"dems.mobile.prefs.v1\\\"\"'" 2>/dev/null | tr -d '\r'; }
unlocked "the start" || exit 77
echo "-- before: prefs $(prefs); simulator auth $(simauth)"

T0=$($A shell "date '+%m-%d %H:%M:%S.000'" | tr -d '\r')
launch 'demsdemo:///login?core=SIMULATED&motion=off'
i=0; until $A logcat -d -T "$T0" -s ReactNativeJS:V 2>/dev/null | grep -q 'runtime.sim.hydrated' || [ "$i" -ge 90 ]; do i=$((i + 1)); sleep 1; done
echo "-- waited $i s for runtime.sim.hydrated"
unlocked "the SIMULATED login" || exit 77
SIM_AUTH=$(simauth); echo "-- simulator auth at the start: $SIM_AUTH"
(cd "$ROOT" && SIM_AUTH="$SIM_AUTH" node --import tsx "$H/device_flow.mts" "$SERIAL" "$OUT" "$T0")
FLOW_EXIT=$?; echo "-- flow exit: $FLOW_EXIT"
[ "$FLOW_EXIT" = 77 ] && exit 77
echo "-- Core request lines of the sitting (method, path template, status):"
$A logcat -d -T "$T0" -s ReactNativeJS:V 2>/dev/null | grep -o 'DEMS-LIVE\] .*' | sed -E 's/ [0-9a-f]{8}-[0-9a-f-]{27}$//' | sort | uniq -c | sort -rn | head -n 20 | sed 's/^/   /'
echo "-- prefs after: $(prefs)"
echo "-- remove this row's dump file only: adb shell rm /sdcard/w82.xml"
$A shell rm -f /sdcard/w82.xml; $A shell ls /sdcard/w82.xml 2>&1 | tr -d '\r' | sed 's/^/   /'
echo "-- lock at end: $($A shell dumpsys window 2>/dev/null|tr -d '\r'|grep -o 'isKeyguardShowing=[a-z]*'|head -n1)"
exit "$FLOW_EXIT"
