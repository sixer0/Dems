def sha(path):
    with open(os.path.join(ROOT, path), 'rb') as handle:
        return hashlib.sha256(handle.read()).hexdigest()


def run_tests(mid, tests):
    env = dict(os.environ, W77_MUT=mid, W77_APPLIED_DIR=APPLIED)
    files = [test.split('::')[0] for test in tests]
    cmd = ['npx', 'vitest', 'run', '--config', os.path.join(ROOT, CONFIG), '--maxWorkers=2', *files]
    started = time.time()
    proc = subprocess.run(cmd, cwd=os.path.join(ROOT, 'apps/core-web'), env=env, capture_output=True, text=True)
    out = proc.stdout + proc.stderr
    summary = [line.strip() for line in out.splitlines() if line.strip().startswith(('Test Files', 'Tests '))]
    return proc.returncode, summary, round(time.time() - started), out


def main():
    mode = sys.argv[1] if len(sys.argv) > 1 else 'check'
    ids = sys.argv[2:]
    with open(os.path.join(HERE, 'mutations.json'), 'w') as handle:
        json.dump(M, handle, indent=1)
    chosen = [entry for entry in M if not ids or entry['id'] in ids]
    if mode == 'check':
        for entry in chosen:
            text = open(os.path.join(ROOT, entry['file'])).read()
            count = text.count(entry['find'])
            print(f"{entry['id']}: occurs {count}x in {entry['file']}" + ('' if count == 1 else '  <-- FIX'))
        return
    files = sorted({entry['file'] for entry in chosen})
    before = {path: sha(path) for path in files}
    results = []
    for entry in chosen:
        marker = os.path.join(APPLIED, entry['id'])
        if os.path.exists(marker):
            os.remove(marker)
        code, summary, secs, out = run_tests(entry['id'], entry['tests'])
        applied = os.path.exists(marker)
        failed = code != 0
        if not applied:
            verdict = 'NOT APPLIED'
        elif entry['expect'] == 'green':
            verdict = 'GREEN (as expected)' if not failed else 'RED (unexpected)'
        else:
            verdict = 'DETECTED' if failed else 'SURVIVED'
        results.append((entry['id'], verdict))
        print(f"{entry['id']} [{verdict}] {entry['what']} :: {' | '.join(summary)} ({secs}s)", flush=True)
        if entry['expect'] == 'green' and failed:
            print('\n'.join(line for line in out.splitlines() if 'FAIL' in line or 'Error' in line)[:3000])
    after = {path: sha(path) for path in files}
    print('files unchanged by the batch:', 'identical' if before == after else 'CHANGED ' + str([p for p in files if before[p] != after[p]]))
    stamp = time.strftime('%Y-%m-%d %H:%M:%S WIB')
    detected = [mid for mid, verdict in results if verdict == 'DETECTED']
    survived = [mid for mid, verdict in results if verdict == 'SURVIVED']
    print(f"## {stamp} {len(results)} entries: {len(detected)} detected, {len(survived)} survived {survived}, others {[r for r in results if r[1] not in ('DETECTED', 'SURVIVED')]}")


if __name__ == '__main__':
    main()
