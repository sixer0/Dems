#!/bin/sh
# W86 README check, device step, on LANE 1 (backend 127.0.0.1:4421, dems_lane_1; phone dems2_phone is
# paired there as DEV-0001). The README documents the main instance (4420); a device credential belongs
# to one database, so the README's mobile lines are exercised here with the lane's port: the device side
# stays tcp:4420 and device-session.sh maps it to host tcp:4421 (DEMS_CORE_HOST_PORT).
# Run from the workspace root, INSIDE the lane lock:
#   /usr/bin/lockf -k -t 300 .locks/core-server.lane1.lock /bin/sh .cache/w86/run_lane.sh
# No lane reset, no db:* command on the lane database, no demo control, no sign-in on the device.
# Never touches dems2_phone2 (reserved) or dems2_tablet. Exit 77 from the device session = locked:
# reported, never worked around.
set -u
ROOT=/Users/gsp-budi/Documents/DEMS
[ "$(pwd)" = "$ROOT" ] || { echo "ABORT: run from $ROOT"; exit 9; }
OUT=.cache/w86/out; PORT=4421; METRO=8083
mkdir -p "$OUT"
stamp() { date '+%H:%M:%S'; }
say() { echo; echo "== $(stamp) $*"; }
listening() { lsof -nP -iTCP:"$1" -sTCP:LISTEN >/dev/null 2>&1; }
BACK_NPM=""
stop_backend() {
  [ -n "$BACK_NPM" ] || return 0
  P=$(lsof -nP -tiTCP:$PORT -sTCP:LISTEN 2>/dev/null | head -n 1)
  [ -n "$P" ] && kill -TERM "$P"
  i=0; while listening $PORT && [ $i -lt 60 ]; do i=$((i + 1)); sleep 0.25; done
  wait "$BACK_NPM" 2>/dev/null; echo "-- $(stamp) lane backend stopped (npm exit $?)"
  BACK_NPM=""
  : > .locks/core-server.lane1.pid
}
trap stop_backend EXIT

say "0. ports $PORT and $METRO free; lock status (read-only)"
for p in $PORT $METRO; do listening $p && { echo "ABORT: $p in use"; lsof -nP -iTCP:$p -sTCP:LISTEN; exit 3; }; done
echo "free: $PORT $METRO"
apps/mobile/scripts/emulator-user-lock.sh status > "$OUT/b00_lock_status_before.txt" 2>&1; echo "lock status exit $?"; cat "$OUT/b00_lock_status_before.txt"

say "1. lane-1 backend: DEMS_CORE_PORT=$PORT npm run start:server (demo flag off)"
env -u DEMS_DEMO_CONTROLS DEMS_CORE_PORT=$PORT npm run start:server > "$OUT/b01_lane_server.log" 2>&1 &
BACK_NPM=$!
i=0; code=000
while [ $i -lt 240 ]; do
  code=$(curl -s -o /dev/null -w '%{http_code}' http://127.0.0.1:$PORT/api/v1/health 2>/dev/null)
  [ "$code" = "200" ] && break
  kill -0 "$BACK_NPM" 2>/dev/null || break
  i=$((i + 1)); sleep 0.25
done
[ "$code" = "200" ] || { echo "ABORT: no health 200 (last $code)"; tail -n 15 "$OUT/b01_lane_server.log"; exit 4; }
lsof -nP -tiTCP:$PORT -sTCP:LISTEN | head -n 1 > .locks/core-server.lane1.pid
echo "-- up after $i quarter seconds; listener PID $(cat .locks/core-server.lane1.pid)"
grep -o '"port":[0-9]*,"database":"[a-z_0-9]*","demoControls":[a-z]*' "$OUT/b01_lane_server.log" | head -n 1
curl -s http://127.0.0.1:$PORT/api/v1/health; echo
LINES=$(grep -c . "$OUT/b01_lane_server.log")

say "2. device session: phone, Metro $METRO, device tcp:4420 -> host tcp:$PORT (cap 600 s)"
METRO_PORT=$METRO DEMS_CORE_HOST_PORT=$PORT DEVICE_SESSION_MAX_SECONDS=600 LOCK_WAIT=300 \
  apps/mobile/scripts/device-session.sh phone "/bin/sh '$ROOT/.cache/w86/device_w86.sh' '$ROOT/$OUT'" > "$OUT/b02_device_session.log" 2>&1
DEVICE_EXIT=$?; echo "device session exit=$DEVICE_EXIT"
grep -E '^-- |^   ' "$OUT/b02_device_session.log" | head -60

say "3. lane backend requests during the session (method, route, status, principal)"
tail -n +$((LINES + 1)) "$OUT/b01_lane_server.log" > "$OUT/b01_lane_server_device_part.log"
grep '"msg":"request completed"' "$OUT/b01_lane_server_device_part.log" | sed -E 's/.*"method":"([^"]*)","route":"([^"]*)","status":([0-9]*).*"principal":([^,}]*).*/\1 \2 \3 principal=\4/' | sed -E 's/principal="?(operator|device|null)[^ ]*/principal=\1/' | sort | uniq -c | sed 's/^/   /'
echo "-- demo operations (must be 0): $(grep '"msg":"request completed"' "$OUT/b01_lane_server_device_part.log" | grep -c '/api/v1/demo/')"
echo "-- credential-shaped values in the lane server log (must be 0): $(grep -c -E 'ses_[A-Za-z0-9]|dvc_[A-Za-z0-9]|bio_[A-Za-z0-9]|[Bb]earer [A-Za-z0-9]' "$OUT/b01_lane_server.log")"

say "4. stop the lane backend; nothing left"
stop_backend
lsof -nP -iTCP:$PORT -iTCP:$METRO -sTCP:LISTEN; echo "lsof exit $? (1: none)"
adb -s emulator-5554 reverse --list; echo "reverse rules listed above (none expected)"
apps/mobile/scripts/emulator-user-lock.sh status > "$OUT/b03_lock_status_after.txt" 2>&1; echo "lock status exit $?"
diff "$OUT/b00_lock_status_before.txt" "$OUT/b03_lock_status_after.txt" && echo "lock status identical before and after"
say "end"; exit "$DEVICE_EXIT"
