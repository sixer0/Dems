// W81 mutation runner (W0's / W64's pattern; rule M15: a mutation never changes a file of the
// workspace tree). The app's own Vite / Vitest configuration, plus ONE plugin that replaces a text in
// ONE module while it is transformed, in memory. Which mutation: the environment variable W81_MUT (an
// id of mutations.json); unset: no mutation (the baseline).
import { readFileSync, writeFileSync } from 'node:fs'
import { fileURLToPath, URL } from 'node:url'
import { mergeConfig } from 'vite'
import appConfig from '../../../apps/core-web/vite.config.ts'

const here = fileURLToPath(new URL('./', import.meta.url))
const web = fileURLToPath(new URL('../../../apps/core-web/', import.meta.url))
const table = JSON.parse(readFileSync(`${here}mutations.json`, 'utf8'))
const id = process.env.W81_MUT ?? ''
const mutation = id === '' ? null : table.find((entry: { id: string }) => entry.id === id)
if (id !== '' && !mutation) throw new Error(`W81 mutation "${id}" is not in mutations.json`)

function mutationPlugin() {
  return {
    name: 'w81-mutation',
    enforce: 'pre' as const,
    transform(code: string, moduleId: string) {
      if (!mutation) return null
      const file = moduleId.split('?')[0] ?? ''
      if (!file.endsWith(`/src/${mutation.file}`)) return null
      const count = code.split(mutation.find).length - 1
      if (count !== 1) throw new Error(`W81 mutation ${mutation.id}: the text to replace occurs ${count} times in ${mutation.file}`)
      writeFileSync(`${process.env.W81_APPLIED_DIR}/${mutation.id}`, 'applied\n')
      const replaced = code.replace(mutation.find, () => mutation.replace)
      return { code: (mutation.prelude ?? '') + replaced, map: null }
    },
  }
}

const base = typeof appConfig === 'function' ? await appConfig({ command: 'serve', mode: 'test', isPreview: false, isSsrBuild: false }) : appConfig
const { plugins = [], ...rest } = base
export default {
  ...mergeConfig(rest, { root: web, cacheDir: `${here}vite-cache`, test: { root: web } }),
  plugins: [mutationPlugin(), ...plugins],
}
