// Scans files for the seeded demo password and for token-shaped values. Prints counts only.
import { readFileSync } from 'node:fs'
import { DEMO_OPERATOR_PASSWORD } from '../../packages/shared/src/sim/dataset.ts'
const MARKER = ['MARKER', 'W55', 'pw', '7f3a9c'].join('-')
let bad = 0
for (const file of process.argv.slice(2)) {
  const buffer = readFileSync(file)
  const text = buffer.toString('latin1')
  const demo = text.split(DEMO_OPERATOR_PASSWORD).length - 1
  const marker = text.split(MARKER).length - 1
  const token = (text.match(/\b(ses|dvc|bio)_[A-Za-z0-9_-]{8,}/g) ?? []).length
  if (demo + marker + token > 0) bad += 1
  console.log(`${demo + marker + token === 0 ? 'clean ' : 'FOUND '} demo-password=${demo} marker=${marker} token-shaped=${token}  ${file}`)
}
console.log(bad === 0 ? 'RESULT: no file holds a secret' : `RESULT: ${bad} file(s) hold a secret`)
process.exit(bad === 0 ? 0 : 1)
