#!/bin/sh
# W66 device check (lane 1, the phone). Modelled on W60's session F (implementation/assets/60_run_w60f.sh).
# The lane database is NOT reset: the phone is paired with dems_lane_1 as DEV-0001 (D143) and stays so.
# Run INSIDE the lane lock: /usr/bin/lockf -k -t 600 .locks/core-server.lane1.lock /bin/sh .cache/w66/run_w66.sh
set -u
ROOT=/Users/gsp-budi/Documents/DEMS; H="$ROOT/.cache/w66"; OUT="$H/out"; PORT=4421; METRO=${METRO_PORT:-8083}
RUN="w66$(date +%H%M%S)"; mkdir -p "$OUT"
say() { printf '\n== %s %s\n' "$(date '+%H:%M:%S')" "$*"; }
listener() { lsof -nP -iTCP:"$1" -sTCP:LISTEN -t 2>/dev/null | head -n 1; }
counts() { (cd "$ROOT" && npm run db:status -w @dems/core-server -- --database dems_lane_1 2>&1) | grep -E "^ +(paired_devices|device_credentials|sessions|audit_log) +[0-9]+|db:status:" | sed 's/^ */   /'; }

say "0. before (no reset): ports free?; dems_lane_1 as it is"
[ -z "$(listener $PORT)" ] || { echo "port $PORT busy"; exit 70; }
[ -z "$(listener "$METRO")" ] || { echo "Metro $METRO busy"; exit 70; }
counts | tee "$OUT/counts_before.txt"

say "1. start the lane backend (demo flag off)"
: >"$OUT/server.log"
(cd "$ROOT" && DEMS_CORE_PORT=$PORT exec npm run start:server) >"$OUT/server.log" 2>&1 &
NPM_PID=$!; i=0
until [ -n "$(listener $PORT)" ] || ! kill -0 "$NPM_PID" 2>/dev/null || [ "$i" -ge 160 ]; do i=$((i + 1)); sleep 0.25; done
SERVER_PID=$(listener $PORT); [ -n "$SERVER_PID" ] || { echo "backend did not boot"; tail -n 20 "$OUT/server.log"; exit 70; }
echo "$SERVER_PID" >"$ROOT/.locks/core-server.lane1.pid"
echo "npm PID $NPM_PID, listener PID $SERVER_PID"
grep -E 'configuration loaded' "$OUT/server.log" | grep -o '"port":[0-9]*,"database":"[a-z_0-9]*","demoControls":[a-z]*' | sed 's/^/   /'
curl -s "http://127.0.0.1:$PORT/api/v1/health"; echo
HOST_LINES=$(grep -c . "$OUT/server.log")

say "2. device session: phone, Metro $METRO, device tcp:4420 -> host tcp:$PORT"
METRO_PORT=$METRO DEMS_CORE_HOST_PORT=$PORT DEVICE_SESSION_MAX_SECONDS=540 LOCK_WAIT=${LOCK_WAIT:-600} \
  "$ROOT/apps/mobile/scripts/device-session.sh" phone "/bin/sh '$H/device_w66.sh' '$RUN' '$OUT' '$OUT/server.log' '$PORT'"
DEVICE_EXIT=$?; echo "device session exit=$DEVICE_EXIT"; echo "$DEVICE_EXIT" >"$OUT/device_exit.txt"

say "3. backend request log during the session"
tail -n +$((HOST_LINES + 1)) "$OUT/server.log" >"$OUT/server_device_part.log"
grep '"msg":"request completed"' "$OUT/server_device_part.log" | sed -E 's/.*"method":"([^"]*)","route":"([^"]*)","status":([0-9]*).*"principal":([^,}]*).*/\1 \2 \3 principal=\4/' | sed -E 's/principal="?(operator|device|null)[^ ]*/principal=\1/' | sort | uniq -c | sed 's/^/   /'
echo "-- socket lines: $(grep -c '"msg":"socket opened"' "$OUT/server_device_part.log") opened, $(grep -c '"msg":"socket closed"' "$OUT/server_device_part.log") closed"
echo "-- bearer/cookie/credential lines in the server log this run (must be 0): $(grep -c -E 'dvc_|ses_|bio_|[Bb]earer |[Cc]ookie' "$OUT/server.log")"

say "4. stop the backend and check nothing is left"
kill "$SERVER_PID" 2>/dev/null; i=0; while [ -n "$(listener $PORT)" ] && [ "$i" -lt 40 ]; do i=$((i + 1)); sleep 0.25; done
kill "$NPM_PID" 2>/dev/null; wait "$NPM_PID" 2>/dev/null
grep -E 'shutdown complete' "$OUT/server.log" | tail -n 1 | sed 's/^/   /'
: >"$ROOT/.locks/core-server.lane1.pid"
lsof -nP -iTCP:$PORT -iTCP:"$METRO" -sTCP:LISTEN; echo "lsof exit $? (1 = none)"
counts | tee "$OUT/counts_after.txt"
say "end of run $RUN"; exit "$DEVICE_EXIT"
