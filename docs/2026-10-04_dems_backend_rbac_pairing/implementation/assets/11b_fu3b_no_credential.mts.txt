import { readFileSync, readdirSync, statSync } from 'node:fs';
import { join } from 'node:path';

const REPO = '/Users/gsp-budi/Documents/DEMS';
const FILES_TO_READ = [
  `${REPO}/.data/pg.owner.env`,
  `${REPO}/.data/pg.env`,
];

function valueOf(file: string, key: string): string {
  const line = readFileSync(file, 'utf8')
    .split('\n')
    .find((text) => text.startsWith(`${key}=`));
  if (line === undefined) throw new Error(`no ${key} in ${file}`);
  return line.slice(key.length + 1);
}

const secrets = [valueOf(FILES_TO_READ[0] ?? '', 'DEMS_PG_OWNER_PASSWORD'), valueOf(FILES_TO_READ[1] ?? '', 'DEMS_PG_APP_PASSWORD')];
if (secrets.some((secret) => secret.length < 8)) throw new Error('a secret looks too short; the read is wrong');

function walk(directory: string, out: string[]): string[] {
  for (const entry of readdirSync(directory)) {
    const full = join(directory, entry);
    const stats = statSync(full);
    if (stats.isDirectory()) walk(full, out);
    else if (stats.isFile()) out.push(full);
  }
  return out;
}

const roots = [
  `${REPO}/apps/core-server/test/helpers/db`,
  `${REPO}/docs/2026-10-04_dems_backend_rbac_pairing/implementation/11b_db_helper_forced_drop_retry.md`,
  `${REPO}/docs/2026-10-04_dems_backend_rbac_pairing/delegation_progress/database_specialist_local_step_w11b3.md`,
  `${REPO}/.cache/rollback/W11b_followup3b.pre`,
  '/var/folders/rn/7bm_xjb14jxcbp36qw981phr0000gn/T/kilo/w11b3-tree',
];
const files: string[] = [];
for (const root of roots) {
  const stats = statSync(root, { throwIfNoEntry: false });
  if (stats === undefined) {
    console.log(`MISSING root: ${root}`);
    continue;
  }
  if (stats.isDirectory()) walk(root, files);
  else files.push(root);
}
for (const name of readdirSync(`${REPO}/docs/2026-10-04_dems_backend_rbac_pairing/implementation/assets`)) {
  if (name.startsWith('11b')) files.push(`${REPO}/docs/2026-10-04_dems_backend_rbac_pairing/implementation/assets/${name}`);
}

let checked = 0;
const hits: string[] = [];
for (const file of [...new Set(files)]) {
  const text = readFileSync(file, 'utf8');
  checked += 1;
  if (secrets.some((secret) => text.includes(secret))) hits.push(file);
}
console.log(`files compared in memory with the two password values: ${checked}`);
console.log(`files that contain one: ${hits.length}`);
for (const hit of hits) console.log(`  HIT ${hit}`);
