#!/bin/sh
# Runs mutations against the row's own tests (unit, then db). Prints one line each.
# Usage: run.sh <scratch dir> [id ...]   (no id: every mutation of mutations.json)
# Vitest writes into a file that is read afterwards: no pipe is closed under a running test run.
# No directory is changed: Vitest gets the workspace as its root. The database tests run only
# when the unit tests did not detect the mutation.
M="$1"; shift
ROOT=/Users/gsp-budi/Documents/DEMS
WS="$ROOT/apps/core-server"
if [ "$#" -eq 0 ]; then set -- $(python3 -c "import json; print(' '.join(x['id'] for x in json.load(open('$M/mutations.json'))))"); fi
summary() { grep -E "^ +Tests |the text to replace is not in|no mutation " "$1" | sed -n '1p' | sed -e 's/^ *//' -e 's/  */ /g'; }
for id in "$@"; do
  note=$(python3 -c "import json,sys; print([x['note'] for x in json.load(open('$M/mutations.json')) if x['id']=='$id'][0])")
  W39_MUTATION="$id" "$ROOT/node_modules/.bin/vitest" run --config "$M/vitest.config.mjs" --root "$WS" --maxWorkers=2 --project unit src/modules/security-audit > "$M/out.unit.txt" 2>&1
  u=$(summary "$M/out.unit.txt")
  case "$u" in
    *failed*) d="not run (detected by the unit tests)" ;;
    *)
      W39_MUTATION="$id" DEMS_TEST_STEP=61 "$ROOT/node_modules/.bin/vitest" run --config "$M/vitest.config.mjs" --root "$WS" --maxWorkers=2 --project db src/modules/security-audit > "$M/out.db.txt" 2>&1
      d=$(summary "$M/out.db.txt") ;;
  esac
  echo "$id | $note | unit: ${u:-NO-SUMMARY} | db: ${d:-NO-SUMMARY}"
done
