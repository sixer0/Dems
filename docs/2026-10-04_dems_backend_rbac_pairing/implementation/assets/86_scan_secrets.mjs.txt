// W86: count-only scan. Reads the three PostgreSQL credential files IN PROCESS, takes their secret
// values (the superuser password file's single line and the two PASSWORD fields), and counts how
// often each occurs as a fixed string in the files named on the command line. Prints counts only:
// never a value, a length, a prefix or a hash of a value.
// Usage: node .cache/w86/scan_secrets.mjs <file>...
import fs from 'node:fs'
import path from 'node:path'

const ROOT = '/Users/gsp-budi/Documents/DEMS'
const data = path.join(ROOT, '.data')
const field = (file, key) => {
  const line = fs.readFileSync(path.join(data, file), 'utf8').split('\n').find((l) => l.startsWith(`${key}=`))
  return line === undefined ? '' : line.slice(key.length + 1).trim()
}
const secrets = {
  'pg.pwfile (dems_admin)': fs.readFileSync(path.join(data, 'pg.pwfile'), 'utf8').trim(),
  'pg.owner.env DEMS_PG_OWNER_PASSWORD': field('pg.owner.env', 'DEMS_PG_OWNER_PASSWORD'),
  'pg.env DEMS_PG_APP_PASSWORD': field('pg.env', 'DEMS_PG_APP_PASSWORD'),
}
for (const [name, value] of Object.entries(secrets)) {
  if (value.length < 16) {
    console.log(`ABORT: ${name} could not be read as a value of the expected size`)
    process.exit(2)
  }
}
let total = 0
for (const target of process.argv.slice(2)) {
  let text
  try {
    text = fs.readFileSync(target, 'latin1')
  } catch {
    console.log(`unreadable: ${target}`)
    continue
  }
  const counts = Object.entries(secrets).map(([name, value]) => [name, text.split(value).length - 1])
  const sum = counts.reduce((s, [, n]) => s + n, 0)
  total += sum
  console.log(`${sum}  ${path.relative(ROOT, path.resolve(target))}  (${counts.map(([n, c]) => `${n}: ${c}`).join('; ')})`)
}
console.log(`TOTAL matches: ${total}`)
